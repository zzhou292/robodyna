cmake_minimum_required(VERSION 3.22)
project(RoboDynaSourceSpinReplay LANGUAGES CXX Fortran)
enable_testing()
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
set(ROBO_DYNA_TL_ROOT "" CACHE PATH "TL checkout with independently qualified native layered recurrence")
if(NOT EXISTS "${ROBO_DYNA_TL_ROOT}/lib_utest/qualification/shell_layered_j2/LayeredNativeRecurrence.cmake")
  message(FATAL_ERROR "Explicit native layered reference checkout required")
endif()
set(TL_SHELL_LAYERED_J2_ENABLE_CUDA OFF CACHE BOOL "Host-only replay qualification" FORCE)
add_subdirectory("${ROBO_DYNA_TL_ROOT}/lib_utest/qualification/shell_layered_j2" native-layered)
include("${CMAKE_CURRENT_LIST_DIR}/SourceAssemblySpin.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../../output/source_assembly/QephSpinFields.cmake")
add_executable(robo_dyna_source_assembly_spin_check tests/AnalysisTest.cpp)
target_link_libraries(robo_dyna_source_assembly_spin_check PRIVATE robo_dyna_source_assembly_spin
  robo_dyna_source_assembly_spin_fields GTest::gtest_main)
set(ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY "" CACHE FILEPATH "Pinned original six-part inventory")
set(ROBO_DYNA_SOURCE_ASSEMBLY_SPIN_TRACE "" CACHE FILEPATH "Optional completed actual spin trace")
add_test(NAME source_assembly_spin COMMAND robo_dyna_source_assembly_spin_check)
set_tests_properties(source_assembly_spin PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
  ENVIRONMENT "ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY};ROBO_DYNA_SOURCE_ASSEMBLY_SPIN_TRACE=${ROBO_DYNA_SOURCE_ASSEMBLY_SPIN_TRACE};OMP_NUM_THREADS=1")
