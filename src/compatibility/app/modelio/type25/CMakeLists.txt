cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaVehicleType25Source LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/VehicleType25Source.cmake")
find_package(GTest REQUIRED)
add_executable(robo_dyna_vehicle_type25_fields_check tests/FieldsTest.cpp)
target_link_libraries(robo_dyna_vehicle_type25_fields_check PRIVATE robo_dyna_vehicle_type25_source GTest::gtest_main)
target_compile_options(robo_dyna_vehicle_type25_fields_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME vehicle_type25_fields COMMAND robo_dyna_vehicle_type25_fields_check)
option(ROBO_DYNA_VEHICLE_TYPE25_ACTUAL_TESTS "Enable complete original mapped TYPE25 source gate" OFF)
if(ROBO_DYNA_VEHICLE_TYPE25_ACTUAL_TESTS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(name CANONICAL SCOPE DECLARATIONS TYPE13_DECLARATION)
    set(ROBO_DYNA_PHYSICAL_${name} "" CACHE PATH "Explicit original ${name} fixture")
    if(NOT EXISTS "${ROBO_DYNA_PHYSICAL_${name}}")
      message(FATAL_ERROR "Vehicle TYPE25 source requires ${name}")
    endif()
  endforeach()
  add_executable(robo_dyna_vehicle_type25_source_check tests/ActualSourceTest.cpp)
  target_link_libraries(robo_dyna_vehicle_type25_source_check PRIVATE robo_dyna_vehicle_type25_source GTest::gtest_main)
  target_compile_options(robo_dyna_vehicle_type25_source_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME vehicle_type25_source COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../../output/full_shell/static_bundle/tests/actual_source_fixture.py"
    "${ROBO_DYNA_PHYSICAL_CANONICAL}" "${ROBO_DYNA_PHYSICAL_SCOPE}"
    "$<TARGET_FILE:robo_dyna_vehicle_type25_source_check>" "VehicleType25Original.*")
  set_tests_properties(vehicle_type25_source PROPERTIES ENVIRONMENT
    "ROBO_VEHICLE_DECLARATIONS=${ROBO_DYNA_PHYSICAL_DECLARATIONS};ROBO_DYNA_TYPE13_DECLARATION=${ROBO_DYNA_PHYSICAL_TYPE13_DECLARATION}")
  set(ROBO_DYNA_SOURCE_BRACKET_INVENTORY "" CACHE FILEPATH "Explicit original seven-part regression fixture")
  if(EXISTS "${ROBO_DYNA_SOURCE_BRACKET_INVENTORY}")
    set(ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY "" CACHE FILEPATH "Explicit original six-part regression fixture")
    if(NOT EXISTS "${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY}")
      message(FATAL_ERROR "Component spotweld regression also requires its six-part negative fixture")
    endif()
    add_executable(robo_dyna_type25_component_regression_check
      "${CMAKE_CURRENT_LIST_DIR}/../source_assembly/tests/SourceBracketSpotweldTest.cpp")
    target_link_libraries(robo_dyna_type25_component_regression_check PRIVATE
      robo_dyna_source_assembly_spotweld_input GTest::gtest_main)
    add_test(NAME vehicle_type25_component_regression COMMAND robo_dyna_type25_component_regression_check)
    set_tests_properties(vehicle_type25_component_regression PROPERTIES ENVIRONMENT
      "ROBO_DYNA_SOURCE_BRACKET_INVENTORY=${ROBO_DYNA_SOURCE_BRACKET_INVENTORY};ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY}")
  endif()
endif()
get_property(type25_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${type25_tests} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE PROCESSORS 1)
