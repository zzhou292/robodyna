cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaRigidPointMass LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/Source.cmake")
find_package(GTest REQUIRED)
add_executable(robo_dyna_rigid_point_mass_fields_check tests/FieldsTest.cpp)
target_link_libraries(robo_dyna_rigid_point_mass_fields_check PRIVATE
  robo_dyna_rigid_point_mass_source GTest::gtest_main)
add_test(NAME rigid_point_mass_fields COMMAND robo_dyna_rigid_point_mass_fields_check)
option(ROBO_DYNA_RIGID_POINT_MASS_ACTUAL_TESTS "Enable the original54-point source gate" OFF)
if(ROBO_DYNA_RIGID_POINT_MASS_ACTUAL_TESTS)
  include("${CMAKE_CURRENT_LIST_DIR}/../../vehicle_sections/VehicleSectionResolution.cmake")
  include("${ROBO_DYNA_TL_ROOT}/lib_src/assembly/ElementMassContributions.cmake")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(name CANONICAL SCOPE DECLARATIONS RESOLUTION GLASS_RESOLUTION)
    set(ROBO_DYNA_VEHICLE_${name} "" CACHE PATH "Explicit original vehicle ${name} fixture")
    if(NOT EXISTS "${ROBO_DYNA_VEHICLE_${name}}")
      message(FATAL_ERROR "Original rigid point-mass gate requires ${name}")
    endif()
  endforeach()
  set(ROBO_DYNA_VEHICLE_GLASS_SHA256 "" CACHE STRING "Authenticated original glass sidecar SHA256")
  string(LENGTH "${ROBO_DYNA_VEHICLE_GLASS_SHA256}" glass_sha_length)
  if(NOT glass_sha_length EQUAL 64)
    message(FATAL_ERROR "Original rigid point-mass gate requires the glass sidecar SHA256")
  endif()
  add_executable(robo_dyna_rigid_point_mass_source_check tests/ActualSourceTest.cpp)
  target_link_libraries(robo_dyna_rigid_point_mass_source_check PRIVATE
    robo_dyna_rigid_point_mass_source robo_dyna_vehicle_sections tl_element_mass_contributions GTest::gtest_main)
  add_test(NAME rigid_point_mass_source COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../../../output/full_shell/static_bundle/tests/actual_source_fixture.py"
    "${ROBO_DYNA_VEHICLE_CANONICAL}" "${ROBO_DYNA_VEHICLE_SCOPE}"
    "$<TARGET_FILE:robo_dyna_rigid_point_mass_source_check>" "RigidPointMassSource.*")
  set_tests_properties(rigid_point_mass_source PROPERTIES
    ENVIRONMENT "ROBO_VEHICLE_DECLARATIONS=${ROBO_DYNA_VEHICLE_DECLARATIONS};ROBO_VEHICLE_RESOLUTION=${ROBO_DYNA_VEHICLE_RESOLUTION};ROBO_VEHICLE_GLASS_RESOLUTION=${ROBO_DYNA_VEHICLE_GLASS_RESOLUTION};ROBO_VEHICLE_GLASS_SHA256=${ROBO_DYNA_VEHICLE_GLASS_SHA256}")
endif()
get_property(point_mass_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${point_mass_tests} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE PROCESSORS 1)
