cmake_minimum_required(VERSION 3.22)
project(RoboDynaSourceAssemblyPilotOptions LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../PilotOptions.cmake")
enable_testing()
find_package(GTest REQUIRED)
add_executable(robo_dyna_source_assembly_pilot_options_check "${CMAKE_CURRENT_LIST_DIR}/../tests/PilotOptionsTest.cpp")
target_link_libraries(robo_dyna_source_assembly_pilot_options_check PRIVATE robo_dyna_source_assembly_pilot_options GTest::gtest_main)
target_compile_options(robo_dyna_source_assembly_pilot_options_check PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME source_assembly_pilot_options_host COMMAND robo_dyna_source_assembly_pilot_options_check)
set_tests_properties(source_assembly_pilot_options_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
