cmake_minimum_required(VERSION 3.22)
project(TLWeightedSurface LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/weighted_surface/WeightedSurface.cmake")
add_executable(weighted_surface_host MappingTest.cpp)
target_link_libraries(weighted_surface_host PRIVATE tl_weighted_surface GTest::gtest_main)
target_compile_options(weighted_surface_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME weighted_surface_host COMMAND weighted_surface_host)
set_tests_properties(weighted_surface_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
