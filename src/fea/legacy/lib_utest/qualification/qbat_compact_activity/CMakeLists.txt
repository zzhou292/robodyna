cmake_minimum_required(VERSION 3.22)
project(TLQbatCompactActivity LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(qbat_compact_activity_host HostTest.cpp "${TL_ROOT}/lib_src/elements/qbat/QbatBatchArena.cpp")
target_include_directories(qbat_compact_activity_host PRIVATE "${TL_ROOT}")
target_link_libraries(qbat_compact_activity_host PRIVATE GTest::gtest_main CUDA::toolkit)
target_compile_features(qbat_compact_activity_host PRIVATE cxx_std_17)
target_compile_options(qbat_compact_activity_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qbat_compact_activity_host COMMAND qbat_compact_activity_host)
add_test(NAME qbat_compact_activity_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
option(QBAT_COMPACT_ACTIVITY_CUDA "Fresh complete GPU validator comparison" OFF)
if(QBAT_COMPACT_ACTIVITY_CUDA)
  enable_language(CUDA)
  add_executable(qbat_compact_activity_cuda CudaTest.cu
    "${TL_ROOT}/lib_src/elements/qbat/QbatBatchArena.cpp"
    "${TL_ROOT}/lib_src/elements/qbat/activity/Kernels.cu")
  target_include_directories(qbat_compact_activity_cuda PRIVATE "${TL_ROOT}")
  target_link_libraries(qbat_compact_activity_cuda PRIVATE GTest::gtest_main CUDA::cudart)
  set_target_properties(qbat_compact_activity_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(qbat_compact_activity_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  add_test(NAME qbat_compact_activity_cuda COMMAND qbat_compact_activity_cuda)
endif()
option(QBAT_COMPACT_ACTIVITY_OWNER "Actual public activity authentication, corruption, alias and CUDA error tests" OFF)
if(QBAT_COMPACT_ACTIVITY_OWNER)
  enable_language(CUDA)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" nodal-owner)
  add_subdirectory("${TL_ROOT}/lib_src/elements/qeph" qeph-values)
  add_subdirectory("${TL_ROOT}/lib_src/elements/t3" t3-values)
  include("${TL_ROOT}/lib_src/elements/qeph/QephBatch.cmake")
  include("${TL_ROOT}/lib_src/elements/t3/T3Batch.cmake")
  include("${TL_ROOT}/lib_src/elements/ShellBatchPublication.cmake")
  add_executable(qbat_compact_activity_owner OwnerTest.cu OwnerProbe.cu
    ../qbat_resident/ResidentStartup.cu ../qbat_resident/ResidentFrames.cu)
  target_link_libraries(qbat_compact_activity_owner PRIVATE tl_shell_batch_publication CUDA::cudart GTest::gtest_main)
  target_link_options(qbat_compact_activity_owner PRIVATE
    "LINKER:--wrap=cudaMemcpyAsync" "LINKER:--wrap=cudaMemsetAsync"
    "LINKER:--wrap=cudaGetLastError" "LINKER:--wrap=cudaStreamSynchronize")
  set_target_properties(qbat_compact_activity_owner PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(qbat_compact_activity_owner PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  add_test(NAME qbat_compact_activity_owner COMMAND qbat_compact_activity_owner)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
