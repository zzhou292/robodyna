cmake_minimum_required(VERSION 3.22)
project(TLQephCandidateDiagnostics LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(qeph_candidate_diagnostics_host HostTest.cpp)
target_include_directories(qeph_candidate_diagnostics_host PRIVATE "${tl_root}")
target_link_libraries(qeph_candidate_diagnostics_host PRIVATE GTest::gtest_main CUDA::toolkit)
target_compile_features(qeph_candidate_diagnostics_host PRIVATE cxx_std_17)
target_compile_options(qeph_candidate_diagnostics_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qeph_candidate_diagnostics_host COMMAND qeph_candidate_diagnostics_host)
set_tests_properties(qeph_candidate_diagnostics_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(QEPH_DIAGNOSTICS_CUDA "Mapped diagnostics versus frozen serial candidate finish" OFF)
if(QEPH_DIAGNOSTICS_CUDA)
  enable_language(CUDA)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  add_subdirectory("${tl_root}/lib_src/elements/qeph" qeph-values)
  include("${tl_root}/lib_src/elements/qeph/QephBatch.cmake")
  add_executable(qeph_candidate_diagnostics_cuda CudaTest.cu)
  target_link_libraries(qeph_candidate_diagnostics_cuda PRIVATE tl_qeph_batch GTest::gtest_main CUDA::cudart)
  set_target_properties(qeph_candidate_diagnostics_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(qeph_candidate_diagnostics_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME qeph_candidate_diagnostics_cuda COMMAND qeph_candidate_diagnostics_cuda)
  set_tests_properties(qeph_candidate_diagnostics_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME qeph_candidate_diagnostics_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(qeph_candidate_diagnostics_identity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
