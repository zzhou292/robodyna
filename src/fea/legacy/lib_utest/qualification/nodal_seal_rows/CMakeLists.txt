cmake_minimum_required(VERSION 3.22)
project(TLNodalSealRows LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/prepare_frozen.py"
  "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_executable(nodal_seal_rows_host HostTest.cpp)
target_compile_features(nodal_seal_rows_host PRIVATE cxx_std_17)
target_include_directories(nodal_seal_rows_host PRIVATE "${tl_root}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(nodal_seal_rows_host PRIVATE GTest::gtest_main)
target_compile_options(nodal_seal_rows_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME nodal_seal_rows_host COMMAND nodal_seal_rows_host)
set_tests_properties(nodal_seal_rows_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(NODAL_SEAL_ROWS_CUDA "Frozen complete seal and actual owner CUDA gates" OFF)
if(NODAL_SEAL_ROWS_CUDA)
  find_package(Eigen3 REQUIRED)
  add_executable(nodal_seal_rows_affected_host "${tl_root}/lib_utest/utest_step_stability.cc")
  target_compile_features(nodal_seal_rows_affected_host PRIVATE cxx_std_17)
  target_include_directories(nodal_seal_rows_affected_host PRIVATE "${tl_root}")
  target_link_libraries(nodal_seal_rows_affected_host PRIVATE Eigen3::Eigen GTest::gtest_main)
  target_compile_options(nodal_seal_rows_affected_host PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME nodal_seal_rows_affected_host COMMAND nodal_seal_rows_affected_host)
  set_tests_properties(nodal_seal_rows_affected_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
  enable_language(CUDA)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  add_executable(nodal_seal_rows_cuda CallerTest.cu OwnerTest.cu Frozen.cu
    ../nodal_vehicle/VehicleOwnerFixture.cu ../nodal_vehicle/SealOrderTest.cu
    "${tl_root}/lib_utest/utest_step_stability_cuda.cu")
  target_include_directories(nodal_seal_rows_cuda PRIVATE "${tl_root}"
    "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(nodal_seal_rows_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main)
  set_target_properties(nodal_seal_rows_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(nodal_seal_rows_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>)
  add_test(NAME nodal_seal_rows_cuda COMMAND nodal_seal_rows_cuda)
  set_tests_properties(nodal_seal_rows_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
add_test(NAME nodal_seal_rows_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(nodal_seal_rows_identity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
