cmake_minimum_required(VERSION 3.18)
project(TLNodalMassQualification LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../../../lib_src/assembly/NodalMassBinding.cmake")
add_executable(nodal_mass_binding_check NodalMassBindingTest.cpp NodalMassCapacityTest.cpp)
target_link_libraries(nodal_mass_binding_check PRIVATE tl_nodal_mass_binding GTest::gtest_main)
target_compile_options(nodal_mass_binding_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME nodal_mass_binding COMMAND nodal_mass_binding_check)
set_tests_properties(nodal_mass_binding PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
option(TL_NODAL_MASS_VEHICLE_CHECKS "Build source-sized combined nodal mass host checks (root-scheduled)" OFF)
if(TL_NODAL_MASS_VEHICLE_CHECKS)
  find_package(CUDAToolkit REQUIRED)
  add_executable(nodal_mass_vehicle_check NodalMassVehicleTest.cpp)
  target_link_libraries(nodal_mass_vehicle_check PRIVATE tl_nodal_mass_binding GTest::gtest_main)
  target_include_directories(nodal_mass_vehicle_check PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
  target_compile_options(nodal_mass_vehicle_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME nodal_mass_vehicle COMMAND nodal_mass_vehicle_check)
  set_tests_properties(nodal_mass_vehicle PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 600)
endif()
option(TL_NODAL_MASS_ENABLE_CUDA "Build actual joined-owner admission checks" OFF)
if(TL_NODAL_MASS_ENABLE_CUDA)
  enable_language(CUDA)
  get_filename_component(tl_mass_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
  add_subdirectory("${tl_mass_root}/lib_src/solvers" nodal-owner)
  include("${tl_mass_root}/lib_src/elements/qeph/QephBatch.cmake")
  include("${tl_mass_root}/lib_src/elements/t3/T3Batch.cmake")
  include("${tl_mass_root}/lib_src/elements/ShellBatchPublication.cmake")
  add_executable(nodal_mass_joined_check NodalMassJoinedTest.cu)
  target_link_libraries(nodal_mass_joined_check PRIVATE tl_nodal_mass_binding
    tl_shell_batch_publication GTest::gtest_main)
  set_target_properties(nodal_mass_joined_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(nodal_mass_joined_check PRIVATE
    --fmad=false --prec-div=true --prec-sqrt=true --ftz=false
    -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME nodal_mass_joined COMMAND nodal_mass_joined_check)
  set_tests_properties(nodal_mass_joined PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
  add_executable(connector_publication_check connector/ConnectorPublicationFixture.cu connector/ConnectorPublicationTest.cu)
  target_link_libraries(connector_publication_check PRIVATE tl_nodal_mass_binding tl_shell_batch_publication GTest::gtest_main)
  set_target_properties(connector_publication_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(connector_publication_check PRIVATE
    --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME connector_publication COMMAND connector_publication_check)
  set_tests_properties(connector_publication PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
