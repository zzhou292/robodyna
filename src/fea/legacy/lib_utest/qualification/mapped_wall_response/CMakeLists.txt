cmake_minimum_required(VERSION 3.22)
project(TLMappedWallResponse LANGUAGES CXX C)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(mapped_wall_response_host HostTest.cpp IncidenceTest.cpp "${TL_ROOT}/lib_src/collision/nodal_wall_mapped/Layout.cpp")
target_include_directories(mapped_wall_response_host PRIVATE "${TL_ROOT}" "${CUDAToolkit_INCLUDE_DIRS}")
target_compile_features(mapped_wall_response_host PRIVATE cxx_std_17)
target_compile_options(mapped_wall_response_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(mapped_wall_response_host PRIVATE GTest::gtest_main)
add_test(NAME mapped_wall_response_host COMMAND mapped_wall_response_host)
option(MAPPED_WALL_RESPONSE_CUDA "Complete frozen validator on actual CUDA" OFF)
option(MAPPED_WALL_RESPONSE_OWNER "Actual mapped owner failure/retry integration" OFF)
if(MAPPED_WALL_RESPONSE_CUDA OR MAPPED_WALL_RESPONSE_OWNER)
  enable_language(CUDA)
endif()
function(configure_cuda target)
  target_link_libraries(${target} PRIVATE Eigen3::Eigen)
  target_include_directories(${target} PRIVATE "${TL_ROOT}" "${CUDAToolkit_INCLUDE_DIRS}")
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(${target} PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
endfunction()
