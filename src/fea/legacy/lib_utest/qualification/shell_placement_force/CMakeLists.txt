cmake_minimum_required(VERSION 3.22)
project(ShellPlacementForce LANGUAGES CXX)
option(TL_PLACEMENT_FORCE_NATIVE "Build complete native placed Q/T force comparisons" OFF)
option(TL_PLACEMENT_FORCE_CUDA "Build device-owned placed Q/T force comparisons" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
include("${TL_ROOT}/lib_src/elements/ShellBatchBinding.cmake")
add_executable(shell_placement_force_values_check PlacementReferenceTest.cpp PlacementForceValueTest.cpp)
target_include_directories(shell_placement_force_values_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_placement_force_values_check PRIVATE cxx_std_17)
target_compile_options(shell_placement_force_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_placement_force_values_check PRIVATE tl_shell_batch_binding GTest::gtest_main)
add_test(NAME shell_placement_force_values COMMAND shell_placement_force_values_check)
set_tests_properties(shell_placement_force_values PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
if(TL_PLACEMENT_FORCE_NATIVE OR TL_PLACEMENT_FORCE_CUDA)
  include(Native.cmake)
endif()
