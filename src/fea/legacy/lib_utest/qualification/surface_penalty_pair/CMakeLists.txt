cmake_minimum_required(VERSION 3.22)
project(TLSurfacePenaltyPair LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Boost REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/penalty_pair/SurfacePenaltyPair.cmake")
add_library(surface_pair_exact STATIC ../surface_jacobian_majorant/ExactOracle.cpp)
target_link_libraries(surface_pair_exact PUBLIC tl_surface_penalty_pair)
target_include_directories(surface_pair_exact PRIVATE ${Boost_INCLUDE_DIRS})
target_compile_options(surface_pair_exact PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(surface_penalty_pair_host ValueTest.cpp)
target_link_libraries(surface_penalty_pair_host PRIVATE surface_pair_exact GTest::gtest_main)
target_compile_options(surface_penalty_pair_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME surface_penalty_pair_host COMMAND surface_penalty_pair_host)
set_tests_properties(surface_penalty_pair_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(SURFACE_PENALTY_PAIR_CUDA "Build actual CUDA pair algebra qualification" OFF)
if(SURFACE_PENALTY_PAIR_CUDA)
  enable_language(CUDA)
  add_executable(surface_penalty_pair_cuda CudaTest.cu)
  target_link_libraries(surface_penalty_pair_cuda PRIVATE surface_pair_exact GTest::gtest_main)
  set_target_properties(surface_penalty_pair_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(surface_penalty_pair_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>)
  add_test(NAME surface_penalty_pair_cuda COMMAND surface_penalty_pair_cuda)
  set_tests_properties(surface_penalty_pair_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
endif()
