cmake_minimum_required(VERSION 3.22)
project(TLQephQualification LANGUAGES CXX Fortran)
option(TL_QEPH_ENABLE_CUDA "Build and require the actual QEPH device qualification" ON)
option(TL_QEPH_ENABLE_BATCH "Build the bounded resident QEPH history/owner qualification" OFF)
option(TL_QEPH_ENABLE_COUPLED "Build the short coupled QEPH recurrence qualification" OFF)
option(TL_QEPH_ENABLE_FREE_RESPONSE_AUDIT "Build the native full-recurrence matrix audit" OFF)
option(TL_QEPH_ENABLE_WALL_RECURRENCE_SUPPORT "Build native moving-map and chronological Gram helper tests" OFF)
option(TL_QEPH_ENABLE_RESPONSE "Build the sustained CUDA response qualification" OFF)
option(TL_QEPH_ENABLE_WALL_COUPLED "Build the short QEPH and finite-wall transaction qualification" OFF)
option(TL_QEPH_ENABLE_WALL_INCOMING "Build the screened CUDA incoming wall-entry qualification" OFF)
option(TL_QEPH_ENABLE_WALL_RESPONSE "Build the screened broadside wall-response qualification" OFF)
if(TL_QEPH_ENABLE_BATCH AND NOT TL_QEPH_ENABLE_CUDA)
  message(FATAL_ERROR "The resident QEPH batch qualification requires CUDA")
endif()
if(TL_QEPH_ENABLE_WALL_RECURRENCE_SUPPORT AND NOT TL_QEPH_ENABLE_FREE_RESPONSE_AUDIT)
  message(FATAL_ERROR "Wall recurrence helpers require the native recurrence audit")
endif()
if(TL_QEPH_ENABLE_COUPLED AND NOT TL_QEPH_ENABLE_BATCH)
  message(FATAL_ERROR "The coupled QEPH qualification requires the resident batch")
endif()
if(TL_QEPH_ENABLE_WALL_COUPLED AND NOT TL_QEPH_ENABLE_COUPLED)
  message(FATAL_ERROR "The finite-wall QEPH qualification requires the coupled fixture")
endif()
if(TL_QEPH_ENABLE_WALL_INCOMING AND
   (NOT TL_QEPH_ENABLE_WALL_COUPLED OR NOT TL_QEPH_ENABLE_WALL_RECURRENCE_SUPPORT))
  message(FATAL_ERROR "Incoming wall entry requires wall coupling and the native wall recurrence screen")
endif()
if(TL_QEPH_ENABLE_RESPONSE AND
   (NOT TL_QEPH_ENABLE_COUPLED OR NOT TL_QEPH_ENABLE_FREE_RESPONSE_AUDIT))
  message(FATAL_ERROR "Response requires coupled QEPH and the native recurrence audit")
endif()
if(TL_QEPH_ENABLE_WALL_RESPONSE AND
   (NOT TL_QEPH_ENABLE_WALL_INCOMING OR NOT TL_QEPH_ENABLE_RESPONSE))
  message(FATAL_ERROR "Wall response requires screened incoming fixtures and reusable response utilities")
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_sources.py"
  RESULT_VARIABLE port_verify_status ERROR_VARIABLE port_verify_error)
if(NOT port_verify_status EQUAL 0)
  message(FATAL_ERROR "QEPH port verification failed: ${port_verify_error}")
endif()
add_custom_target(qeph_verify_port
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_sources.py" VERBATIM)
add_subdirectory("${TL_ROOT}/lib_src/elements/qeph" "tl-qeph")
# Only this explicit qualification harness links the native Fortran reference.
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/qeph" "native-qeph")
if(TL_QEPH_ENABLE_FREE_RESPONSE_AUDIT)
  add_subdirectory(free_response)
endif()
if(TL_QEPH_ENABLE_WALL_RECURRENCE_SUPPORT)
  add_subdirectory(wall_recurrence)
endif()

add_executable(qeph_startup_check QephStartupTest.cpp)
add_dependencies(qeph_startup_check qeph_verify_port)
target_link_libraries(qeph_startup_check PRIVATE tl_qeph qeph_q1_native GTest::gtest_main)
target_compile_options(qeph_startup_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qeph_startup_check COMMAND qeph_startup_check)
set_tests_properties(qeph_startup_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)

add_executable(qeph_kinematics_port_check QephKinematicsTest.cpp)
add_dependencies(qeph_kinematics_port_check qeph_verify_port)
target_link_libraries(qeph_kinematics_port_check PRIVATE tl_qeph qeph_q1_native GTest::gtest_main)
target_compile_options(qeph_kinematics_port_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qeph_kinematics_port_check COMMAND qeph_kinematics_port_check)
set_tests_properties(qeph_kinematics_port_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)

add_executable(qeph_force_port_check QephForceTest.cpp)
add_dependencies(qeph_force_port_check qeph_verify_port)
target_link_libraries(qeph_force_port_check PRIVATE tl_qeph qeph_q1_native GTest::gtest_main)
target_compile_options(qeph_force_port_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qeph_force_port_check COMMAND qeph_force_port_check)
set_tests_properties(qeph_force_port_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)

if(TL_QEPH_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(TL_QEPH_ENABLE_BATCH)
    add_subdirectory("${TL_ROOT}/lib_src/solvers" "tl-nodal-owner")
    include("${TL_ROOT}/lib_src/elements/qeph/QephBatch.cmake")
    add_subdirectory(batch)
    if(TL_QEPH_ENABLE_COUPLED)
      add_subdirectory(coupled)
    endif()
    if(TL_QEPH_ENABLE_RESPONSE)
      add_subdirectory(response)
    endif()
    if(TL_QEPH_ENABLE_WALL_COUPLED)
      add_subdirectory(wall_coupled)
    endif()
    if(TL_QEPH_ENABLE_WALL_INCOMING)
      add_subdirectory(wall_incoming)
    endif()
    if(TL_QEPH_ENABLE_WALL_RESPONSE)
      add_subdirectory(wall_response)
    endif()
  endif()
  add_executable(qeph_startup_cuda_check QephStartupCudaTest.cu)
  add_dependencies(qeph_startup_cuda_check qeph_verify_port)
  set_target_properties(qeph_startup_cuda_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_link_libraries(qeph_startup_cuda_check PRIVATE tl_qeph qeph_q1_native CUDA::cudart GTest::gtest_main)
  target_compile_options(qeph_startup_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME qeph_startup_cuda_check COMMAND qeph_startup_cuda_check)
  set_tests_properties(qeph_startup_cuda_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 45)

  add_executable(qeph_kinematics_port_cuda_check QephKinematicsCudaTest.cu)
  add_dependencies(qeph_kinematics_port_cuda_check qeph_verify_port)
  set_target_properties(qeph_kinematics_port_cuda_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_link_libraries(qeph_kinematics_port_cuda_check PRIVATE tl_qeph qeph_q1_native CUDA::cudart GTest::gtest_main)
  target_compile_options(qeph_kinematics_port_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME qeph_kinematics_port_cuda_check COMMAND qeph_kinematics_port_cuda_check)
  set_tests_properties(qeph_kinematics_port_cuda_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 45)

  add_executable(qeph_force_port_cuda_check QephForceCudaTest.cu)
  add_dependencies(qeph_force_port_cuda_check qeph_verify_port)
  set_target_properties(qeph_force_port_cuda_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_link_libraries(qeph_force_port_cuda_check PRIVATE tl_qeph qeph_q1_native CUDA::cudart GTest::gtest_main)
  target_compile_options(qeph_force_port_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME qeph_force_port_cuda_check COMMAND qeph_force_port_cuda_check)
  set_tests_properties(qeph_force_port_cuda_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 45)
endif()
