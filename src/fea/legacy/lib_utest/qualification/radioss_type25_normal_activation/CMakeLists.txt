cmake_minimum_required(VERSION 3.22)
project(TLType25NormalActivation LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TYPE25_NORMAL_ACTIVATION_CUDA "Build actual parallel mask qualification" OFF)
if(TYPE25_NORMAL_ACTIVATION_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/RadiossType25NormalActivation.cmake")
set(native "${CMAKE_CURRENT_BINARY_DIR}/native")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE result ERROR_VARIABLE error)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Pinned activation reference preparation failed: ${error}")
endif()
file(GLOB native_inputs CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/native/*"
  "${CMAKE_CURRENT_LIST_DIR}/native/original/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${native_inputs})
file(MAKE_DIRECTORY "${native}/modules")
add_library(type25_normal_activation_oracle STATIC NativeOracle.cpp
  "${native}/Constants.F90" "${native}/Boundary.F90" "${native}/Tag.F"
  "${native}/Free.F" "${native}/Wrapper.F90")
set_target_properties(type25_normal_activation_oracle PROPERTIES Fortran_MODULE_DIRECTORY "${native}/modules")
target_include_directories(type25_normal_activation_oracle PRIVATE "${native}" "${native}/modules")
target_link_libraries(type25_normal_activation_oracle PUBLIC tl_radioss_type25_normal_activation)
target_compile_options(type25_normal_activation_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(type25_normal_activation_host HostTest.cpp)
target_link_libraries(type25_normal_activation_host PRIVATE type25_normal_activation_oracle GTest::gtest_main)
add_test(NAME type25_normal_activation_host COMMAND type25_normal_activation_host)
add_executable(type25_normal_activation_consumer Consumer.cpp)
target_link_libraries(type25_normal_activation_consumer PRIVATE tl_radioss_type25_normal_activation)
add_test(NAME type25_normal_activation_consumer COMMAND type25_normal_activation_consumer)
add_test(NAME type25_normal_activation_source COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
set_tests_properties(type25_normal_activation_host type25_normal_activation_consumer type25_normal_activation_source
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
if(TYPE25_NORMAL_ACTIVATION_CUDA)
  add_executable(type25_normal_activation_cuda CudaTest.cu)
  target_link_libraries(type25_normal_activation_cuda PRIVATE type25_normal_activation_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_normal_activation_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME type25_normal_activation_cuda COMMAND type25_normal_activation_cuda)
  set_tests_properties(type25_normal_activation_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
