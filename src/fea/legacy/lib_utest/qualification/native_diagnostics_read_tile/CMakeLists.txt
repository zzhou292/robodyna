cmake_minimum_required(VERSION 3.22)
project(TLNativeDiagnosticsReadTile LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(existing "${CMAKE_CURRENT_LIST_DIR}/../native_contact_diagnostics")
set(oracle "${CMAKE_CURRENT_BINARY_DIR}/oracle")
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(CUDAToolkit REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" "${CMAKE_CURRENT_LIST_DIR}/source-manifest.json"
  "${CMAKE_CURRENT_LIST_DIR}/frozen/Kernels.cu"
  "${TL_ROOT}/lib_src/collision/radioss_type25/runtime/Kernels.cu")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py"
  --output "${oracle}" COMMAND_ERROR_IS_FATAL ANY)
add_library(native_read_tile_layout STATIC "${TL_ROOT}/lib_src/collision/radioss_type25/runtime/Layout.cpp")
target_include_directories(native_read_tile_layout PUBLIC "${TL_ROOT}" ${CUDAToolkit_INCLUDE_DIRS})
target_compile_features(native_read_tile_layout PUBLIC cxx_std_17)
target_compile_options(native_read_tile_layout PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(native_read_tile_host "${existing}/LayoutTest.cpp" ReadTest.cpp)
target_link_libraries(native_read_tile_host PRIVATE native_read_tile_layout GTest::gtest_main)
target_compile_options(native_read_tile_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME native_read_tile_host COMMAND native_read_tile_host)
add_test(NAME native_read_tile_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py")
option(TL_NATIVE_READ_TILE_CUDA "Literal ad610/current diagnostic kernels with complete Control comparison" OFF)
if(TL_NATIVE_READ_TILE_CUDA)
  enable_language(CUDA)
  add_executable(native_read_tile_cuda BoundaryTest.cu)
  target_include_directories(native_read_tile_cuda PRIVATE "${oracle}")
  target_link_libraries(native_read_tile_cuda PRIVATE native_read_tile_layout GTest::gtest_main CUDA::cudart)
  set_target_properties(native_read_tile_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(native_read_tile_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;--ptxas-options=-v;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  add_test(NAME native_read_tile_cuda COMMAND native_read_tile_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
