cmake_minimum_required(VERSION 3.18)
project(TLRigidPartModel LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/constraints/NodalRigidPartAssemblyModel.cmake")
add_executable(rigid_part_model_check ValueTest.cpp RoundingTest.cpp SolidTest.cpp)
target_link_libraries(rigid_part_model_check PRIVATE tl_nodal_rigid_part_assembly_model GTest::gtest_main)
target_compile_options(rigid_part_model_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME rigid_part_model_values COMMAND rigid_part_model_check)
option(RIGID_PART_MODEL_NATIVE "Build the independent original raw/merge/correction native gate" OFF)
if(RIGID_PART_MODEL_NATIVE)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(rigid_native "${CMAKE_CURRENT_LIST_DIR}/../nodal_rigid_group/native")
  set(assembly_native "${CMAKE_CURRENT_LIST_DIR}/../nodal_rigid_group/assembly/native")
  foreach(kind rigid assembly)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${${kind}_native}/verify_sources.py"
      RESULT_VARIABLE identity_result ERROR_VARIABLE identity_error)
    if(NOT identity_result EQUAL 0)
      message(FATAL_ERROR "Existing ${kind} native source identity failed: ${identity_error}")
    endif()
    add_test(NAME rigid_part_model_${kind}_identity
      COMMAND "${Python3_EXECUTABLE}" "${${kind}_native}/verify_sources.py")
  endforeach()
  add_library(rigid_part_model_native STATIC "${assembly_native}/NativeAssembly.F"
    "${rigid_native}/NativeRigidGroup.F")
  target_include_directories(rigid_part_model_native PRIVATE "${assembly_native}" "${rigid_native}")
  target_compile_options(rigid_part_model_native PRIVATE -cpp -ffixed-line-length-none
    -fcheck=bounds -fbacktrace -fno-fast-math -ffp-contract=off)
  add_executable(rigid_part_model_native_check NativeTest.cpp SolidNativeTest.cpp
    "../nodal_rigid_group/assembly/AssemblyNativeOracle.cpp")
  target_link_libraries(rigid_part_model_native_check PRIVATE
    tl_nodal_rigid_part_assembly_model rigid_part_model_native GTest::gtest_main)
  target_compile_options(rigid_part_model_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME rigid_part_model_native COMMAND rigid_part_model_native_check)
endif()
get_property(model_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${model_tests} PROPERTIES PROCESSORS 1 RUN_SERIAL TRUE TIMEOUT 120)
