cmake_minimum_required(VERSION 3.22)
project(TLExtendedSolidCoefficients LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory(../nodal_coefficients legacy)
add_executable(extended_solid_coefficients_host HostTest.cpp BoundsTest.cpp LifetimeTest.cpp)
target_include_directories(extended_solid_coefficients_host PRIVATE "${TL_ROOT}")
target_compile_features(extended_solid_coefficients_host PRIVATE cxx_std_17)
target_compile_options(extended_solid_coefficients_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(extended_solid_coefficients_host PRIVATE tl_nodal_coefficient_ledger
  tl_nodal_mass_binding GTest::gtest_main)
add_test(NAME extended_solid_coefficients_host COMMAND extended_solid_coefficients_host)
set_tests_properties(extended_solid_coefficients_host PROPERTIES PROCESSORS 1 RUN_SERIAL TRUE TIMEOUT 120)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME extended_solid_coefficients_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(extended_solid_coefficients_identity PROPERTIES PROCESSORS 1 RUN_SERIAL TRUE TIMEOUT 60)
