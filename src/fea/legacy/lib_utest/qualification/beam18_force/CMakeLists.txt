cmake_minimum_required(VERSION 3.22)
project(TLBeam18Force LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(beam18_force_host HostTest.cpp)
target_include_directories(beam18_force_host PRIVATE "${TL_ROOT}")
target_compile_features(beam18_force_host PRIVATE cxx_std_17)
target_compile_options(beam18_force_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(beam18_force_host PRIVATE GTest::gtest_main)
add_test(NAME beam18_force_host COMMAND beam18_force_host)
add_test(NAME beam18_force_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
option(BEAM18_FORCE_NATIVE "Independent complete native beam force caller" OFF)
option(BEAM18_FORCE_CUDA "Actual CUDA beam recurrence and rejection" OFF)
set(BEAM18_FORCE_SOURCE_FIXTURE "" CACHE PATH "Authenticated original142 beam geometry")
if(BEAM18_FORCE_NATIVE OR BEAM18_FORCE_CUDA)
  enable_language(Fortran)
  add_subdirectory(native)
  add_library(beam18_force_oracle STATIC NativeOracle.cpp ../beam18_reference/NativeOracle.cpp)
  target_include_directories(beam18_force_oracle PUBLIC "${TL_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_features(beam18_force_oracle PUBLIC cxx_std_17)
  target_compile_options(beam18_force_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(beam18_force_oracle PUBLIC beam18_force_native GTest::gtest)
  add_executable(beam18_force_native_check NativeTest.cpp)
  target_link_libraries(beam18_force_native_check PRIVATE beam18_force_oracle GTest::gtest_main)
  target_compile_options(beam18_force_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME beam18_force_native_check COMMAND beam18_force_native_check)
endif()
if(BEAM18_FORCE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(beam18_force_cuda_check CudaTest.cu)
  target_compile_features(beam18_force_cuda_check PRIVATE cuda_std_17)
  target_compile_options(beam18_force_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(beam18_force_cuda_check PRIVATE beam18_force_oracle CUDA::cudart GTest::gtest_main)
  add_test(NAME beam18_force_cuda_check COMMAND beam18_force_cuda_check)
endif()
if(BEAM18_FORCE_SOURCE_FIXTURE)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_fixture.py"
    --fixture "${BEAM18_FORCE_SOURCE_FIXTURE}" --output "${CMAKE_CURRENT_BINARY_DIR}/fixture/OriginalFixture.h"
    RESULT_VARIABLE status)
  if(NOT status EQUAL 0)
    message(FATAL_ERROR "Beam18 force original fixture authentication failed")
  endif()
  foreach(target beam18_force_native_check beam18_force_cuda_check)
    if(TARGET ${target})
      target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture")
      target_compile_definitions(${target} PRIVATE BEAM18_FORCE_ORIGINAL)
    endif()
  endforeach()
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
