cmake_minimum_required(VERSION 3.22)
project(TLVehicleShellResident LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/elements/ShellBatchPlasticityBinding.cmake")
add_executable(vehicle_shell_resident_host_check
  ResidentLayoutTest.cpp ResidentStartupTest.cpp
  "${TL_ROOT}/lib_src/elements/qeph/QephBatchModel.cpp"
  "${TL_ROOT}/lib_src/elements/t3/T3BatchModel.cpp")
target_include_directories(vehicle_shell_resident_host_check PRIVATE "${TL_ROOT}" "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(vehicle_shell_resident_host_check PRIVATE tl_shell_batch_plasticity_binding GTest::gtest_main)
target_compile_options(vehicle_shell_resident_host_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME vehicle_shell_resident_host_check COMMAND vehicle_shell_resident_host_check)
set_tests_properties(vehicle_shell_resident_host_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(TL_VEHICLE_SHELL_RESIDENT_CUDA "Build full-count resident CUDA qualification" OFF)
if(TL_VEHICLE_SHELL_RESIDENT_CUDA)
  enable_language(CUDA)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" "tl-nodal-owner")
  include("${TL_ROOT}/lib_src/elements/qeph/QephBatch.cmake")
  include("${TL_ROOT}/lib_src/elements/t3/T3Batch.cmake")
  include("${TL_ROOT}/lib_src/elements/ShellBatchPublication.cmake")
  include("${CMAKE_CURRENT_LIST_DIR}/VehicleResident.cmake")
endif()
