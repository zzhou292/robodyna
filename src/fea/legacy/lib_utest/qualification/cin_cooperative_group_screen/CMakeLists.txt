cmake_minimum_required(VERSION 3.22)
project(TLCinCooperativeGroupScreen LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(packet_dir "${CMAKE_CURRENT_LIST_DIR}/../cin_parallel_ordinary")
include("${tl_root}/lib_src/constraints/tied_shell/TiedCinAttachment.cmake")
add_executable(cin_cooperative_group_host MemberTest.cpp GroupTest.cpp ../cin_parallel_groups/LayoutTest.cpp "${packet_dir}/Packet.cpp")
target_include_directories(cin_cooperative_group_host PRIVATE "${tl_root}" "${tl_root}/lib_src/solvers")
target_link_libraries(cin_cooperative_group_host PRIVATE tl_tied_cin_attachment GTest::gtest_main CUDA::toolkit)
target_compile_features(cin_cooperative_group_host PRIVATE cxx_std_17)
target_compile_options(cin_cooperative_group_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME cin_cooperative_group_host COMMAND cin_cooperative_group_host)
set_tests_properties(cin_cooperative_group_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(CIN_COOPERATIVE_GROUP_CUDA "Frozen complete screen and actual owner CUDA tests" OFF)
if(CIN_COOPERATIVE_GROUP_CUDA)
  enable_language(CUDA)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  add_executable(cin_cooperative_group_cuda CudaTest.cu FrozenScreen.cu
    "${packet_dir}/DevicePacket.cu" "${packet_dir}/Serial.cu"
    "${packet_dir}/Packet.cpp" "${packet_dir}/Compare.cpp"
    ../cin_parallel_groups/CallerTest.cu ../cin_parallel_groups/OwnerTest.cu
    ../cin_parallel_groups/Frozen.cu ../rigid_assembly_owner/Fixture.cpp)
  target_include_directories(cin_cooperative_group_cuda PRIVATE "${tl_root}" "${tl_root}/lib_src/solvers")
  target_link_libraries(cin_cooperative_group_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main CUDA::cudart)
  set_target_properties(cin_cooperative_group_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(cin_cooperative_group_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;--ptxas-options=-v;-Xcompiler=-fno-fast-math,-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>)
  add_test(NAME cin_cooperative_group_cuda COMMAND cin_cooperative_group_cuda)
  set_tests_properties(cin_cooperative_group_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
add_test(NAME cin_cooperative_group_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(cin_cooperative_group_identity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
