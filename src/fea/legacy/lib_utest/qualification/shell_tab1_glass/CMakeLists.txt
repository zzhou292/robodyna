cmake_minimum_required(VERSION 3.22)
project(ShellTab1Glass LANGUAGES CXX)
option(TL_TAB1_GLASS_NATIVE "Build complete native TAB1/table/caller comparisons" OFF)
option(TL_TAB1_GLASS_CUDA "Build device-owned TAB1 caller comparisons" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_executable(shell_tab1_glass_values_check Tab1PointTest.cpp Tab1SectionTest.cpp LegacyWorkParityTest.cpp)
target_include_directories(shell_tab1_glass_values_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_tab1_glass_values_check PRIVATE cxx_std_17)
target_compile_options(shell_tab1_glass_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_tab1_glass_values_check PRIVATE GTest::gtest_main)
add_test(NAME shell_tab1_glass_values COMMAND shell_tab1_glass_values_check)
set(TL_LAYERED_FAILURE_NATIVE ${TL_TAB1_GLASS_NATIVE})
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_failure" layered-failure)
if(TL_TAB1_GLASS_NATIVE)
  enable_language(Fortran)
  set(prepared "${CMAKE_CURRENT_BINARY_DIR}/prepared")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/native/prepare_sources.py"
    --output "${prepared}" RESULT_VARIABLE status ERROR_VARIABLE diagnostic)
  if(NOT status EQUAL 0)
    message(FATAL_ERROR "TAB1 native source preparation failed: ${diagnostic}")
  endif()
  add_custom_target(shell_tab1_glass_sources
    COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/native/prepare_sources.py"
      --output "${prepared}" --check VERBATIM)
  add_library(shell_tab1_glass_native STATIC native/NativeSupport.F90 native/NativeDefaults.F
    native/NativePoint.F native/NativeCaller.F90
    "${prepared}/TableModule.F" "${prepared}/TableInterp.F" "${prepared}/TableVinterp.F"
    "${prepared}/NativeFailure.F"
    "${prepared}/LegacyCaller.F90" "${prepared}/LegacyParent.F" "${prepared}/LegacyInterfaces.F90")
  add_dependencies(shell_tab1_glass_native shell_tab1_glass_sources)
  foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(value shell_layered_failure_native ${property})
    set_property(TARGET shell_tab1_glass_native PROPERTY ${property} "${value}")
  endforeach()
  get_target_property(caller_modules shell_layered_failure_native Fortran_MODULE_DIRECTORY)
  target_include_directories(shell_tab1_glass_native BEFORE PRIVATE "${prepared}" "${caller_modules}")
  target_include_directories(shell_tab1_glass_native PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/../native/qeph/original/starter/share/includes")
  set_target_properties(shell_tab1_glass_native PROPERTIES
    Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tab1-modules")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/tab1-modules")
  target_include_directories(shell_tab1_glass_native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/tab1-modules")
  target_link_libraries(shell_tab1_glass_native PUBLIC shell_layered_failure_native)
  add_library(shell_tab1_glass_test_support STATIC Tab1NativeSupport.cpp
    "${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_failure/FailureNativeFixture.cpp")
  target_compile_features(shell_tab1_glass_test_support PUBLIC cxx_std_17)
  target_compile_options(shell_tab1_glass_test_support PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(shell_tab1_glass_test_support PUBLIC shell_tab1_glass_native GTest::gtest_main)
  add_executable(shell_tab1_glass_native_check Tab1NativePointTest.cpp Tab1NativeSectionTest.cpp LegacyNativeParityTest.cpp)
  target_compile_options(shell_tab1_glass_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(shell_tab1_glass_native_check PRIVATE shell_tab1_glass_test_support)
  add_test(NAME shell_tab1_glass_native COMMAND shell_tab1_glass_native_check)
endif()
if(TL_TAB1_GLASS_CUDA)
  if(NOT TL_TAB1_GLASS_NATIVE)
    message(FATAL_ERROR "CUDA TAB1 qualification requires the complete native oracle")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_tab1_glass_cuda_check Tab1CudaTest.cu Tab1PointCudaTest.cu)
  target_link_libraries(shell_tab1_glass_cuda_check PRIVATE shell_tab1_glass_test_support CUDA::cudart)
  target_compile_features(shell_tab1_glass_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_tab1_glass_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_tab1_glass_cuda COMMAND shell_tab1_glass_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
