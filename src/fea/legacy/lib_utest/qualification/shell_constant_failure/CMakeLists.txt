cmake_minimum_required(VERSION 3.22)
project(ShellConstantFailureQualification LANGUAGES CXX Fortran)
option(TL_CONSTANT_FAILURE_CUDA "Compare actual CUDA point failure with native code" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()

add_executable(shell_constant_failure_value_check FailureValueTest.cpp)
target_include_directories(shell_constant_failure_value_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_constant_failure_value_check PRIVATE cxx_std_17)
target_compile_options(shell_constant_failure_value_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_constant_failure_value_check PRIVATE GTest::gtest_main)
add_test(NAME shell_constant_failure_values COMMAND shell_constant_failure_value_check)

# Reuse the existing authenticated precision/constants/include environment.
set(LAW44_POINT_BUILD_TESTS OFF)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/law44" native-law44)
set(prepared "${CMAKE_CURRENT_BINARY_DIR}/prepared")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/native/prepare_sources.py"
  --output "${prepared}" RESULT_VARIABLE status ERROR_VARIABLE diagnostic)
if(NOT status EQUAL 0)
  message(FATAL_ERROR "Native failure source preparation failed: ${diagnostic}")
endif()
add_custom_target(shell_constant_failure_sources
  COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/native/prepare_sources.py"
  --output "${prepared}" VERBATIM)
add_library(shell_constant_failure_native STATIC native/NativePoint.F "${prepared}/NativeFailure.F")
add_dependencies(shell_constant_failure_native shell_constant_failure_sources)
foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
  get_target_property(value law44_point_native ${property})
  set_property(TARGET shell_constant_failure_native PROPERTY ${property} "${value}")
endforeach()
get_target_property(modules law44_point_native Fortran_MODULE_DIRECTORY)
target_include_directories(shell_constant_failure_native PRIVATE "${modules}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../native/qeph/original/starter/share/includes")
target_link_libraries(shell_constant_failure_native PUBLIC law44_point_native)
add_executable(shell_constant_failure_native_check FailureNativeTest.cpp)
target_link_libraries(shell_constant_failure_native_check PRIVATE shell_constant_failure_native GTest::gtest_main)
target_compile_options(shell_constant_failure_native_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME shell_constant_failure_native COMMAND shell_constant_failure_native_check)

if(TL_CONSTANT_FAILURE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_constant_failure_cuda_check FailureCudaTest.cu)
  target_link_libraries(shell_constant_failure_cuda_check PRIVATE shell_constant_failure_native GTest::gtest_main CUDA::cudart)
  target_compile_features(shell_constant_failure_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_constant_failure_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_constant_failure_cuda COMMAND shell_constant_failure_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
