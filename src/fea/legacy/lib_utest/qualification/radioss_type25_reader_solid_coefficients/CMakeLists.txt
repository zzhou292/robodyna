cmake_minimum_required(VERSION 3.22)
project(TLType25ReaderSolidCoefficients LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TYPE25_COEFFICIENTS_CUDA "Build original coefficient GPU comparisons" OFF)
if(TYPE25_COEFFICIENTS_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_coated_coefficients" coated)
set(native "${CMAKE_CURRENT_BINARY_DIR}/native")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Original reader solid coefficient extraction failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/prepare.py native/Reference.F.in
  ../radioss_type25_coefficients/native/prepare.py ../radioss_type25_coefficients/native/original/i25sti3.F
  ../radioss_type25_main_geometry/native/prepare.py ../radioss_type25_main_geometry/native/original/volint.F)
add_library(reader_solid_oracle STATIC NativeOracle.cpp "${native}/Reference.F")
target_link_libraries(reader_solid_oracle PUBLIC tl_radioss_type25_coefficients)
target_compile_options(reader_solid_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(reader_solid_host HostTest.cpp)
target_link_libraries(reader_solid_host PRIVATE reader_solid_oracle GTest::gtest_main)
add_test(NAME reader_solid_host COMMAND reader_solid_host)
add_executable(reader_solid_consumer Consumer.cpp)
target_link_libraries(reader_solid_consumer PRIVATE tl_radioss_type25_coefficients)
add_test(NAME reader_solid_consumer COMMAND reader_solid_consumer)
add_test(NAME reader_solid_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
if(TYPE25_COEFFICIENTS_CUDA)
  add_executable(reader_solid_cuda CudaTest.cu)
  target_link_libraries(reader_solid_cuda PRIVATE reader_solid_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(reader_solid_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME reader_solid_cuda COMMAND reader_solid_cuda)
endif()
