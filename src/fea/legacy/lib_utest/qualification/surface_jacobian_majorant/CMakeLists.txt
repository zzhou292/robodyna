cmake_minimum_required(VERSION 3.22)
project(TLSurfaceJacobianMajorant LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Boost REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/SurfaceJacobianMajorant.cmake")
add_library(surface_majorant_exact STATIC ExactOracle.cpp)
target_link_libraries(surface_majorant_exact PUBLIC tl_surface_jacobian_majorant)
target_include_directories(surface_majorant_exact PRIVATE ${Boost_INCLUDE_DIRS})
target_compile_options(surface_majorant_exact PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(surface_jacobian_majorant_host ValueTest.cpp CongruenceTest.cpp)
target_link_libraries(surface_jacobian_majorant_host PRIVATE surface_majorant_exact GTest::gtest_main)
target_compile_options(surface_jacobian_majorant_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME surface_jacobian_majorant_host COMMAND surface_jacobian_majorant_host)
set_tests_properties(surface_jacobian_majorant_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(SURFACE_MAJORANT_CUDA "Build actual device value qualification" OFF)
if(SURFACE_MAJORANT_CUDA)
  enable_language(CUDA)
  add_executable(surface_jacobian_majorant_cuda CudaTest.cu)
  target_link_libraries(surface_jacobian_majorant_cuda PRIVATE surface_majorant_exact GTest::gtest_main)
  set_target_properties(surface_jacobian_majorant_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(surface_jacobian_majorant_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>)
  add_test(NAME surface_jacobian_majorant_cuda COMMAND surface_jacobian_majorant_cuda)
  set_tests_properties(surface_jacobian_majorant_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
endif()
