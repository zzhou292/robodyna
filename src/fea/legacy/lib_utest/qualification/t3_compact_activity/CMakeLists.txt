cmake_minimum_required(VERSION 3.22)
project(TLT3CompactActivity LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(t3_compact_activity_host HostTest.cpp "${tl_root}/lib_src/elements/t3/mapped/ActivityReport.cpp")
target_include_directories(t3_compact_activity_host PRIVATE "${tl_root}")
target_link_libraries(t3_compact_activity_host PRIVATE CUDA::toolkit GTest::gtest_main)
target_compile_features(t3_compact_activity_host PRIVATE cxx_std_17)
target_compile_options(t3_compact_activity_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME t3_compact_activity_host COMMAND t3_compact_activity_host)
add_test(NAME t3_compact_activity_sources COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(t3_compact_activity_host t3_compact_activity_sources PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
option(T3_COMPACT_ACTIVITY_CUDA "Complete frozen predicate and actual-owner qualification" OFF)
if(T3_COMPACT_ACTIVITY_CUDA)
  enable_language(CUDA)
  add_executable(t3_compact_activity_cuda KernelTest.cu
    "${tl_root}/lib_src/elements/t3/mapped/ActivityKernels.cu" "${tl_root}/lib_src/elements/t3/mapped/ActivityReport.cpp")
  target_include_directories(t3_compact_activity_cuda PRIVATE "${tl_root}")
  target_link_libraries(t3_compact_activity_cuda PRIVATE CUDA::cudart GTest::gtest_main)
  set(PHYSICAL_PUBLICATION_CUDA ON CACHE BOOL "Owning physical fixture" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../physical_publication" physical-owner)
  add_executable(t3_compact_activity_owner OwnerTest.cu OwnerProbe.cpp
    ../physical_publication/OwnerStartup.cu ../physical_publication/OwnerAttempt.cu)
  target_link_libraries(t3_compact_activity_owner PRIVATE physical_publication_fixture
    tl_shell_batch_publication CUDA::cudart GTest::gtest_main)
  target_link_options(t3_compact_activity_owner PRIVATE
    "-Wl,--wrap=cudaMemcpyAsync" "-Wl,--wrap=cudaStreamSynchronize" "-Wl,--wrap=cudaGetLastError")
  foreach(target IN ITEMS t3_compact_activity_cuda t3_compact_activity_owner)
    set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(${target} PRIVATE
      $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>
      $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>)
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 300)
  endforeach()
endif()
