cmake_minimum_required(VERSION 3.22)
project(ShellReferencePlacement LANGUAGES CXX)
option(TL_SHELL_PLACEMENT_NATIVE "Build exact native placement and glass caller comparisons" OFF)
option(TL_SHELL_PLACEMENT_CUDA "Build device-owned shifted glass caller qualification" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
add_executable(shell_reference_placement_values_check PlacementValueTest.cpp)
target_include_directories(shell_reference_placement_values_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_reference_placement_values_check PRIVATE cxx_std_17)
target_compile_options(shell_reference_placement_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_reference_placement_values_check PRIVATE GTest::gtest_main)
add_test(NAME shell_reference_placement_values COMMAND shell_reference_placement_values_check)
if(TL_SHELL_PLACEMENT_NATIVE)
  include(Native.cmake)
endif()
if(TL_SHELL_PLACEMENT_CUDA)
  if(NOT TL_SHELL_PLACEMENT_NATIVE)
    message(FATAL_ERROR "Device placement qualification requires its complete native oracle")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_reference_placement_cuda_check PlacementCudaTest.cu PlacementCoefficientCudaTest.cu)
  target_link_libraries(shell_reference_placement_cuda_check PRIVATE placement_native_test_support CUDA::cudart)
  target_compile_features(shell_reference_placement_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_reference_placement_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_reference_placement_cuda COMMAND shell_reference_placement_cuda_check)
endif()
get_property(placement_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${placement_tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
  ENVIRONMENT "OMP_NUM_THREADS=1;OPENBLAS_NUM_THREADS=1;MKL_NUM_THREADS=1")
