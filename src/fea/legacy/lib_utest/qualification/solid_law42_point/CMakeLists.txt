cmake_minimum_required(VERSION 3.22)
project(TLLaw42Point LANGUAGES CXX C)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(law42_test_target name)
  target_include_directories(${name} PRIVATE "${TL_ROOT}")
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_link_libraries(${name} PRIVATE GTest::gtest_main Eigen3::Eigen)
  target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
endfunction()
add_executable(law42_host_check HostTest.cpp)
law42_test_target(law42_host_check)
add_test(NAME law42_host COMMAND law42_host_check)
option(LAW42_NATIVE "Complete independent native material oracle" OFF)
option(LAW42_CUDA "Actual CUDA material values" OFF)
if(LAW42_NATIVE OR LAW42_CUDA)
  enable_language(Fortran)
  add_subdirectory(native)
  add_executable(law42_native_check NativeTest.cpp)
  law42_test_target(law42_native_check)
  target_link_libraries(law42_native_check PRIVATE law42_native)
  add_test(NAME law42_native COMMAND law42_native_check)
endif()
if(LAW42_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(law42_cuda_check CudaTest.cu)
  law42_test_target(law42_cuda_check)
  target_compile_features(law42_cuda_check PRIVATE cuda_std_17)
  target_compile_options(law42_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr;--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(law42_cuda_check PRIVATE law42_native CUDA::cudart)
  add_test(NAME law42_cuda COMMAND law42_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
