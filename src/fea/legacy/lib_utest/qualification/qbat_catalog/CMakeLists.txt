cmake_minimum_required(VERSION 3.22)
project(TLQbatCatalogQualification LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include("${tl_root}/lib_src/elements/ShellFormulationScope.cmake")
add_executable(qbat_catalog_host_test ValueTest.cpp FailureTest.cpp ScopeTest.cpp)
target_link_libraries(qbat_catalog_host_test PRIVATE tl_shell_formulation_scope GTest::gtest_main)
target_compile_options(qbat_catalog_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qbat_catalog_host COMMAND qbat_catalog_host_test)

option(QBAT_CATALOG_SOURCE_CHECKS "Build root-scheduled complete original 4250+1 catalog" OFF)
if(QBAT_CATALOG_SOURCE_CHECKS)
  add_executable(qbat_catalog_original_test OriginalTest.cpp "../qbat_binding/OriginalFixture.cpp")
  target_link_libraries(qbat_catalog_original_test PRIVATE tl_shell_formulation_scope GTest::gtest_main)
  target_compile_options(qbat_catalog_original_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME qbat_catalog_original COMMAND qbat_catalog_original_test)
  add_test(NAME qbat_catalog_source_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../qbat/source_fixture/verify_fixture.py")
endif()
option(QBAT_CATALOG_LEGACY_CHECKS "Build affected small legacy host catalogs" ON)
if(QBAT_CATALOG_LEGACY_CHECKS)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../t3/mixed_binding" legacy-binding)
  include("${CMAKE_CURRENT_LIST_DIR}/../plasticity_binding/PlasticityBinding.cmake")
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
