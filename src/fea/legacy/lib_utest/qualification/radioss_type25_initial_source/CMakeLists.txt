cmake_minimum_required(VERSION 3.22)
project(TLType25InitialSource LANGUAGES C CXX CUDA Fortran)
enable_testing()
get_filename_component(TYPE25_INITIAL_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(TYPE25_INITIAL_ROOT "${TYPE25_INITIAL_SOURCE_ROOT}")
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
include("${TYPE25_INITIAL_SOURCE_ROOT}/lib_src/collision/RadiossType25InitialState.cmake")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_post_gapm" topology-regression)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_tied_removal" tied-regression)
set(TYPE25_CANDIDATE_CUDA ON CACHE BOOL "Qualify factored existing inventory kernels" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_candidates" candidate-regression)
include("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_initial_state/native/Native.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/native/Native.cmake")
add_executable(type25_initial_source_host HostTest.cpp)
target_compile_features(type25_initial_source_host PRIVATE cxx_std_17)
target_link_libraries(type25_initial_source_host PRIVATE tl_radioss_type25_initial_state type25_initial_inventory_native
  type25_search_startup_oracle type25_startup_oracle GTest::gtest_main)
target_compile_options(type25_initial_source_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type25_initial_source_host COMMAND type25_initial_source_host)
set_tests_properties(type25_initial_source_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
add_executable(type25_initial_source_cuda CudaTest.cu)
set_target_properties(type25_initial_source_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_link_libraries(type25_initial_source_cuda PRIVATE tl_radioss_type25_initial_state type25_initial_inventory_native
  type25_initial_state_native type25_search_startup_oracle type25_tied_removal_oracle type25_startup_oracle GTest::gtest_main CUDA::cudart)
target_link_libraries(type25_initial_source_cuda PRIVATE type25_post_gapm_oracle tl_radioss_type25_interface_surface type25_initial_source_history_native)
target_compile_options(type25_initial_source_cuda PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
add_test(NAME type25_initial_source_cuda COMMAND type25_initial_source_cuda)
set_tests_properties(type25_initial_source_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 2 TIMEOUT 180)
add_executable(type25_initial_source_consumer Consumer.cpp)
target_compile_features(type25_initial_source_consumer PRIVATE cxx_std_17)
target_link_libraries(type25_initial_source_consumer PRIVATE tl_radioss_type25_initial_state)
add_test(NAME type25_initial_source_consumer COMMAND type25_initial_source_consumer)
set_tests_properties(type25_initial_source_consumer PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
