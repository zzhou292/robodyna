cmake_minimum_required(VERSION 3.22)
project(TLT3ActivityReadback LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(t3_activity_readback_host HostTest.cpp)
target_include_directories(t3_activity_readback_host PRIVATE "${tl_root}")
target_link_libraries(t3_activity_readback_host PRIVATE CUDA::toolkit GTest::gtest_main)
target_compile_features(t3_activity_readback_host PRIVATE cxx_std_17)
target_compile_options(t3_activity_readback_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME t3_activity_readback_host COMMAND t3_activity_readback_host)
add_test(NAME t3_activity_readback_source_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(t3_activity_readback_host t3_activity_readback_source_identity
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(T3_ACTIVITY_READBACK_CUDA "Actual mapped owner, same-call readback and rejection gates" OFF)
if(T3_ACTIVITY_READBACK_CUDA)
  include("${CMAKE_CURRENT_LIST_DIR}/Cuda.cmake")
endif()
