cmake_minimum_required(VERSION 3.18)
project(TLNodalCoefficients LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/assembly/NodalCoefficientLedger.cmake")
include("${tl_root}/lib_src/assembly/NodalMassBinding.cmake")
add_executable(nodal_coefficient_check ValueTest.cpp RoundingTest.cpp PointMassTest.cpp SolidTest.cpp)
target_link_libraries(nodal_coefficient_check PRIVATE tl_nodal_coefficient_ledger tl_nodal_mass_binding GTest::gtest_main)
target_compile_options(nodal_coefficient_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME nodal_coefficient_values COMMAND nodal_coefficient_check)
option(NODAL_COEFFICIENT_FULL_COUNT "Build actual source-count synthetic capacity gate" OFF)
if(NODAL_COEFFICIENT_FULL_COUNT)
  add_executable(nodal_coefficient_size_check SizeTest.cpp)
  target_link_libraries(nodal_coefficient_size_check PRIVATE tl_nodal_coefficient_ledger GTest::gtest_main)
  target_compile_options(nodal_coefficient_size_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME nodal_coefficient_size COMMAND nodal_coefficient_size_check)
endif()
option(NODAL_COEFFICIENT_SOLID_NATIVE "Build actual selected solid source/native coefficient gate" OFF)
if(NODAL_COEFFICIENT_SOLID_NATIVE)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_subdirectory(../solid18_reference/native solid18_native)
  add_subdirectory(../solid24_reference/native solid24_native)
  add_subdirectory(../solid6z_reference/native solid6z_native)
  add_executable(nodal_coefficient_solid_native_check SolidSourceNativeTest.cpp
    ../solid18_reference/NativeOracle.cpp)
  target_link_libraries(nodal_coefficient_solid_native_check PRIVATE tl_nodal_coefficient_ledger
    solid18_reference_native solid24_reference_native solid6z_reference_native GTest::gtest_main)
  target_compile_options(nodal_coefficient_solid_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME nodal_coefficient_solid_native COMMAND nodal_coefficient_solid_native_check)
  add_test(NAME nodal_coefficient_solid_source_identity COMMAND "${Python3_EXECUTABLE}"
    -B "${CMAKE_CURRENT_LIST_DIR}/../solid18_reference/verify_source.py")
  add_test(NAME nodal_coefficient_rubber_source_identity COMMAND "${Python3_EXECUTABLE}"
    -B "${CMAKE_CURRENT_LIST_DIR}/../solid_common/source_fixture/verify_fixture.py")
endif()
option(NODAL_COEFFICIENT_NATIVE "Build independent qualified native spring coefficient controls" OFF)
if(NODAL_COEFFICIENT_NATIVE)
  enable_language(Fortran)
  set(TYPE13_NATIVE_CHECKS ON CACHE BOOL "Reuse native coefficients" FORCE)
  set(TYPE25_NATIVE_CHECKS ON CACHE BOOL "Reuse native coefficients" FORCE)
  set(TYPE13_CUDA_CHECKS OFF CACHE BOOL "No GPU needed" FORCE)
  set(TYPE25_CUDA_CHECKS OFF CACHE BOOL "No GPU needed" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../type13" native13)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../type25" native25)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(rigid_assembly_native "${CMAKE_CURRENT_LIST_DIR}/../nodal_rigid_group/assembly/native")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${rigid_assembly_native}/verify_sources.py"
    RESULT_VARIABLE point_mass_source_check ERROR_VARIABLE point_mass_source_error)
  if(NOT point_mass_source_check EQUAL 0)
    message(FATAL_ERROR "Native ADMAS5 source identity failed: ${point_mass_source_error}")
  endif()
  add_library(nodal_element_mass_native STATIC "${rigid_assembly_native}/NativeAssembly.F")
  target_include_directories(nodal_element_mass_native PRIVATE "${rigid_assembly_native}")
  target_compile_options(nodal_element_mass_native PRIVATE -cpp -ffixed-line-length-none
    -fcheck=bounds -fbacktrace -fno-fast-math -ffp-contract=off)
  add_test(NAME element_mass_native_source_identity
    COMMAND "${Python3_EXECUTABLE}" "${rigid_assembly_native}/verify_sources.py")
  add_executable(nodal_coefficient_native_check NativeTest.cpp PointMassNativeTest.cpp "../type13/NativeOracle.cpp")
  target_link_libraries(nodal_coefficient_native_check PRIVATE tl_nodal_coefficient_ledger
    type13_native type25_native nodal_element_mass_native GTest::gtest_main)
  target_compile_options(nodal_coefficient_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME nodal_coefficient_native COMMAND nodal_coefficient_native_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES PROCESSORS 1 RUN_SERIAL TRUE TIMEOUT 120)
