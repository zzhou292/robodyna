cmake_minimum_required(VERSION 3.22)
project(TLQbatMappedGather LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(qbat_mapped_gather_host HostTest.cpp
  "${tl_root}/lib_src/elements/qbat/QbatBatchArena.cpp"
  "${tl_root}/lib_src/elements/qbat/QbatBatchIdentity.cpp")
target_include_directories(qbat_mapped_gather_host PRIVATE "${tl_root}")
target_link_libraries(qbat_mapped_gather_host PRIVATE GTest::gtest_main CUDA::toolkit)
target_compile_features(qbat_mapped_gather_host PRIVATE cxx_std_17)
target_compile_options(qbat_mapped_gather_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qbat_mapped_gather_host COMMAND qbat_mapped_gather_host)
set_tests_properties(qbat_mapped_gather_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(QBAT_GATHER_CUDA "Ordered gather and measurements against frozen serial kernels" OFF)
if(QBAT_GATHER_CUDA)
  enable_language(CUDA)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  include("${tl_root}/lib_src/elements/qbat/QbatBatch.cmake")
  add_executable(qbat_mapped_gather_cuda CudaTest.cu)
  target_link_libraries(qbat_mapped_gather_cuda PRIVATE tl_qbat_batch GTest::gtest_main CUDA::cudart)
  set_target_properties(qbat_mapped_gather_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(qbat_mapped_gather_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME qbat_mapped_gather_cuda COMMAND qbat_mapped_gather_cuda)
  set_tests_properties(qbat_mapped_gather_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME qbat_mapped_gather_source_identity
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
