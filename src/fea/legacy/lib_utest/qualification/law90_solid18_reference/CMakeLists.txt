cmake_minimum_required(VERSION 3.22)
project(TLLaw90Solid18Reference LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(law90_solid18_reference_host HostTest.cpp)
target_compile_features(law90_solid18_reference_host PRIVATE cxx_std_17)
target_include_directories(law90_solid18_reference_host PRIVATE "${TL_ROOT}")
target_compile_options(law90_solid18_reference_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(law90_solid18_reference_host PRIVATE GTest::gtest_main)
add_test(NAME law90_solid18_reference_host COMMAND law90_solid18_reference_host)
option(TL_LAW90_SOLID18_NATIVE "Build complete native reference/current oracle" OFF)
option(TL_LAW90_SOLID18_CUDA "Build actual device packet gate" OFF)
set(TL_LAW90_RADIATOR_FIXTURE "" CACHE PATH "Authenticated yaris-radiator-geometry-1 directory")
if(TL_LAW90_SOLID18_CUDA AND NOT TL_LAW90_SOLID18_NATIVE)
  message(FATAL_ERROR "Device gate requires the independent native oracle")
endif()
if(TL_LAW90_SOLID18_NATIVE)
  if(NOT TL_LAW90_RADIATOR_FIXTURE)
    message(FATAL_ERROR "Native gate requires the authenticated original fixture directory")
  endif()
  set(fixture "${CMAKE_CURRENT_BINARY_DIR}/source_fixture")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/source_fixture/prepare_fixture.py"
    --input "${TL_LAW90_RADIATOR_FIXTURE}" --output "${fixture}/RadiatorSourceFixture.h"
    RESULT_VARIABLE result)
  if(NOT result EQUAL 0)
    message(FATAL_ERROR "Original radiator fixture authentication failed")
  endif()
  add_test(NAME law90_solid18_original_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/source_fixture/prepare_fixture.py"
    --input "${TL_LAW90_RADIATOR_FIXTURE}" --output "${fixture}/RadiatorSourceFixture.h" --check)
  add_test(NAME law90_solid18_fixture_reader COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/source_fixture/FixtureReaderTest.py"
    --input "${TL_LAW90_RADIATOR_FIXTURE}")
  enable_language(Fortran)
  add_subdirectory(../solid18_reference/native reference_native)
  add_subdirectory(native)
  add_library(law90_solid18_oracle STATIC NativeOracle.cpp ../solid18_reference/NativeOracle.cpp)
  target_compile_features(law90_solid18_oracle PUBLIC cxx_std_17)
  target_include_directories(law90_solid18_oracle PUBLIC "${TL_ROOT}" "${fixture}")
  target_compile_options(law90_solid18_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(law90_solid18_oracle PUBLIC law90_solid18_current_native)
  foreach(group IN ITEMS Native SourceNative)
    string(TOLOWER "${group}" lower)
    set(target "law90_solid18_reference_${lower}")
    add_executable(${target} "${group}Test.cpp")
    target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off)
    target_link_libraries(${target} PRIVATE law90_solid18_oracle GTest::gtest_main)
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
  endforeach()
endif()
if(TL_LAW90_SOLID18_CUDA)
  enable_language(CUDA)
  add_executable(law90_solid18_reference_cuda CudaTest.cu)
  target_compile_features(law90_solid18_reference_cuda PRIVATE cuda_std_17)
  target_compile_options(law90_solid18_reference_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;-Xptxas=-v;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  target_link_libraries(law90_solid18_reference_cuda PRIVATE law90_solid18_oracle GTest::gtest_main)
  add_test(NAME law90_solid18_reference_cuda COMMAND law90_solid18_reference_cuda)
  set_tests_properties(law90_solid18_reference_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
endif()
add_test(NAME law90_solid18_owning_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
