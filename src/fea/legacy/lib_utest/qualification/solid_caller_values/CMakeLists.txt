cmake_minimum_required(VERSION 3.22)
project(TLSolidCallerValues LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(solid_caller_values_check HostTest.cpp
  ../solid_law36_point/HostTest.cpp ../solid18_force/HostTest.cpp)
target_include_directories(solid_caller_values_check PRIVATE "${TL_ROOT}")
target_compile_features(solid_caller_values_check PRIVATE cxx_std_17)
target_compile_options(solid_caller_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(solid_caller_values_check PRIVATE GTest::gtest_main)
add_test(NAME solid_caller_values COMMAND solid_caller_values_check)
add_test(NAME solid_caller_values_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
set_tests_properties(solid_caller_values solid_caller_values_identity PROPERTIES PROCESSORS 1 RUN_SERIAL TRUE)
