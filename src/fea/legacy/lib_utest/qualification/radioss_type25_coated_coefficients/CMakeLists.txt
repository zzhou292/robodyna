cmake_minimum_required(VERSION 3.22)
project(TLType25CoatedCoefficients LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(TYPE25_COEFFICIENTS_CUDA "Build native coefficient GPU comparisons" OFF)
if(TYPE25_COEFFICIENTS_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_coefficients" ordinary)
set(native "${CMAKE_CURRENT_BINARY_DIR}/native")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Coating source extraction failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/prepare.py native/Coating.F.in
  ../radioss_type25_coefficients/native/prepare.py ../radioss_type25_coefficients/native/original/i25sti3.F)
add_library(coated_coefficient_oracle STATIC NativeOracle.cpp "${native}/Coating.F")
target_link_libraries(coated_coefficient_oracle PUBLIC tl_radioss_type25_coefficients)
target_compile_options(coated_coefficient_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(coated_coefficients_host HostTest.cpp)
target_link_libraries(coated_coefficients_host PRIVATE coated_coefficient_oracle GTest::gtest_main)
target_compile_options(coated_coefficients_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME coated_coefficients_host COMMAND coated_coefficients_host)
add_executable(coated_coefficients_consumer Consumer.cpp)
target_link_libraries(coated_coefficients_consumer PRIVATE tl_radioss_type25_coefficients)
add_test(NAME coated_coefficients_consumer COMMAND coated_coefficients_consumer)
add_test(NAME coated_coefficients_source COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
if(TYPE25_COEFFICIENTS_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(coated_coefficients_cuda CudaTest.cu)
  target_link_libraries(coated_coefficients_cuda PRIVATE coated_coefficient_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(coated_coefficients_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME coated_coefficients_cuda COMMAND coated_coefficients_cuda)
endif()
