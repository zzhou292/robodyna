cmake_minimum_required(VERSION 3.18)
project(tl_type13_qualification LANGUAGES CXX)
enable_testing()
get_filename_component(tl_type13_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_type13_root}/lib_src/elements/type13/Type13Startup.cmake")
find_package(GTest REQUIRED)
add_executable(type13_startup_test PropertyTest.cpp ReferenceTest.cpp)
target_link_libraries(type13_startup_test PRIVATE tl_type13_startup GTest::gtest_main)
target_compile_options(type13_startup_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type13_startup COMMAND type13_startup_test)
option(TYPE13_NATIVE_CHECKS "Compile pinned native TYPE13 startup arithmetic" ON)
if(TYPE13_NATIVE_CHECKS)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME type13_native_source_identity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/native/verify_sources.py")
  add_library(type13_native STATIC native/NativeProperty.F native/NativeReference.F)
  target_include_directories(type13_native PRIVATE "${CMAKE_CURRENT_LIST_DIR}/native")
  target_compile_options(type13_native PRIVATE -cpp -ffixed-line-length-none -fno-fast-math -ffp-contract=off -fcheck=all -finit-real=snan)
  add_executable(type13_native_test NativeTest.cpp NativeOracle.cpp)
  target_link_libraries(type13_native_test PRIVATE tl_type13_startup type13_native GTest::gtest_main)
  target_compile_options(type13_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME type13_native COMMAND type13_native_test)
  add_test(NAME type13_original_fixture_identity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/source_fixture/verify_fixture.py")
  add_executable(type13_source_native_test SourceNativeTest.cpp NativeOracle.cpp)
  target_link_libraries(type13_source_native_test PRIVATE tl_type13_startup type13_native GTest::gtest_main)
  target_compile_options(type13_source_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME type13_original_source_native COMMAND type13_source_native_test)
endif()
option(TYPE13_CUDA_CHECKS "Compile optional CUDA startup value checks (root scheduled)" OFF)
if(TYPE13_CUDA_CHECKS)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(type13_startup_cuda_test StartupCudaTest.cu)
  target_link_libraries(type13_startup_cuda_test PRIVATE tl_type13_startup GTest::gtest_main CUDA::cudart)
  set_target_properties(type13_startup_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(type13_startup_cuda_test PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME type13_startup_cuda COMMAND type13_startup_cuda_test)
  set_tests_properties(type13_startup_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
endif()
