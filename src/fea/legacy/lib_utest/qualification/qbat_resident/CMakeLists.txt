cmake_minimum_required(VERSION 3.22)
project(TLQbatResidentQualification LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/qbat/QbatBatchValues.cmake")
add_executable(qbat_resident_host_test HostTest.cpp)
target_link_libraries(qbat_resident_host_test PRIVATE tl_qbat_batch_values GTest::gtest_main)
add_test(NAME qbat_resident_host COMMAND qbat_resident_host_test)
target_compile_options(qbat_resident_host_test PRIVATE -fno-fast-math -ffp-contract=off)
option(QBAT_RESIDENT_NATIVE "Independent complete native four-point resident packets" OFF)
option(QBAT_RESIDENT_CUDA "Actual complete-formulation owner publication" OFF)
if(QBAT_RESIDENT_NATIVE OR QBAT_RESIDENT_CUDA)
  enable_language(Fortran)
  set(QBAT_FORCE_NATIVE_CHECKS ON CACHE BOOL "" FORCE)
  set(QBAT_FORCE_CUDA_CHECKS OFF CACHE BOOL "" FORCE)
  add_subdirectory("../qbat_force" force)
endif()
if(QBAT_RESIDENT_NATIVE)
  add_executable(qbat_resident_native_test NativeTest.cpp ../qbat_force/NativeOracle.cpp)
  target_link_libraries(qbat_resident_native_test PRIVATE tl_qbat_batch_values qbat_force_native GTest::gtest_main)
  target_compile_options(qbat_resident_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME qbat_resident_native COMMAND qbat_resident_native_test)
endif()
if(QBAT_RESIDENT_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  add_subdirectory("${tl_root}/lib_src/elements/qeph" qeph-values)
  add_subdirectory("${tl_root}/lib_src/elements/t3" t3-values)
  include("${tl_root}/lib_src/elements/qeph/QephBatch.cmake")
  include("${tl_root}/lib_src/elements/t3/T3Batch.cmake")
  include("${tl_root}/lib_src/elements/ShellBatchPublication.cmake")
  include("${CMAKE_CURRENT_LIST_DIR}/Resident.cmake")
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
if(QBAT_RESIDENT_CUDA)
  set_tests_properties(qbat_resident_original_cuda PROPERTIES TIMEOUT 240)
endif()
