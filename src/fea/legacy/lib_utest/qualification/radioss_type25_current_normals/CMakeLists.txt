cmake_minimum_required(VERSION 3.22)
project(TLType25CurrentNormals LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/RadiossType25CurrentNormals.cmake")
# Changed shared float leaves require the complete old startup regression gate.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_fixed_main_startup" startup-regression)
include("${CMAKE_CURRENT_LIST_DIR}/native/Native.cmake")
add_executable(type25_current_normals_host HostTest.cpp CoatedTopologyTest.cpp)
target_link_libraries(type25_current_normals_host PRIVATE tl_radioss_type25_current_normals type25_current_normals_oracle GTest::gtest_main)
target_compile_features(type25_current_normals_host PRIVATE cxx_std_17)
add_test(NAME type25_current_normals_host COMMAND type25_current_normals_host)
add_executable(type25_current_normals_consumer Consumer.cpp)
target_link_libraries(type25_current_normals_consumer PRIVATE tl_radioss_type25_current_normals)
add_test(NAME type25_current_normals_consumer COMMAND type25_current_normals_consumer)
add_test(NAME type25_current_normals_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/check.py")
set_tests_properties(type25_current_normals_host type25_current_normals_consumer type25_current_normals_source
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
