cmake_minimum_required(VERSION 3.22)
project(TLT3OnePointResidentQualification LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../plasticity_binding/PlasticityBinding.cmake")
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/ShellBatchPlasticity.cmake")
add_executable(t3_one_point_resident_host_test HostTest.cpp)
target_link_libraries(t3_one_point_resident_host_test PRIVATE tl_shell_batch_plasticity GTest::gtest_main)
target_compile_options(t3_one_point_resident_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME t3_one_point_resident_host COMMAND t3_one_point_resident_host_test)
set_tests_properties(t3_one_point_resident_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(T3_ONE_POINT_RESIDENT_NATIVE "Original triangle native startup gate" OFF)
option(T3_ONE_POINT_RESIDENT_CUDA "Actual complete mixed T3 one-point transactions" OFF)
if(T3_ONE_POINT_RESIDENT_NATIVE OR T3_ONE_POINT_RESIDENT_CUDA)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(T3_ONE_POINT_RESIDENT_CUDA)
    enable_language(CUDA)
    set(TL_T3_ENABLE_CUDA ON)
    set(TL_T3_ENABLE_FORCE ON)
    set(TL_T3_ENABLE_BATCH ON)
    set(TL_T3_ENABLE_MIXED_BINDING ON)
    set(TL_T3_ENABLE_MIXED ON)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../t3" t3)
  else()
    set(T3_R2_BUILD OFF CACHE BOOL "" FORCE)
    set(T3_R3_BUILD OFF CACHE BOOL "" FORCE)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native/t3" t3-native)
  endif()
  add_executable(t3_one_point_startup_native_test NativeStartupTest.cpp)
  target_link_libraries(t3_one_point_startup_native_test PRIVATE tl_shell_batch_plasticity_binding t3_r1_native GTest::gtest_main)
  target_compile_options(t3_one_point_startup_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME t3_one_point_startup_native COMMAND t3_one_point_startup_native_test)
  add_test(NAME t3_one_point_resident_source_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../t3_one_point/native/verify_sources.py")
endif()
if(T3_ONE_POINT_RESIDENT_CUDA)
  set(TL_CONSTANT_FAILURE_CUDA OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../shell_constant_failure" failure-native)
  include("${CMAKE_CURRENT_LIST_DIR}/../native/law44_one_point/Law44OnePoint.cmake")
  include("${CMAKE_CURRENT_LIST_DIR}/../t3_one_point/OnePointNative.cmake")
  include("${CMAKE_CURRENT_LIST_DIR}/Resident.cmake")
endif()
