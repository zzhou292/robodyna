cmake_minimum_required(VERSION 3.22)
project(TLRearSolid18Force LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(rear18_force_host_check HostTest.cpp)
target_include_directories(rear18_force_host_check PRIVATE "${TL_ROOT}")
target_compile_features(rear18_force_host_check PRIVATE cxx_std_17)
target_compile_options(rear18_force_host_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(rear18_force_host_check PRIVATE GTest::gtest_main)
add_test(NAME rear18_force_host COMMAND rear18_force_host_check)

option(REAR18_FORCE_NATIVE "Complete selected native rear LAW44 recurrence" OFF)
option(REAR18_FORCE_CUDA "Actual device value and rollback" OFF)
set(REAR18_SOURCE_FIXTURE "" CACHE PATH "Authenticated original rear306 binary fixture")
if(REAR18_FORCE_NATIVE OR REAR18_FORCE_CUDA)
  enable_language(Fortran)
  add_subdirectory(../solid_law36_point/native law36-native)
  add_subdirectory(../solid18_reference/native reference-native)
  add_subdirectory(../solid18_force/native legacy-force)
  add_subdirectory(../solid_law44_point/native law44-native)
  add_subdirectory(native)
  add_library(rear18_reference_packet STATIC ../solid18_law44_reference/native/NativePacket.F90
    ../solid18_law44_reference/native/NativeScatter.F)
  add_dependencies(rear18_reference_packet solid18_reference_native)
  target_compile_definitions(rear18_reference_packet PRIVATE MYREAL8 CPP_mach=CPP_p4linux964 COMP_GFORTRAN)
  target_compile_options(rear18_reference_packet PRIVATE -cpp -ffixed-line-length-none -ffree-line-length-none
    -fcheck=bounds -fbacktrace -fno-fast-math -ffp-contract=off)
  target_include_directories(rear18_reference_packet PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}/reference-native/prepared" "${CMAKE_CURRENT_BINARY_DIR}/reference-native/modules")
  target_link_libraries(rear18_reference_packet PUBLIC solid18_reference_native)
  add_library(rear18_force_oracle STATIC NativeOracle.cpp PacketValues.cpp PacketChecks.cpp
    ../solid18_law44_reference/NativeOracle.cpp ../solid18_reference/NativeOracle.cpp)
  target_compile_features(rear18_force_oracle PUBLIC cxx_std_17)
  target_include_directories(rear18_force_oracle PUBLIC "${TL_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_options(rear18_force_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(rear18_force_oracle PUBLIC rear18_force_native rear18_reference_packet GTest::gtest)
  add_executable(rear18_force_native_check NativeTest.cpp)
  target_link_libraries(rear18_force_native_check PRIVATE rear18_force_oracle GTest::gtest_main)
  target_compile_options(rear18_force_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME rear18_force_native COMMAND rear18_force_native_check)
endif()
if(REAR18_SOURCE_FIXTURE)
  set(fixture_header "${CMAKE_CURRENT_BINARY_DIR}/fixture/OriginalFixture.h")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/../solid18_law44_reference/prepare_fixture.py"
    --fixture "${REAR18_SOURCE_FIXTURE}" --output "${fixture_header}" RESULT_VARIABLE status)
  if(NOT status EQUAL 0)
    message(FATAL_ERROR "Original rear306 fixture authentication failed")
  endif()
  if(TARGET rear18_force_native_check)
    target_sources(rear18_force_native_check PRIVATE SourceNativeTest.cpp)
    target_include_directories(rear18_force_native_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture")
  endif()
endif()
if(REAR18_FORCE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(rear18_force_cuda_check CudaTest.cu)
  target_compile_features(rear18_force_cuda_check PRIVATE cuda_std_17)
  target_compile_options(rear18_force_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(rear18_force_cuda_check PRIVATE rear18_force_oracle CUDA::cudart GTest::gtest_main)
  if(REAR18_SOURCE_FIXTURE)
    target_compile_definitions(rear18_force_cuda_check PRIVATE REAR18_ORIGINAL)
    target_include_directories(rear18_force_cuda_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture")
  endif()
  add_test(NAME rear18_force_cuda COMMAND rear18_force_cuda_check)
endif()
add_test(NAME rear18_force_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
