cmake_minimum_required(VERSION 3.22)
project(TLCinPhysicalTimestep LANGUAGES CXX)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
add_executable(cin_physical_timestep_host ValuesTest.cpp)
# Keep the independent 6x6 eigenvalue oracle inexpensive to compile.
set_source_files_properties(MatrixTest.cpp PROPERTIES COMPILE_OPTIONS "-O0"
  COMPILE_DEFINITIONS "EIGEN_UNROLLING_LIMIT=0")
target_include_directories(cin_physical_timestep_host PRIVATE "${TL_ROOT}")
target_link_libraries(cin_physical_timestep_host PRIVATE GTest::gtest_main CUDA::cudart)
target_compile_features(cin_physical_timestep_host PRIVATE cxx_std_17)
target_compile_options(cin_physical_timestep_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME cin_physical_timestep_host COMMAND cin_physical_timestep_host)
# Independent matrix oracle is a separate owning gate. Eigen compiler memory
# exceeds the 512 MiB source-author budget; root runs the enabled default.
option(TL_CIN_PHYSICAL_TIMESTEP_MATRIX "Build independent dense 6x6 eigenvalue oracle" ON)
if(TL_CIN_PHYSICAL_TIMESTEP_MATRIX)
  find_package(Eigen3 REQUIRED)
  add_executable(cin_physical_timestep_matrix MatrixTest.cpp)
  target_include_directories(cin_physical_timestep_matrix PRIVATE "${TL_ROOT}")
  target_link_libraries(cin_physical_timestep_matrix PRIVATE GTest::gtest_main Eigen3::Eigen CUDA::cudart)
  target_compile_features(cin_physical_timestep_matrix PRIVATE cxx_std_17)
  target_compile_options(cin_physical_timestep_matrix PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME cin_physical_timestep_matrix COMMAND cin_physical_timestep_matrix)
endif()
option(TL_CIN_PHYSICAL_TIMESTEP_NATIVE "Build exact independent native scalar/rigid-producer extracts" OFF)
option(TL_CIN_PHYSICAL_TIMESTEP_CUDA "Build actual combined owner rollback gate" OFF)
if(TL_CIN_PHYSICAL_TIMESTEP_NATIVE)
  enable_language(Fortran)
  include(Native.cmake)
endif()
if(TL_CIN_PHYSICAL_TIMESTEP_CUDA)
  enable_language(CUDA)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" owner)
  add_executable(cin_physical_timestep_cuda OwnerTest.cu ../rigid_assembly_owner/Fixture.cpp)
  target_link_libraries(cin_physical_timestep_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main)
  set_target_properties(cin_physical_timestep_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(cin_physical_timestep_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>")
  add_test(NAME cin_physical_timestep_cuda COMMAND cin_physical_timestep_cuda)
  set_tests_properties(cin_physical_timestep_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
