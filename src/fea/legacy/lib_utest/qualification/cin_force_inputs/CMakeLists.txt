cmake_minimum_required(VERSION 3.22)
project(TLCinForceInputs LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/constraints/tied_shell/TiedCinAttachment.cmake")
set(packet_dir "${CMAKE_CURRENT_LIST_DIR}/../cin_parallel_ordinary")
add_executable(cin_force_inputs_host HostTest.cpp SerialHost.cpp "${packet_dir}/Packet.cpp" "${packet_dir}/Compare.cpp")
target_include_directories(cin_force_inputs_host PRIVATE "${tl_root}/lib_src/solvers")
target_link_libraries(cin_force_inputs_host PRIVATE tl_tied_cin_attachment GTest::gtest_main CUDA::toolkit)
target_compile_features(cin_force_inputs_host PRIVATE cxx_std_17)
target_compile_options(cin_force_inputs_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME cin_force_inputs_host COMMAND cin_force_inputs_host)
set_tests_properties(cin_force_inputs_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(CIN_FORCE_INPUTS_CUDA "Compare staged force inputs with the complete frozen caller" OFF)
if(CIN_FORCE_INPUTS_CUDA)
  enable_language(CUDA)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  add_executable(cin_force_inputs_cuda CudaTest.cu Serial.cu
    "${packet_dir}/DevicePacket.cu" "${packet_dir}/Serial.cu"
    "${packet_dir}/Packet.cpp" "${packet_dir}/Compare.cpp")
  target_include_directories(cin_force_inputs_cuda PRIVATE "${tl_root}/lib_src/solvers")
  target_link_libraries(cin_force_inputs_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main CUDA::cudart)
  set_target_properties(cin_force_inputs_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(cin_force_inputs_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>)
  add_test(NAME cin_force_inputs_cuda COMMAND cin_force_inputs_cuda)
  set_tests_properties(cin_force_inputs_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME cin_force_inputs_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(cin_force_inputs_identity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
