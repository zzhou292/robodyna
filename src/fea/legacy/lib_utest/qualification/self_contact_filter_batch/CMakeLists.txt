cmake_minimum_required(VERSION 3.22)
project(TLSelfContactFilterBatch LANGUAGES CXX CUDA)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/SelfContactFilterBatch.cmake")
add_executable(self_contact_filter_batch_values ValueTest.cpp)
target_link_libraries(self_contact_filter_batch_values PRIVATE tl_self_contact_filter_batch_values GTest::gtest_main)
target_compile_options(self_contact_filter_batch_values PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(self_contact_filter_batch_cuda CudaTest.cu)
target_link_libraries(self_contact_filter_batch_cuda PRIVATE tl_self_contact_filter_batch GTest::gtest_main)
set_target_properties(self_contact_filter_batch_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
target_compile_options(self_contact_filter_batch_cuda PRIVATE
  $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
add_test(NAME self_contact_filter_batch_values COMMAND self_contact_filter_batch_values)
add_test(NAME self_contact_filter_batch_cuda COMMAND self_contact_filter_batch_cuda)
add_test(NAME self_contact_filter_batch_source COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/verify_extraction.py")
set_tests_properties(self_contact_filter_batch_values self_contact_filter_batch_source PROPERTIES TIMEOUT 30)
set_tests_properties(self_contact_filter_batch_cuda PROPERTIES TIMEOUT 90 RESOURCE_LOCK gpu0)
