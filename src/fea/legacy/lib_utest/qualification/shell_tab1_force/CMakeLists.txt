cmake_minimum_required(VERSION 3.22)
project(ShellTab1Force LANGUAGES CXX)
option(TL_TAB1_FORCE_NATIVE "Build full native glass Q/T force comparisons" OFF)
option(TL_TAB1_FORCE_CUDA "Build device-owned glass Q/T force comparisons" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
add_executable(shell_tab1_force_values_check Tab1ForceValueTest.cpp)
target_include_directories(shell_tab1_force_values_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_tab1_force_values_check PRIVATE cxx_std_17)
target_compile_options(shell_tab1_force_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_tab1_force_values_check PRIVATE GTest::gtest_main)
add_test(NAME shell_tab1_force_values COMMAND shell_tab1_force_values_check)
set_tests_properties(shell_tab1_force_values PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
if(TL_TAB1_FORCE_NATIVE OR TL_TAB1_FORCE_CUDA)
  include("${CMAKE_CURRENT_SOURCE_DIR}/Native.cmake")
endif()
