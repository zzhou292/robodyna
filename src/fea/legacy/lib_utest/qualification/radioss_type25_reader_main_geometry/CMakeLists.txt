cmake_minimum_required(VERSION 3.22)
project(TLType25ReaderMainGeometry LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TYPE25_READER_GEOMETRY_CUDA "Build actual reader geometry device comparison" OFF)
set(TYPE25_MAIN_GEOMETRY_CUDA ${TYPE25_READER_GEOMETRY_CUDA})
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_main_geometry" exterior-regression)
set(native "${CMAKE_CURRENT_BINARY_DIR}/native")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
 --output "${native}" RESULT_VARIABLE status)
if(NOT status EQUAL 0)
 message(FATAL_ERROR "Pinned reader geometry native preparation failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native/prepare.py
 ../radioss_type25_main_geometry/native/prepare.py ../radioss_type25_main_geometry/native/Reference.F.in)
add_library(type25_reader_geometry_native STATIC NativeOracle.cpp "${native}/Reader.F")
target_link_libraries(type25_reader_geometry_native PUBLIC tl_radioss_type25_main_geometry)
target_compile_options(type25_reader_geometry_native PRIVATE
 "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(type25_reader_geometry_host HostTest.cpp)
target_link_libraries(type25_reader_geometry_host PRIVATE type25_reader_geometry_native type25_main_geometry_oracle GTest::gtest_main)
add_executable(type25_reader_geometry_consumer Consumer.cpp)
target_link_libraries(type25_reader_geometry_consumer PRIVATE tl_radioss_type25_main_geometry)
add_test(NAME type25_reader_geometry_host COMMAND type25_reader_geometry_host)
add_test(NAME type25_reader_geometry_consumer COMMAND type25_reader_geometry_consumer)
add_test(NAME type25_reader_geometry_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
set_tests_properties(type25_reader_geometry_host type25_reader_geometry_consumer type25_reader_geometry_source
 PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
if(TYPE25_READER_GEOMETRY_CUDA)
 enable_language(CUDA)
 find_package(CUDAToolkit REQUIRED)
 add_executable(type25_reader_geometry_cuda CudaTest.cu)
 target_link_libraries(type25_reader_geometry_cuda PRIVATE type25_reader_geometry_native type25_main_geometry_oracle GTest::gtest_main CUDA::cudart)
 set_target_properties(type25_reader_geometry_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
 add_test(NAME type25_reader_geometry_cuda COMMAND type25_reader_geometry_cuda)
 set_tests_properties(type25_reader_geometry_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
endif()
