cmake_minimum_required(VERSION 3.22)
project(TLBeam18Resident LANGUAGES CXX C)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(BEAM18_RESIDENT_CUDA "Actual common-owner and independent native beam gate" OFF)
set(BEAM18_RESIDENT_SOURCE_FIXTURE "" CACHE PATH "Authenticated original142 beam fixture")
if(BEAM18_RESIDENT_SOURCE_FIXTURE AND NOT BEAM18_RESIDENT_CUDA)
  message(FATAL_ERROR "Original resident constructor requires the CUDA/native gate")
endif()
if(BEAM18_RESIDENT_CUDA)
  enable_language(CUDA)
  enable_language(Fortran)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" nodal-owner)
  set(TL_BEAM18_BATCH_CUDA ON)
endif()
add_subdirectory("${TL_ROOT}/lib_src/elements/beam18/resident" resident)
include("${TL_ROOT}/lib_src/constraints/tied_shell/TiedCinAttachment.cmake")
include("${TL_ROOT}/lib_src/constraints/NodalRigidAssemblyBinding.cmake")
add_library(beam18_resident_fixture STATIC OwnerFixture.cpp ../rigid_assembly_owner/Fixture.cpp)
target_link_libraries(beam18_resident_fixture PUBLIC tl_beam18_batch_values
  tl_tied_cin_attachment tl_nodal_rigid_assembly_binding GTest::gtest)
target_compile_options(beam18_resident_fixture PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(beam18_resident_host ValueTest.cpp EndpointTest.cpp)
target_link_libraries(beam18_resident_host PRIVATE beam18_resident_fixture GTest::gtest_main)
target_compile_options(beam18_resident_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME beam18_resident_host COMMAND beam18_resident_host)
add_test(NAME beam18_resident_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
if(BEAM18_RESIDENT_CUDA)
  set(BEAM18_FORCE_NATIVE ON)
  set(BEAM18_FORCE_CUDA OFF)
  set(BEAM18_FORCE_SOURCE_FIXTURE "${BEAM18_RESIDENT_SOURCE_FIXTURE}")
  add_subdirectory(../beam18_force force-native)
  add_executable(beam18_resident_cuda CudaFixture.cu NativeChecks.cpp AssemblyChecks.cu OwnerTest.cu)
  target_link_libraries(beam18_resident_cuda PRIVATE beam18_resident_fixture tl_beam18_batch
    beam18_force_oracle GTest::gtest_main)
  set_target_properties(beam18_resident_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  if(BEAM18_RESIDENT_SOURCE_FIXTURE)
    target_sources(beam18_resident_cuda PRIVATE OriginalTest.cpp)
    target_include_directories(beam18_resident_cuda PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/force-native/fixture")
  endif()
  target_compile_options(beam18_resident_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>)
  add_test(NAME beam18_resident_cuda COMMAND beam18_resident_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
