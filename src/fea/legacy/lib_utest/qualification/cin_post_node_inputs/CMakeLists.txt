cmake_minimum_required(VERSION 3.22)
project(TLCinPostNodeInputs LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/constraints/tied_shell/TiedCinAttachment.cmake")
set(packet "${CMAKE_CURRENT_LIST_DIR}/../cin_parallel_ordinary")
set(prior "${CMAKE_CURRENT_LIST_DIR}/../cin_force_inputs")
add_executable(cin_post_node_inputs_host HostTest.cpp "${prior}/HostTest.cpp" "${prior}/SerialHost.cpp" "${packet}/Packet.cpp" "${packet}/Compare.cpp")
target_include_directories(cin_post_node_inputs_host PRIVATE "${tl_root}" "${tl_root}/lib_src/solvers")
target_link_libraries(cin_post_node_inputs_host PRIVATE tl_tied_cin_attachment GTest::gtest_main CUDA::toolkit)
target_compile_features(cin_post_node_inputs_host PRIVATE cxx_std_17)
target_compile_options(cin_post_node_inputs_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME cin_post_node_inputs_host COMMAND cin_post_node_inputs_host)
add_test(NAME cin_post_node_inputs_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(cin_post_node_inputs_host cin_post_node_inputs_source PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(CIN_POST_NODE_INPUTS_CUDA "Frozen complete caller qualification" OFF)
if(CIN_POST_NODE_INPUTS_CUDA)
  enable_language(CUDA)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  add_executable(cin_post_node_inputs_cuda CudaTest.cu "${prior}/CudaTest.cu" "${prior}/Serial.cu"
    "${packet}/DevicePacket.cu" "${packet}/Serial.cu" "${packet}/Packet.cpp" "${packet}/Compare.cpp")
  target_include_directories(cin_post_node_inputs_cuda PRIVATE "${tl_root}" "${tl_root}/lib_src/solvers")
  target_link_libraries(cin_post_node_inputs_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main CUDA::cudart)
  set_target_properties(cin_post_node_inputs_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(cin_post_node_inputs_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>)
  add_test(NAME cin_post_node_inputs_cuda COMMAND cin_post_node_inputs_cuda)
  set_tests_properties(cin_post_node_inputs_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
endif()
