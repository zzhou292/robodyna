cmake_minimum_required(VERSION 3.22)
project(TLTiedSearchFinalization LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/constraints/tied_shell/search/TiedSearchFinalization.cmake")
find_package(GTest REQUIRED)
add_executable(tied_search_finalization_values_test ValueTest.cpp)
target_link_libraries(tied_search_finalization_values_test PRIVATE tl_tied_search_finalization GTest::gtest_main)
add_test(NAME tied_search_finalization_values COMMAND tied_search_finalization_values_test)
option(TL_TIED_FINALIZATION_NATIVE "Build complete independent I2TID3 oracle" OFF)
if(TL_TIED_FINALIZATION_NATIVE)
  enable_language(Fortran)
  include("${CMAKE_CURRENT_LIST_DIR}/Native.cmake")
  add_executable(tied_search_finalization_native_test NativeTest.cpp)
  target_link_libraries(tied_search_finalization_native_test PRIVATE tied_search_finalization_oracle GTest::gtest_main)
  add_test(NAME tied_search_finalization_native COMMAND tied_search_finalization_native_test)
  add_test(NAME tied_search_finalization_sources COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/verify_sources.py")
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
