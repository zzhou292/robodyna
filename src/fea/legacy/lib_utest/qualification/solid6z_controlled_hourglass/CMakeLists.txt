cmake_minimum_required(VERSION 3.22)
project(TLS6ControlledAdapter LANGUAGES CXX C Fortran)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TL_S6_CONTROL_CUDA "Qualify controlled S6 on actual CUDA" OFF)
set(TL_SOLID6Z_FORCE_NATIVE ON CACHE BOOL "Legacy native regression" FORCE)
set(TL_SOLID6Z_FORCE_CUDA ${TL_S6_CONTROL_CUDA} CACHE BOOL "Legacy CUDA regression" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../solid6z_force" legacy_s6)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../controlled_hourglass/native" raw_hourglass)
add_subdirectory(native)
add_executable(s6_control_host HostTest.cpp)
target_compile_features(s6_control_host PRIVATE cxx_std_17)
target_include_directories(s6_control_host PRIVATE "${TL_ROOT}")
target_compile_options(s6_control_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(s6_control_host PRIVATE s6_control_native GTest::gtest_main Eigen3::Eigen)
add_test(NAME s6_control_host COMMAND s6_control_host)
add_test(NAME s6_control_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/source_check.py")
if(TL_S6_CONTROL_CUDA)
 enable_language(CUDA)
 add_executable(s6_control_cuda CudaTest.cu)
 target_compile_features(s6_control_cuda PRIVATE cuda_std_17)
 target_include_directories(s6_control_cuda PRIVATE "${TL_ROOT}")
 target_compile_options(s6_control_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true -Xcompiler=-fno-fast-math,-ffp-contract=off)
 target_link_libraries(s6_control_cuda PRIVATE s6_control_native GTest::gtest_main Eigen3::Eigen)
 add_test(NAME s6_control_cuda COMMAND s6_control_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
