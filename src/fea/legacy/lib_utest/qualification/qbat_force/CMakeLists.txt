cmake_minimum_required(VERSION 3.22)
project(tl_qbat_force_qualification LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/qbat/QbatForce.cmake")
find_package(GTest REQUIRED)
add_executable(qbat_force_host_test MathTest.cpp)
target_link_libraries(qbat_force_host_test PRIVATE tl_qbat_force GTest::gtest_main)
target_compile_options(qbat_force_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qbat_force_host COMMAND qbat_force_host_test)

option(QBAT_FORCE_NATIVE_CHECKS "Compile complete selected four-point native force caller" ON)
if(QBAT_FORCE_NATIVE_CHECKS)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(QBAT_NATIVE_CHECKS ON CACHE BOOL "" FORCE)
  set(QBAT_CUDA_CHECKS OFF CACHE BOOL "" FORCE)
  set(TL_CONSTANT_FAILURE_CUDA OFF CACHE BOOL "" FORCE)
  add_subdirectory("../qbat" geometry)
  add_subdirectory("../shell_constant_failure" failure)
  add_test(NAME qbat_force_source_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/native/verify_sources.py")
  set(native "${CMAKE_CURRENT_LIST_DIR}/native")
  include("${CMAKE_CURRENT_LIST_DIR}/../native/law44_one_point/Law44OnePoint.cmake")
  get_target_property(modules law44_one_point_native Fortran_MODULE_DIRECTORY)
  add_library(qbat_force_native STATIC native/NativeKinematics.F native/NativeParent.F
    native/NativeForce.F native/original/cbafori.F native/original/cbaener.F
    native/original/cbavisc.F native/original/cbaproj.F native/original/cbastra3.F)
  add_dependencies(qbat_force_native law44_one_point_native qbat_native)
  target_link_libraries(qbat_force_native PUBLIC qbat_native law44_one_point_native)
  foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(value qbat_native ${property})
    set_property(TARGET qbat_force_native PROPERTY ${property} "${value}")
  endforeach()
  target_include_directories(qbat_force_native PRIVATE "${native}" "${modules}")
  target_compile_definitions(qbat_force_native PRIVATE CBAFORI=QBF_CBAFORI CBAFORI1=QBF_CBAFORI1
    CBAFORCT=QBF_CBAFORCT CBAFORRZ=QBF_CBAFORRZ CBAENER=QBF_CBAENER CBAENERS=QBF_CBAENERS
    CBAVISC=QBF_CBAVISC CBAVISNP1=QBF_CBAVISNP1 CBAPROJ=QBF_CBAPROJ CBASTRA3=QBF_CBASTRA3)
  add_executable(qbat_force_native_test NativeTest.cpp NativeOracle.cpp OriginalSourceTest.cpp)
  target_link_libraries(qbat_force_native_test PRIVATE tl_qbat_force qbat_force_native GTest::gtest_main)
  target_compile_options(qbat_force_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME qbat_force_native COMMAND qbat_force_native_test)
  # Directly affected old shell-entry material/rate paths remain in this gate.
  set(law "${CMAKE_CURRENT_LIST_DIR}/../native/law44")
  foreach(suite point rate analytic)
    if(suite STREQUAL "point")
      set(sources "${law}/NativePointTest.cpp" "${law}/NativePhysicalThicknessTest.cpp")
    elseif(suite STREQUAL "rate")
      set(sources "${law}/NativeRateTest.cpp" "${law}/NativeRateSectionTest.cpp")
    else()
      set(sources "${law}/NativeAnalyticTest.cpp" "${law}/NativeAnalyticSectionTest.cpp")
    endif()
    add_executable(qbat_legacy_${suite}_test ${sources})
    target_link_libraries(qbat_legacy_${suite}_test PRIVATE law44_point_native GTest::gtest_main)
    target_compile_options(qbat_legacy_${suite}_test PRIVATE -fno-fast-math -ffp-contract=off)
    add_test(NAME qbat_legacy_${suite} COMMAND qbat_legacy_${suite}_test)
  endforeach()
endif()

option(QBAT_FORCE_CUDA_CHECKS "Build root-scheduled actual four-point history checks" OFF)

  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)

  target_link_libraries(qbat_force_cuda_test PRIVATE tl_qbat_force GTest::gtest_main CUDA::cudart)
  set_target_properties(qbat_force_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(qbat_force_cuda_test PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME qbat_force_cuda COMMAND qbat_force_cuda_test)
endif()
