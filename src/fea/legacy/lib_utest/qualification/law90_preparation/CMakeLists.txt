cmake_minimum_required(VERSION 3.22)
project(TLLaw90Preparation LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(law90_target name)
  target_include_directories(${name} PRIVATE "${TL_ROOT}")
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_link_libraries(${name} PRIVATE GTest::gtest_main)
  target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
endfunction()
add_executable(law90_preparation_host_check HostTest.cpp)
law90_target(law90_preparation_host_check)
add_test(NAME law90_preparation_host COMMAND law90_preparation_host_check)
add_test(NAME law90_preparation_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
option(LAW90_NATIVE "Independent complete native preparation/curve oracle" OFF)
option(LAW90_CUDA "Actual CUDA values against the independent native packet" OFF)
if(LAW90_NATIVE OR LAW90_CUDA)
  enable_language(Fortran)
  add_subdirectory(native)
  add_executable(law90_preparation_native_check NativeTest.cpp)
  law90_target(law90_preparation_native_check)
  target_link_libraries(law90_preparation_native_check PRIVATE law90_preparation_native)
  add_test(NAME law90_preparation_native COMMAND law90_preparation_native_check)
endif()
if(LAW90_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(law90_preparation_cuda_check CudaTest.cu)
  law90_target(law90_preparation_cuda_check)
  target_compile_features(law90_preparation_cuda_check PRIVATE cuda_std_17)
  target_compile_options(law90_preparation_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(law90_preparation_cuda_check PRIVATE law90_preparation_native CUDA::cudart)
  add_test(NAME law90_preparation_cuda COMMAND law90_preparation_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
