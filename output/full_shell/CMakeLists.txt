cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaFullShellRecords LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
if(NOT TARGET robo_dyna_bounded_array_io)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../arrays" "${CMAKE_CURRENT_BINARY_DIR}/arrays")
endif()
if(NOT TARGET robo_dyna_interval_values)
  add_subdirectory(interval_values)
endif()
add_library(robo_dyna_full_shell_records STATIC FullShellVisualizationSchema.cpp FullShellVisualizationPlan.cpp
  FullShellFrameDescription.cpp FullShellVisualizationRecords.cpp FullShellIdentityFields.cpp)
target_link_libraries(robo_dyna_full_shell_records PUBLIC robo_dyna_bounded_array_io robo_dyna_interval_values)
target_compile_features(robo_dyna_full_shell_records PUBLIC cxx_std_17)
target_compile_options(robo_dyna_full_shell_records PRIVATE -fno-fast-math -ffp-contract=off)
add_library(robo_dyna_interval_segments STATIC IntervalSequence.cpp IntervalSegmentChecks.cpp
  IntervalSegmentFields.cpp IntervalSegmentReader.cpp IntervalSegmentWriter.cpp)
target_link_libraries(robo_dyna_interval_segments PUBLIC robo_dyna_full_shell_records)
target_compile_features(robo_dyna_interval_segments PUBLIC cxx_std_17)
target_compile_options(robo_dyna_interval_segments PRIVATE -fno-fast-math -ffp-contract=off)
option(ROBO_DYNA_FULL_SHELL_RECORD_TESTS "Build pure array/forecast/record contract checks" ON)
if(ROBO_DYNA_FULL_SHELL_RECORD_TESTS)
  find_package(GTest REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(robo_dyna_full_shell_record_check tests/BoundedArrayTest.cpp
    tests/FullShellPlanTest.cpp tests/FullShellRecordTest.cpp)
  target_link_libraries(robo_dyna_full_shell_record_check PRIVATE robo_dyna_full_shell_records GTest::gtest_main)
  target_compile_options(robo_dyna_full_shell_record_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_executable(robo_dyna_interval_segment_check tests/IntervalSegmentTest.cpp tests/IntervalWriterFailureTest.cpp)
  target_link_libraries(robo_dyna_interval_segment_check PRIVATE robo_dyna_interval_segments GTest::gtest_main)
  target_compile_options(robo_dyna_interval_segment_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME full_shell_interval_segments COMMAND robo_dyna_interval_segment_check)
  set_tests_properties(full_shell_interval_segments PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
  option(ROBO_DYNA_INTERVAL_FACTORY_CHECKS "Compile the shared native-stamp interval factory without solver/runtime" OFF)
  if(ROBO_DYNA_INTERVAL_FACTORY_CHECKS)
    find_package(CUDAToolkit REQUIRED)
    set(ROBO_DYNA_TL_ROOT "" CACHE PATH "Explicit TL-FEA headers for host interval factory checks")
    if(NOT EXISTS "${ROBO_DYNA_TL_ROOT}/lib_src/solvers/FENodalState.h")
      message(FATAL_ERROR "Interval factory checks need explicit ROBO_DYNA_TL_ROOT")
    endif()
    add_executable(robo_dyna_interval_factory_check tests/IntervalFactoryTest.cpp ../source_assembly/SourceAssemblyWallInterval.cpp)
    target_include_directories(robo_dyna_interval_factory_check PRIVATE
      "${ROBO_DYNA_TL_ROOT}" "${ROBO_DYNA_TL_ROOT}/lib_src" "${CUDAToolkit_INCLUDE_DIRS}")
    target_link_libraries(robo_dyna_interval_factory_check PRIVATE robo_dyna_interval_values GTest::gtest_main)
    target_compile_features(robo_dyna_interval_factory_check PRIVATE cxx_std_17)
    target_compile_options(robo_dyna_interval_factory_check PRIVATE -fno-fast-math -ffp-contract=off)
    add_test(NAME full_shell_interval_factory COMMAND robo_dyna_interval_factory_check)
    set_tests_properties(full_shell_interval_factory PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
  endif()
  add_test(NAME full_shell_visualization_records COMMAND robo_dyna_full_shell_record_check
    --gtest_filter=-BoundedArrays.CanonicalImporterInteroperability)
  set_tests_properties(full_shell_visualization_records PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
  add_test(NAME full_shell_canonical_array_compatibility COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/tests/canonical_array_compatibility.py"
    "${CMAKE_CURRENT_LIST_DIR}/../.." "$<TARGET_FILE:robo_dyna_full_shell_record_check>")
  set_tests_properties(full_shell_canonical_array_compatibility PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
endif()
