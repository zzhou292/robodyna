cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)

  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/ActivityRecords.cmake")

add_executable(robo_dyna_parent_activity_check tests/ActivityRecordTest.cpp tests/ActivityCorruptionTest.cpp tests/ActivityPlanTest.cpp)

add_test(NAME full_shell_parent_activity COMMAND robo_dyna_parent_activity_check)
set_tests_properties(full_shell_parent_activity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 90)
