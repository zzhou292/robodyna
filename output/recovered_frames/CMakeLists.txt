cmake_minimum_required(VERSION 3.22)
project(RoboDynaRecoveredFrames LANGUAGES CXX)
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
enable_testing()
set(ROBO_DYNA_TL_ROOT "" CACHE PATH "Qualified TL source for bounded arena")
if(NOT EXISTS "${ROBO_DYNA_TL_ROOT}/lib_utils/BoundedArena.h")
  message(FATAL_ERROR "Recovery requires an explicit TL source root")
endif()
include(RecoveredFrames.cmake)
add_executable(robo_dyna_recovered_frames_check tests/ValuesTest.cpp)
target_link_libraries(robo_dyna_recovered_frames_check PRIVATE robo_dyna_recovered_frames GTest::gtest_main)
target_compile_options(robo_dyna_recovered_frames_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME recovered_frames_values COMMAND robo_dyna_recovered_frames_check)
set_tests_properties(recovered_frames_values PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
add_executable(recover_physical_samples cli/Main.cpp)
target_link_libraries(recover_physical_samples PRIVATE robo_dyna_recovered_frames)
target_compile_options(recover_physical_samples PRIVATE -fno-fast-math -ffp-contract=off)
