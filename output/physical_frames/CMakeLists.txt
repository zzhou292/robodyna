cmake_minimum_required(VERSION 3.22)
project(RoboDynaPhysicalAcceptedFrames LANGUAGES CXX)
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
option(ROBO_DYNA_PHYSICAL_CAPTURE_RUNTIME "Build actual accepted physical-owner capture" OFF)
if(ROBO_DYNA_PHYSICAL_CAPTURE_RUNTIME)
  enable_language(CUDA)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/PhysicalFrames.cmake")
add_executable(robo_dyna_physical_capture_values_check tests/ValuesTest.cpp tests/JointPhaseTest.cpp)
target_link_libraries(robo_dyna_physical_capture_values_check PRIVATE robo_dyna_physical_frame_values GTest::gtest_main)
target_compile_options(robo_dyna_physical_capture_values_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME physical_capture_values COMMAND robo_dyna_physical_capture_values_check)
set_tests_properties(physical_capture_values PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
option(ROBO_DYNA_PHYSICAL_CAPTURE_ORIGINAL "Author the full original initial snapshot gate" OFF)
if(ROBO_DYNA_PHYSICAL_CAPTURE_ORIGINAL)
  if(NOT ROBO_DYNA_PHYSICAL_CAPTURE_RUNTIME)
    message(FATAL_ERROR "Original snapshot gate requires the actual runtime capture")
  endif()
  foreach(name CANONICAL SCOPE DECLARATIONS GLASS_RESOLUTION TYPE13_DECLARATION)
    set(ROBO_DYNA_VEHICLE_${name} "" CACHE PATH "Explicit original vehicle ${name} fixture")
    if(NOT EXISTS "${ROBO_DYNA_VEHICLE_${name}}")
      message(FATAL_ERROR "Physical capture requires ${name}")
    endif()
  endforeach()
  set(ROBO_DYNA_VEHICLE_GLASS_SHA256 "" CACHE STRING "Authenticated glass sidecar SHA256")
  string(LENGTH "${ROBO_DYNA_VEHICLE_GLASS_SHA256}" sha_length)
  if(NOT sha_length EQUAL 64)
    message(FATAL_ERROR "Physical capture requires glass SHA256")
  endif()
  add_executable(robo_dyna_physical_capture_original_check tests/OriginalTest.cpp
    "${CMAKE_CURRENT_LIST_DIR}/../../case/vehicle_startup/physical_attachments/tests/Source.cpp")
  target_link_libraries(robo_dyna_physical_capture_original_check PRIVATE robo_dyna_physical_accepted_frames GTest::gtest_main)
  target_compile_options(robo_dyna_physical_capture_original_check PRIVATE -fno-fast-math -ffp-contract=off)
  foreach(gate forecast initial)
    if(gate STREQUAL forecast)
      set(filter "PhysicalCaptureOriginal.MappingAndArchiveForecast")
    else()
      set(filter "PhysicalCaptureOriginal.InitialCompleteSnapshotAndBinaryRoundTrip")
    endif()
    add_test(NAME physical_capture_${gate} COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_LIST_DIR}/../../case/vehicle_startup/tied_classification/tests/actual_fixture.py"
      "${ROBO_DYNA_VEHICLE_CANONICAL}" "${ROBO_DYNA_VEHICLE_SCOPE}" "${ROBO_DYNA_VEHICLE_DECLARATIONS}"
      "$<TARGET_FILE:robo_dyna_physical_capture_original_check>" "${filter}")
    set_tests_properties(physical_capture_${gate} PROPERTIES TIMEOUT 600 RUN_SERIAL TRUE PROCESSORS 2
      ENVIRONMENT "ROBO_VEHICLE_DECLARATIONS=${ROBO_DYNA_VEHICLE_DECLARATIONS};ROBO_VEHICLE_GLASS_RESOLUTION=${ROBO_DYNA_VEHICLE_GLASS_RESOLUTION};ROBO_VEHICLE_GLASS_SHA256=${ROBO_DYNA_VEHICLE_GLASS_SHA256};ROBO_DYNA_TYPE13_DECLARATION=${ROBO_DYNA_VEHICLE_TYPE13_DECLARATION}")
  endforeach()
endif()
