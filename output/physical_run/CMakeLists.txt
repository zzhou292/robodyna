cmake_minimum_required(VERSION 3.22)
project(RoboDynaPhysicalRun LANGUAGES CXX)
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
enable_testing()
option(ROBO_DYNA_PHYSICAL_RUN_LIVE_FACTORY "Build actual dynamics accepted interval factory" OFF)
option(ROBO_DYNA_PHYSICAL_RUN_ORIGINAL "Build original-source archive gates; root execution only" OFF)
option(ROBO_DYNA_PHYSICAL_RUN_WALL "Build named actual wall archive adapter" OFF)
if(ROBO_DYNA_PHYSICAL_RUN_WALL AND NOT ROBO_DYNA_PHYSICAL_RUN_LIVE_FACTORY)
  message(FATAL_ERROR "Physical wall archive requires the actual live factory")
endif()
if(ROBO_DYNA_PHYSICAL_RUN_LIVE_FACTORY)
  enable_language(CUDA)
endif()
set(ROBO_DYNA_TL_ROOT "" CACHE PATH "Qualified TL source root for shared bounded arena")
if(NOT EXISTS "${ROBO_DYNA_TL_ROOT}/lib_utils/BoundedArena.h")
  message(FATAL_ERROR "Physical run needs the explicit qualified TL root")
endif()
include(PhysicalRun.cmake)
add_executable(robo_dyna_physical_run_check tests/ValuesTest.cpp tests/ReplayBudgetTest.cpp tests/RecordsTest.cpp tests/FailureTest.cpp tests/WallTest.cpp tests/ViewerInputTest.cpp
  tests/SelfContactRecordsTest.cpp tests/NativeContactRecordsTest.cpp tests/NativeGroupRecordsTest.cpp tests/EnvironmentRecordsTest.cpp tests/EnvironmentMeshTest.cpp)
target_link_libraries(robo_dyna_physical_run_check PRIVATE robo_dyna_physical_run_records GTest::gtest_main)
target_compile_options(robo_dyna_physical_run_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME physical_run_records COMMAND robo_dyna_physical_run_check)
set_tests_properties(physical_run_records PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
add_executable(robo_dyna_wall_composition_check tests/WallCompositionTest.cpp)
target_link_libraries(robo_dyna_wall_composition_check PRIVATE robo_dyna_wall_composition GTest::gtest_main)
target_compile_options(robo_dyna_wall_composition_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME physical_wall_composition COMMAND robo_dyna_wall_composition_check)
set_tests_properties(physical_wall_composition PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
if(ROBO_DYNA_PHYSICAL_RUN_ORIGINAL)
  include(Original.cmake)
endif()
