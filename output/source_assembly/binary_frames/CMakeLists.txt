if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaComponentBinaryFrames LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  find_package(CUDAToolkit REQUIRED)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../SourceAssemblyWallValues.cmake")
if(NOT TARGET robo_dyna_full_shell_records)
  set(ROBO_DYNA_FULL_SHELL_RECORD_TESTS OFF)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../full_shell" "${CMAKE_CURRENT_BINARY_DIR}/records")
endif()
add_library(robo_dyna_component_binary_values STATIC
  ComponentMapping.cpp ComponentContext.cpp ComponentFrameFields.cpp)
target_link_libraries(robo_dyna_component_binary_values PUBLIC
  robo_dyna_full_shell_records robo_dyna_source_assembly_wall_values)
target_compile_options(robo_dyna_component_binary_values PRIVATE -fno-fast-math -ffp-contract=off)
find_package(GTest REQUIRED)
set(ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY "" CACHE FILEPATH "Frozen six-part source inventory")
set(ROBO_DYNA_SOURCE_ASSEMBLY_WALL "" CACHE FILEPATH "Original finite mesh wall")
foreach(required IN ITEMS ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY ROBO_DYNA_SOURCE_ASSEMBLY_WALL)
  if(NOT EXISTS "${${required}}")
    message(FATAL_ERROR "Component binary checks need explicit ${required}")
  endif()
endforeach()
set(component_binary_environment
  "ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY}"
  "ROBO_DYNA_SOURCE_ASSEMBLY_WALL=${ROBO_DYNA_SOURCE_ASSEMBLY_WALL}")
add_executable(robo_dyna_component_binary_values_check tests/ComponentFrameValuesTest.cpp
  "${CMAKE_CURRENT_LIST_DIR}/../tests/SourceAssemblyWallFieldTest.cpp")
target_link_libraries(robo_dyna_component_binary_values_check PRIVATE robo_dyna_component_binary_values GTest::gtest_main)
target_compile_options(robo_dyna_component_binary_values_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME component_binary_values COMMAND robo_dyna_component_binary_values_check)
set_tests_properties(component_binary_values PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
  ENVIRONMENT "${component_binary_environment}")
option(ROBO_DYNA_COMPONENT_BINARY_LIVE "Build actual accepted six/seven-part CUDA parity gate" OFF)

  enable_language(CUDA)
  set(CRASH_TL_FEA_SOURCE_DIR "${ROBO_DYNA_TL_ROOT}")
  if(NOT TARGET tl_explicit_nodal_state)
    add_subdirectory("${ROBO_DYNA_TL_ROOT}/lib_src/solvers" tl-owner)
  endif()
  include("${CMAKE_CURRENT_LIST_DIR}/../SourceAssemblyWallArtifacts.cmake")
  add_library(robo_dyna_component_binary_frames STATIC SourceAssemblyBinaryFrames.cpp)
  target_link_libraries(robo_dyna_component_binary_frames PUBLIC
    robo_dyna_component_binary_values robo_dyna_source_assembly_wall_adapter)
  target_compile_options(robo_dyna_component_binary_frames PRIVATE -fno-fast-math -ffp-contract=off)
  add_executable(robo_dyna_component_binary_live_check tests/ComponentFrameLiveTest.cpp
    "${ROBO_DYNA_TL_ROOT}/lib_utest/qualification/nodal_rigid_group/PreparedSnapshotCudaProbe.cpp")
  target_link_libraries(robo_dyna_component_binary_live_check PRIVATE robo_dyna_component_binary_frames GTest::gtest_main CUDA::cudart)
  target_link_options(robo_dyna_component_binary_live_check PRIVATE -Wl,--wrap=cudaMemcpyAsync)
  target_compile_options(robo_dyna_component_binary_live_check PRIVATE -fno-fast-math -ffp-contract=off)
  set(ROBO_DYNA_SOURCE_BRACKET_INVENTORY "" CACHE FILEPATH "Frozen seven-part source inventory")
  if(ROBO_DYNA_SOURCE_BRACKET_INVENTORY)
    if(NOT EXISTS "${ROBO_DYNA_SOURCE_BRACKET_INVENTORY}")

    target_sources(robo_dyna_component_binary_live_check PRIVATE tests/ComponentFrameBracketTest.cpp)
    list(APPEND component_binary_environment "ROBO_DYNA_SOURCE_BRACKET_INVENTORY=${ROBO_DYNA_SOURCE_BRACKET_INVENTORY}")
  endif()
  add_test(NAME component_binary_live COMMAND robo_dyna_component_binary_live_check)
  set_tests_properties(component_binary_live PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 600
    ENVIRONMENT "${component_binary_environment}")
