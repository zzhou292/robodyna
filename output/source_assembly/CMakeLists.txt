cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaSourceAssemblyOutput LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/SourceAssemblyOutput.cmake")
find_package(GTest REQUIRED)
set(ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY "" CACHE FILEPATH "Frozen authenticated six-part Yaris inventory")
if(NOT EXISTS "${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY}")
  message(FATAL_ERROR "Assembly output checks require the explicit frozen inventory")
endif()
add_executable(robo_dyna_source_assembly_output_check tests/SourceAssemblySurfaceTest.cpp tests/SourceAssemblySectionFieldsTest.cpp)
target_link_libraries(robo_dyna_source_assembly_output_check PRIVATE robo_dyna_source_assembly_fields GTest::gtest_main)
add_test(NAME source_assembly_output_host COMMAND robo_dyna_source_assembly_output_check)
set_tests_properties(source_assembly_output_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
  ENVIRONMENT "ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY}")
option(ROBO_DYNA_SOURCE_ASSEMBLY_OUTPUT_CUDA_CHECK "Opt in to actual source accepted-output GPU qualification" OFF)
if(ROBO_DYNA_SOURCE_ASSEMBLY_OUTPUT_CUDA_CHECK)
  include("${CMAKE_CURRENT_LIST_DIR}/SourceAssemblyOutputCudaCheck.cmake")
endif()
