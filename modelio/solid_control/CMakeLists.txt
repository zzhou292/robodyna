cmake_minimum_required(VERSION 3.22)
project(RoboDynaSolidControlSource LANGUAGES CXX)
enable_testing()
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
set(ROBO_DYNA_TL_ROOT "" CACHE PATH "Qualified TL source")
include("${CMAKE_CURRENT_LIST_DIR}/SolidControlSource.cmake")
add_executable(solid_control_source_values tests/SourceValuesTest.cpp)
target_link_libraries(solid_control_source_values PRIVATE robo_dyna_solid_control_source GTest::gtest_main)
target_compile_options(solid_control_source_values PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME solid_control_source_values COMMAND solid_control_source_values)
set_tests_properties(solid_control_source_values PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
