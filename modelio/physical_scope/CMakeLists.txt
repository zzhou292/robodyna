cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaPhysicalScope LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/PhysicalScope.cmake")
find_package(GTest REQUIRED)
add_executable(robo_dyna_physical_scope_fields_check tests/FieldsTest.cpp)
target_link_libraries(robo_dyna_physical_scope_fields_check PRIVATE robo_dyna_physical_scope GTest::gtest_main)
add_test(NAME physical_scope_fields COMMAND robo_dyna_physical_scope_fields_check)
option(ROBO_DYNA_PHYSICAL_SCOPE_ACTUAL_TESTS "Enable complete original source coverage census" OFF)
if(ROBO_DYNA_PHYSICAL_SCOPE_ACTUAL_TESTS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(name CANONICAL SCOPE DECLARATIONS TYPE13_DECLARATION)
    set(ROBO_DYNA_PHYSICAL_${name} "" CACHE PATH "Explicit original ${name} fixture")
    if(NOT EXISTS "${ROBO_DYNA_PHYSICAL_${name}}")
      message(FATAL_ERROR "Physical census requires ${name}")
    endif()
  endforeach()
  set(ROBO_DYNA_PHYSICAL_REPORT "" CACHE FILEPATH "Optional create-only report output")
  add_executable(robo_dyna_physical_scope_source_check tests/ActualSourceTest.cpp)
  target_link_libraries(robo_dyna_physical_scope_source_check PRIVATE robo_dyna_physical_scope GTest::gtest_main)
  add_test(NAME physical_scope_source COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../../output/full_shell/static_bundle/tests/actual_source_fixture.py"
    "${ROBO_DYNA_PHYSICAL_CANONICAL}" "${ROBO_DYNA_PHYSICAL_SCOPE}"
    "$<TARGET_FILE:robo_dyna_physical_scope_source_check>" "PhysicalScopeOriginal.*")
  set_tests_properties(physical_scope_source PROPERTIES ENVIRONMENT
    "ROBO_VEHICLE_DECLARATIONS=${ROBO_DYNA_PHYSICAL_DECLARATIONS};ROBO_DYNA_TYPE13_DECLARATION=${ROBO_DYNA_PHYSICAL_TYPE13_DECLARATION};ROBO_PHYSICAL_SCOPE_REPORT=${ROBO_DYNA_PHYSICAL_REPORT}")
endif()
get_property(physical_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${physical_tests} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE PROCESSORS 1)
