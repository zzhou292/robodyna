cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaSourceAssemblyInput LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/SourceAssembly.cmake")
find_package(GTest REQUIRED)
set(ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY "" CACHE FILEPATH "Frozen authenticated six-part Yaris assembly inventory")
if(NOT EXISTS "${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY}")
  message(FATAL_ERROR "The assembly input check requires the explicit frozen inventory path")
endif()
add_executable(robo_dyna_source_assembly_input_check
  tests/SourceAssemblyReaderTest.cpp tests/SourceAssemblyInputTest.cpp tests/SourceAssemblyRejectionTest.cpp)
target_link_libraries(robo_dyna_source_assembly_input_check PRIVATE
  robo_dyna_source_assembly_shell_input robo_dyna_source_assembly_material_input GTest::gtest_main)
target_compile_options(robo_dyna_source_assembly_input_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME source_assembly_input COMMAND robo_dyna_source_assembly_input_check)
set_tests_properties(source_assembly_input PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
  ENVIRONMENT "ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY}")
set(ROBO_DYNA_SOURCE_ANALYTIC_INVENTORY "" CACHE FILEPATH "Optional original analytic LAW44 source V2 fixture")
set(ROBO_DYNA_SOURCE_ANALYTIC_MIXED_INVENTORY "" CACHE FILEPATH "Optional mixed original analytic/table V2 fixture")
if(ROBO_DYNA_SOURCE_ANALYTIC_INVENTORY)
  if(NOT EXISTS "${ROBO_DYNA_SOURCE_ANALYTIC_INVENTORY}" OR NOT EXISTS "${ROBO_DYNA_SOURCE_ANALYTIC_MIXED_INVENTORY}")
    message(FATAL_ERROR "Both explicit analytic source fixtures must exist")
  endif()
  include("${ROBO_DYNA_TL_ROOT}/lib_src/elements/ShellBatchPlasticityBinding.cmake")
  add_executable(robo_dyna_source_analytic_input_check tests/SourceAnalyticInputTest.cpp)
  target_link_libraries(robo_dyna_source_analytic_input_check PRIVATE robo_dyna_source_assembly_shell_input
    robo_dyna_source_assembly_material_input tl_shell_batch_plasticity_binding GTest::gtest_main)
  target_compile_options(robo_dyna_source_analytic_input_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME source_analytic_input COMMAND robo_dyna_source_analytic_input_check)
  set_tests_properties(source_analytic_input PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
    ENVIRONMENT "ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY};ROBO_DYNA_SOURCE_ANALYTIC_INVENTORY=${ROBO_DYNA_SOURCE_ANALYTIC_INVENTORY};ROBO_DYNA_SOURCE_ANALYTIC_MIXED_INVENTORY=${ROBO_DYNA_SOURCE_ANALYTIC_MIXED_INVENTORY}")
endif()
set(ROBO_DYNA_SOURCE_BRACKET_INVENTORY "" CACHE FILEPATH "Optional frozen original seven-part bracket inventory")
if(ROBO_DYNA_SOURCE_BRACKET_INVENTORY)
  if(NOT EXISTS "${ROBO_DYNA_SOURCE_BRACKET_INVENTORY}")
    message(FATAL_ERROR "Declared bracket inventory is missing")
  endif()
  add_executable(robo_dyna_source_bracket_input_check tests/SourceBracketInputTest.cpp)
  target_link_libraries(robo_dyna_source_bracket_input_check PRIVATE
    robo_dyna_source_assembly_shell_input robo_dyna_source_assembly_material_input GTest::gtest_main)
  target_compile_options(robo_dyna_source_bracket_input_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME source_bracket_input COMMAND robo_dyna_source_bracket_input_check)
  set_tests_properties(source_bracket_input PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
    ENVIRONMENT "ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY};ROBO_DYNA_SOURCE_BRACKET_INVENTORY=${ROBO_DYNA_SOURCE_BRACKET_INVENTORY}")
  include("${CMAKE_CURRENT_LIST_DIR}/SourceAssemblySpotweldInput.cmake")
  add_executable(robo_dyna_source_bracket_spotweld_check tests/SourceBracketSpotweldTest.cpp)
  target_link_libraries(robo_dyna_source_bracket_spotweld_check PRIVATE
    robo_dyna_source_assembly_spotweld_input GTest::gtest_main)
  target_compile_options(robo_dyna_source_bracket_spotweld_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME source_bracket_spotweld COMMAND robo_dyna_source_bracket_spotweld_check)
  set_tests_properties(source_bracket_spotweld PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
    ENVIRONMENT "ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY=${ROBO_DYNA_SOURCE_ASSEMBLY_INVENTORY};ROBO_DYNA_SOURCE_BRACKET_INVENTORY=${ROBO_DYNA_SOURCE_BRACKET_INVENTORY}")
endif()
