cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaVehiclePointMassSource LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/VehiclePointMassSource.cmake")
find_package(GTest REQUIRED)
add_executable(robo_dyna_vehicle_point_mass_fields_check tests/FieldsTest.cpp)
target_link_libraries(robo_dyna_vehicle_point_mass_fields_check PRIVATE robo_dyna_vehicle_point_mass_source GTest::gtest_main)
target_compile_options(robo_dyna_vehicle_point_mass_fields_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME vehicle_point_mass_fields COMMAND robo_dyna_vehicle_point_mass_fields_check)
option(ROBO_DYNA_VEHICLE_POINT_MASS_ACTUAL_TESTS "Original declared-domain point-mass source gate" OFF)
if(ROBO_DYNA_VEHICLE_POINT_MASS_ACTUAL_TESTS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(name CANONICAL SCOPE DECLARATIONS TYPE13_DECLARATION)
    set(ROBO_DYNA_PHYSICAL_${name} "" CACHE PATH "Explicit original ${name} fixture")
    if(NOT EXISTS "${ROBO_DYNA_PHYSICAL_${name}}")
      message(FATAL_ERROR "Vehicle point-mass source requires ${name}")
    endif()
  endforeach()
  add_executable(robo_dyna_vehicle_point_mass_source_check tests/ActualSourceTest.cpp)
  target_link_libraries(robo_dyna_vehicle_point_mass_source_check PRIVATE robo_dyna_vehicle_point_mass_source GTest::gtest_main)
  target_compile_options(robo_dyna_vehicle_point_mass_source_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME vehicle_point_mass_source COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../../output/full_shell/static_bundle/tests/actual_source_fixture.py"
    "${ROBO_DYNA_PHYSICAL_CANONICAL}" "${ROBO_DYNA_PHYSICAL_SCOPE}"
    "$<TARGET_FILE:robo_dyna_vehicle_point_mass_source_check>" "VehiclePointMassOriginal.*")
  set_tests_properties(vehicle_point_mass_source PROPERTIES ENVIRONMENT
    "ROBO_VEHICLE_DECLARATIONS=${ROBO_DYNA_PHYSICAL_DECLARATIONS};ROBO_DYNA_TYPE13_DECLARATION=${ROBO_DYNA_PHYSICAL_TYPE13_DECLARATION}")
endif()
get_property(point_mass_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${point_mass_tests} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE PROCESSORS 1)
