cmake_minimum_required(VERSION 3.18)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(RoboDynaVehiclePhysicalDomain LANGUAGES CXX)
  find_package(Chrono REQUIRED CONFIG)
  enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/VehiclePhysicalDomain.cmake")
find_package(GTest REQUIRED)
add_executable(robo_dyna_vehicle_physical_domain_fields_check tests/SelectionTest.cpp tests/ProfileTest.cpp)
target_link_libraries(robo_dyna_vehicle_physical_domain_fields_check PRIVATE robo_dyna_vehicle_physical_domain GTest::gtest_main)
target_compile_options(robo_dyna_vehicle_physical_domain_fields_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME vehicle_physical_domain_fields COMMAND robo_dyna_vehicle_physical_domain_fields_check)
option(ROBO_DYNA_VEHICLE_PHYSICAL_DOMAIN_ACTUAL_TESTS "Original complete physical domain source gate" OFF)
if(ROBO_DYNA_VEHICLE_PHYSICAL_DOMAIN_ACTUAL_TESTS)
  include("${CMAKE_CURRENT_LIST_DIR}/../point_mass/VehiclePointMassSource.cmake")
  include("${CMAKE_CURRENT_LIST_DIR}/../type25/VehicleType25Source.cmake")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  foreach(name CANONICAL SCOPE DECLARATIONS TYPE13_DECLARATION)
    set(ROBO_DYNA_PHYSICAL_${name} "" CACHE PATH "Explicit original ${name} fixture")
    if(NOT EXISTS "${ROBO_DYNA_PHYSICAL_${name}}")
      message(FATAL_ERROR "Vehicle physical domain requires ${name}")
    endif()
  endforeach()
  add_executable(robo_dyna_vehicle_physical_domain_source_check tests/ActualSourceTest.cpp)
  target_link_libraries(robo_dyna_vehicle_physical_domain_source_check PRIVATE robo_dyna_vehicle_physical_domain
    robo_dyna_vehicle_point_mass_source robo_dyna_vehicle_type25_source GTest::gtest_main)
  target_compile_options(robo_dyna_vehicle_physical_domain_source_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME vehicle_physical_domain_source COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../../output/full_shell/static_bundle/tests/actual_source_fixture.py"
    "${ROBO_DYNA_PHYSICAL_CANONICAL}" "${ROBO_DYNA_PHYSICAL_SCOPE}"
    "$<TARGET_FILE:robo_dyna_vehicle_physical_domain_source_check>" "VehiclePhysicalDomainOriginal.*")
  set_tests_properties(vehicle_physical_domain_source PROPERTIES ENVIRONMENT
    "ROBO_VEHICLE_DECLARATIONS=${ROBO_DYNA_PHYSICAL_DECLARATIONS};ROBO_DYNA_TYPE13_DECLARATION=${ROBO_DYNA_PHYSICAL_TYPE13_DECLARATION}")
endif()
get_property(domain_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${domain_tests} PROPERTIES TIMEOUT 240 RUN_SERIAL TRUE PROCESSORS 1)
