cmake_minimum_required(VERSION 3.22)
project(RoboDynaCorrectedNodalSource LANGUAGES CXX Fortran)
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
get_filename_component(ROBO_CORRECTION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
include("${CMAKE_CURRENT_LIST_DIR}/../CorrectedNodalSource.cmake")
# Keep the already-qualified pre-correction and source-selection regressions.
set(ROBO_DYNA_VEHICLE_CONTACT_SEED_NATIVE_TESTS ON)

  add_subdirectory("${ROBO_DYNA_TL_ROOT}/lib_utest/qualification/law90_preparation/native" law90-native)

add_executable(corrected_nodal_source_values tests/SharingTest.cpp tests/MaterialTest.cpp)

  robo_dyna_law42_contact_slots_native law90_preparation_native GTest::gtest_main)
target_compile_options(corrected_nodal_source_values PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME corrected_nodal_source_values COMMAND corrected_nodal_source_values)
set_tests_properties(corrected_nodal_source_values PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
add_executable(corrected_nodal_source_consumer tests/Consumer.cpp)
target_link_libraries(corrected_nodal_source_consumer PRIVATE robo_dyna_vehicle_corrected_nodal_source)
add_test(NAME corrected_nodal_source_consumer COMMAND corrected_nodal_source_consumer)
option(ROBO_DYNA_CORRECTED_NODAL_ACTUAL_TESTS "Build explicit complete V5 corrected nodal source gate" OFF)
if(ROBO_DYNA_CORRECTED_NODAL_ACTUAL_TESTS)

  target_link_libraries(corrected_nodal_source_actual PRIVATE robo_dyna_vehicle_corrected_nodal_source GTest::gtest_main)

  # Explicit guarded forecast/preparation commands own the source-sized run.
