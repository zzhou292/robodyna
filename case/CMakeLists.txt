find_package(GTest REQUIRED)
set(CRASH_CANONICAL_WALL "" CACHE FILEPATH "Prepared canonical Yaris wall manifest")
if(NOT EXISTS "${CRASH_CANONICAL_WALL}")
  message(FATAL_ERROR "CRASH_CANONICAL_WALL must name the prepared manifest; no fetch or skipped gate")
endif()
add_library(crash_canonical_wall STATIC CanonicalWall.cpp)
target_compile_features(crash_canonical_wall PUBLIC cxx_std_17)
target_include_directories(crash_canonical_wall PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(crash_canonical_wall PRIVATE Chrono::Chrono_core)
target_compile_options(crash_canonical_wall PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(crash_canonical_wall_check canonical_wall_check.cpp)
target_link_libraries(crash_canonical_wall_check PRIVATE crash_canonical_wall Chrono::Chrono_core GTest::gtest)
add_test(NAME canonical_wall_reader COMMAND crash_canonical_wall_check "${CRASH_CANONICAL_WALL}")
set_tests_properties(canonical_wall_reader PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE PROCESSORS 1)
