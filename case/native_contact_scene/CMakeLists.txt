option(ROBO_DYNA_NATIVE_CAPTURE_RUNTIME "Build actual native TYPE25 accepted capture" OFF)

  message(FATAL_ERROR "Physical source qualification requires explicit declared scene export")

  ENVIRONMENT "ROBO_DYNA_NATIVE_SCENE_EXPORT=${ROBO_DYNA_NATIVE_SCENE_EXPORT}")
