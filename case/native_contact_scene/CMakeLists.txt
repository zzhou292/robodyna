cmake_minimum_required(VERSION 3.22)
project(RoboNativeScenePhysicalSource LANGUAGES CXX)
enable_testing()
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
set(ROBO_DYNA_TL_ROOT "" CACHE PATH "Explicit qualified TL source with empty owner scopes")
include("${CMAKE_CURRENT_LIST_DIR}/PhysicalSource.cmake")
set(ROBO_DYNA_NATIVE_SCENE_EXPORT "" CACHE FILEPATH "Pinned capped declared-scene.json")
if(NOT EXISTS "${ROBO_DYNA_NATIVE_SCENE_EXPORT}")
  message(FATAL_ERROR "Physical source qualification requires explicit declared scene export")
endif()
add_executable(native_scene_physical_source_test PhysicalSourceTest.cpp)
target_link_libraries(native_scene_physical_source_test PRIVATE robo_dyna_native_scene_physical_source GTest::gtest_main)
target_compile_options(native_scene_physical_source_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME native_scene_physical_source COMMAND native_scene_physical_source_test)
set_tests_properties(native_scene_physical_source PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120
  ENVIRONMENT "ROBO_DYNA_NATIVE_SCENE_EXPORT=${ROBO_DYNA_NATIVE_SCENE_EXPORT}")
