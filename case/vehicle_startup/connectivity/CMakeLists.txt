cmake_minimum_required(VERSION 3.22)
project(RoboDynaVehicleConnectivity LANGUAGES CXX)
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
enable_testing()
set(ROBO_DYNA_TL_ROOT "" CACHE PATH "Explicit TL source root")
if(NOT EXISTS "${ROBO_DYNA_TL_ROOT}/lib_src/assembly/NodalNodeDomain.h")
  message(FATAL_ERROR "Connectivity requires ROBO_DYNA_TL_ROOT")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/ConnectivityValues.cmake")
add_executable(robo_dyna_vehicle_connectivity_values_check tests/ComponentsTest.cpp)
target_link_libraries(robo_dyna_vehicle_connectivity_values_check PRIVATE robo_dyna_connectivity_values GTest::gtest_main)
add_test(NAME vehicle_connectivity_values COMMAND robo_dyna_vehicle_connectivity_values_check)
option(ROBO_DYNA_CONNECTIVITY_ORIGINAL "Complete physical original-source gate; root lane" OFF)
if(ROBO_DYNA_CONNECTIVITY_ORIGINAL)
  enable_language(CUDA)
  include("${CMAKE_CURRENT_LIST_DIR}/Actual.cmake")
endif()
