cmake_minimum_required(VERSION 3.22)
project(RoboDynaVehicleContactGeometry LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
set(ROBO_DYNA_TL_ROOT "" CACHE PATH "Explicit TL-FEA source with Vehicle wall weights")
if(NOT EXISTS "${ROBO_DYNA_TL_ROOT}/lib_src/collision/NodalWallWeightLimits.h")
    message(FATAL_ERROR "Vehicle contact geometry needs explicit TL Vehicle weight support")
endif()
include("ShellCollectionContactGeometry.cmake")
add_executable(robo_dyna_vehicle_contact_geometry_check tests/VehicleContactGeometryTest.cpp)
target_include_directories(robo_dyna_vehicle_contact_geometry_check PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(robo_dyna_vehicle_contact_geometry_check PRIVATE robo_dyna_shell_collection_contact_geometry GTest::gtest_main)
target_compile_options(robo_dyna_vehicle_contact_geometry_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME vehicle_contact_geometry COMMAND robo_dyna_vehicle_contact_geometry_check)
set_tests_properties(vehicle_contact_geometry PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
