cmake_minimum_required(VERSION 3.18)
project(tl_type25_qualification LANGUAGES CXX)
enable_testing()
get_filename_component(tl_type25_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_type25_root}/lib_src/elements/type25/Type25Model.cmake")
find_package(GTest REQUIRED)
add_executable(type25_startup_test ModelTest.cpp FrameTest.cpp)
target_link_libraries(type25_startup_test PRIVATE tl_type25_model GTest::gtest_main)
target_compile_options(type25_startup_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type25_startup COMMAND type25_startup_test)
option(TYPE25_NATIVE_CHECKS "Compile retained source arithmetic and host math tests" ON)
if(TYPE25_NATIVE_CHECKS)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  include("${tl_type25_root}/lib_src/elements/type25/Type25Math.cmake")
  add_test(NAME type25_native_source_identity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/native/verify_sources.py")
  add_library(type25_native STATIC native/NativeReference.F native/NativeFrame.F native/NativeDeformation.F
    native/NativeResponse.F native/NativeScatter.F native/NativeCoefficients.F)
  target_include_directories(type25_native PRIVATE "${CMAKE_CURRENT_LIST_DIR}/native" "${CMAKE_CURRENT_LIST_DIR}/native/stubs")
  target_compile_options(type25_native PRIVATE -cpp -ffixed-line-length-none -fno-fast-math -ffp-contract=off -fcheck=all -finit-real=snan)
  add_executable(type25_math_test NativeOracle.cpp NativeTest.cpp MathTest.cpp)
  target_link_libraries(type25_math_test PRIVATE tl_type25_math type25_native GTest::gtest_main)
  target_compile_options(type25_math_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME type25_math COMMAND type25_math_test)
endif()
option(TYPE25_CUDA_CHECKS "Compile CUDA pure-math qualification (root-scheduled)" OFF)
if(TYPE25_CUDA_CHECKS)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  include("${tl_type25_root}/lib_src/elements/type25/Type25Math.cmake")
  add_executable(type25_math_cuda_test MathCudaTest.cu)
  target_link_libraries(type25_math_cuda_test PRIVATE tl_type25_math GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_math_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(type25_math_cuda_test PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME type25_math_cuda COMMAND type25_math_cuda_test)
  set_tests_properties(type25_math_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
endif()
