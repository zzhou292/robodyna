cmake_minimum_required(VERSION 3.22)
project(TLDiagnosticChannelFolds LANGUAGES CXX CUDA)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(oracle "${CMAKE_CURRENT_BINARY_DIR}/oracle")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" --output "${oracle}" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/../finalizer_local_control/generate_oracles.py" --output "${oracle}/qbat" COMMAND_ERROR_IS_FATAL ANY)
add_library(channel_values STATIC
 "${TL_ROOT}/lib_src/elements/solids/resident/Identity.cpp"
 "${TL_ROOT}/lib_src/elements/qbat/QbatBatchArena.cpp"
 "${TL_ROOT}/lib_src/elements/qbat/QbatBatchIdentity.cpp")
target_include_directories(channel_values PUBLIC "${TL_ROOT}" ${CUDAToolkit_INCLUDE_DIRS})
target_link_libraries(channel_values PUBLIC Eigen3::Eigen)
target_compile_features(channel_values PUBLIC cxx_std_17)
target_compile_options(channel_values PRIVATE -fno-fast-math -ffp-contract=off)
foreach(family IN ITEMS Solid Qbat)
 string(TOLOWER "${family}" lower)
 foreach(kind IN ITEMS Host Cuda)
  string(TOLOWER "${kind}" suffix)
  if(kind STREQUAL "Host")
   set(source "${family}HostTest.cpp")
  else()
   set(source "${family}CudaTest.cu")
  endif()
  set(target "${lower}_channels_${suffix}")
  add_executable(${target} "${source}")
  target_include_directories(${target} PRIVATE "${oracle}/${lower}")
  target_link_libraries(${target} PRIVATE channel_values GTest::gtest_main CUDA::cudart)
  target_compile_options(${target} PRIVATE
   $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>
   $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;--ptxas-options=-v;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME ${target} COMMAND ${target})
 endforeach()
endforeach()
target_sources(qbat_channels_cuda PRIVATE "${TL_ROOT}/lib_src/elements/qbat/mapped/Measurement.cu")
add_test(NAME diagnostic_channels_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py")
get_property(checks DIRECTORY PROPERTY TESTS)
set_tests_properties(${checks} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
