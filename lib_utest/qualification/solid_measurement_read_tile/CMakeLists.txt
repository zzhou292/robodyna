cmake_minimum_required(VERSION 3.22)
project(TLSolidMeasurementReadTile LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(oracle "${CMAKE_CURRENT_BINARY_DIR}/oracle")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" "${CMAKE_CURRENT_LIST_DIR}/source-manifest.json"
  "${TL_ROOT}/lib_src/elements/solids/resident/Candidate.cu")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" --output "${oracle}" COMMAND_ERROR_IS_FATAL ANY)
add_library(solid_read_tile_identity STATIC "${TL_ROOT}/lib_src/elements/solids/resident/Identity.cpp")
target_include_directories(solid_read_tile_identity PUBLIC "${TL_ROOT}" "${oracle}" ${CUDAToolkit_INCLUDE_DIRS})
target_compile_features(solid_read_tile_identity PUBLIC cxx_std_17)
target_compile_options(solid_read_tile_identity PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(solid_read_tile_host HostTest.cpp)
target_link_libraries(solid_read_tile_host PRIVATE solid_read_tile_identity GTest::gtest_main)
target_compile_options(solid_read_tile_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME solid_read_tile_host COMMAND solid_read_tile_host)
add_test(NAME solid_read_tile_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py")
option(SOLID_READ_TILE_CUDA "Literal full finalizer and complete five-family controls" OFF)
if(SOLID_READ_TILE_CUDA)
  enable_language(CUDA)
  add_executable(solid_read_tile_cuda CudaTest.cu)
  target_link_libraries(solid_read_tile_cuda PRIVATE solid_read_tile_identity GTest::gtest_main CUDA::cudart)
  set_target_properties(solid_read_tile_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(solid_read_tile_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;--ptxas-options=-v;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  add_test(NAME solid_read_tile_cuda COMMAND solid_read_tile_cuda)
endif()
# Reuse every assertion in the earlier real all-family Control/repair suite.
# Its current production wrappers now use the actual64-lane launch; frozen
# wrappers stay single-threaded. Existing serial host checks use the exact
# current fallback generated above, not a fabricated cooperative host kernel.
option(SOLID_READ_TILE_REAL_CUDA "Real all-family arenas and prior complete-Control fixtures" OFF)
if(SOLID_READ_TILE_REAL_CUDA)
  if(NOT SOLID_READ_TILE_CUDA)
    message(FATAL_ERROR "Real CUDA gate requires the focused CUDA option")
  endif()
  set(legacy "${CMAKE_CURRENT_LIST_DIR}/../finalizer_local_control")
  set(validation "${CMAKE_CURRENT_LIST_DIR}/../solid_candidate_validation")
  foreach(generator IN ITEMS "${validation}/generate_oracle.py" "${legacy}/generate_oracles.py" "${CMAKE_CURRENT_LIST_DIR}/source_proof.py")
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${generator}" --output "${oracle}" COMMAND_ERROR_IS_FATAL ANY)
  endforeach()
  add_subdirectory("${TL_ROOT}/lib_src/elements/solids/resident" real-values)
  add_executable(solid_read_tile_real_host "${legacy}/SolidHostTest.cpp")
  add_executable(solid_read_tile_real_cuda "${legacy}/SolidCudaTest.cu" "${legacy}/SolidCurrentCandidate.cu"
    "${validation}/FrozenCandidate.cu" "${TL_ROOT}/lib_src/elements/solids/resident/ResultValidation.cu")
  foreach(target IN ITEMS solid_read_tile_real_host solid_read_tile_real_cuda)
    target_include_directories(${target} PRIVATE "${TL_ROOT}" "${oracle}")
    target_link_libraries(${target} PRIVATE tl_solid_batch_values GTest::gtest_main CUDA::cudart)
    target_compile_options(${target} PRIVATE
      $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>
      $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
    set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    add_test(NAME ${target} COMMAND ${target})
  endforeach()
endif()
get_property(checks DIRECTORY PROPERTY TESTS)
set_tests_properties(${checks} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
