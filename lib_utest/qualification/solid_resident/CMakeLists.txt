cmake_minimum_required(VERSION 3.22)
project(TLSolidResident LANGUAGES CXX C)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(SOLID_RESIDENT_CUDA "Build actual owner and independent native solid recurrence gates" OFF)
option(SOLID_RESIDENT_ORIGINAL "Include all2412 original constructor values (root lane)" OFF)
if(SOLID_RESIDENT_ORIGINAL AND NOT SOLID_RESIDENT_CUDA)
  message(FATAL_ERROR "Original resident qualification requires its CUDA/native gate")
endif()
if(SOLID_RESIDENT_CUDA)
  enable_language(CUDA)
  enable_language(Fortran)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" nodal-owner)
  set(TL_SOLID_BATCH_CUDA ON)
endif()
add_subdirectory("${TL_ROOT}/lib_src/elements/solids/resident" "${CMAKE_CURRENT_BINARY_DIR}/resident")
include("${TL_ROOT}/lib_src/constraints/tied_shell/TiedCinAttachment.cmake")
include("${TL_ROOT}/lib_src/constraints/NodalRigidAssemblyBinding.cmake")
add_executable(solid_resident_host HostTest.cpp FixtureTest.cpp ../rigid_assembly_owner/Fixture.cpp)
target_link_libraries(solid_resident_host PRIVATE tl_solid_batch_values tl_tied_cin_attachment
  tl_nodal_rigid_assembly_binding GTest::gtest_main)
target_compile_options(solid_resident_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME solid_resident_host COMMAND solid_resident_host)
set_tests_properties(solid_resident_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
if(SOLID_RESIDENT_CUDA)
  set(TL_SOLID_FORCE_STARTUP_NATIVE ON)
  set(TL_SOLID_FORCE_STARTUP_CUDA OFF)
  add_subdirectory(../solid_force_startup startup-native)
  add_library(solid_resident_native_checks STATIC NativeTrajectory.cpp
    Native18Checks.cpp Native24Checks.cpp Native6zChecks.cpp)
  target_link_libraries(solid_resident_native_checks PUBLIC solid_startup_oracles tl_solid_batch_values)
  target_compile_options(solid_resident_native_checks PRIVATE -fno-fast-math -ffp-contract=off)
  add_executable(solid_resident_cuda OwnerFixture.cu AssemblyChecks.cu OwnerTest.cu
    RejectionTest.cu ScopeTest.cu ../rigid_assembly_owner/Fixture.cpp)
  if(SOLID_RESIDENT_ORIGINAL)
    target_sources(solid_resident_cuda PRIVATE OriginalTest.cpp)
    add_test(NAME solid_resident_rubber_identity COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/../solid_common/source_fixture/verify_fixture.py")
    add_test(NAME solid_resident_adhesive_identity COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/../solid18_reference/verify_source.py")
    add_test(NAME solid_resident_material_identity COMMAND "${Python3_EXECUTABLE}" -B
      "${CMAKE_CURRENT_SOURCE_DIR}/../solid_law36_point/verify_source.py")
  endif()
  target_link_libraries(solid_resident_cuda PRIVATE tl_solid_batch solid_resident_native_checks GTest::gtest_main)
  set_target_properties(solid_resident_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(solid_resident_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>)
  add_test(NAME solid_resident_cuda COMMAND solid_resident_cuda)
  set_tests_properties(solid_resident_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 300)
endif()
