if(NOT TARGET solid_law44_native)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../solid_law44_point/native" "${CMAKE_CURRENT_BINARY_DIR}/law44")
endif()
if(NOT TARGET beam18_native)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../beam18_reference/native" "${CMAKE_CURRENT_BINARY_DIR}/reference")
endif()
get_target_property(law44_modules solid_law44_native Fortran_MODULE_DIRECTORY)
set(prepared "${CMAKE_CURRENT_BINARY_DIR}/prepared")
set(modules "${CMAKE_CURRENT_BINARY_DIR}/modules")
file(MAKE_DIRECTORY "${modules}")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
  --output "${prepared}" RESULT_VARIABLE status)
if(NOT status EQUAL 0)
  message(FATAL_ERROR "Beam18 force donor authentication failed")
endif()
set(leaves main_beam18 mulaw_ib pevec3 pdefo3 pcurv3 pdlen3 pdamp3 pfint3 pfcum3 pmcum3 sigeps44pi pibuf3)
set(sources)
foreach(leaf IN LISTS leaves)
  list(APPEND sources "${prepared}/${leaf}.F")
endforeach()
add_library(beam18_force_native STATIC Context.F90 Excluded.F90 NativeCaller.F ${sources})
set_target_properties(beam18_force_native PROPERTIES Fortran_MODULE_DIRECTORY "${modules}")
add_dependencies(beam18_force_native solid_law44_native)
target_include_directories(beam18_force_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/context" "${law44_modules}" "${modules}")
target_compile_options(beam18_force_native PRIVATE -cpp -ffixed-line-length-none -ffree-line-length-none
  -fcheck=bounds -fbacktrace -fno-fast-math -ffp-contract=off)
target_link_libraries(beam18_force_native PUBLIC solid_law44_native beam18_native)
add_test(NAME beam18_force_native_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py" --output "${prepared}" --check)
