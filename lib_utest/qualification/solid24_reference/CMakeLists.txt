cmake_minimum_required(VERSION 3.22)
project(TLSolid24Reference LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(solid24_reference_host HostTest.cpp)
target_compile_features(solid24_reference_host PRIVATE cxx_std_17)
target_include_directories(solid24_reference_host PRIVATE "${TL_ROOT}")
target_compile_options(solid24_reference_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(solid24_reference_host PRIVATE GTest::gtest_main)
add_test(NAME solid24_reference_host COMMAND solid24_reference_host)
option(TL_SOLID24_REFERENCE_NATIVE "Build selected native startup oracle" OFF)
option(TL_SOLID24_REFERENCE_CUDA "Build CUDA versus native startup gate" OFF)
if(TL_SOLID24_REFERENCE_CUDA AND NOT TL_SOLID24_REFERENCE_NATIVE)
  message(FATAL_ERROR "CUDA gate requires the native oracle")
endif()
if(TL_SOLID24_REFERENCE_NATIVE)
  enable_language(Fortran)
  add_subdirectory(native)
  add_executable(solid24_reference_native_test NativeTest.cpp SourceNativeTest.cpp)
  target_include_directories(solid24_reference_native_test PRIVATE "${TL_ROOT}")
  target_compile_features(solid24_reference_native_test PRIVATE cxx_std_17)
  target_compile_options(solid24_reference_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(solid24_reference_native_test PRIVATE solid24_reference_native GTest::gtest_main)
  add_test(NAME solid24_reference_native_test COMMAND solid24_reference_native_test)
endif()
if(TL_SOLID24_REFERENCE_CUDA)
  enable_language(CUDA)
  add_executable(solid24_reference_cuda CudaTest.cu)
  target_include_directories(solid24_reference_cuda PRIVATE "${TL_ROOT}")
  target_compile_features(solid24_reference_cuda PRIVATE cuda_std_17)
  target_compile_options(solid24_reference_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-fno-fast-math,-ffp-contract=off)
  target_link_libraries(solid24_reference_cuda PRIVATE solid24_reference_native GTest::gtest_main)
  add_test(NAME solid24_reference_cuda COMMAND solid24_reference_cuda)
endif()
add_test(NAME solid24_reference_fixture_identity
  COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/../solid_common/source_fixture/verify_fixture.py")
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
