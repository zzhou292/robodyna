cmake_minimum_required(VERSION 3.22)
project(TLType25PostGapmStartup LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TYPE25_CURRENT_NORMALS_CUDA "Build actual shared GPU normal stages" OFF)
if(TYPE25_CURRENT_NORMALS_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/RadiossType25InterfaceSurface.cmake")
# Reuse complete old startup/current-normal qualification and unchanged donors.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_current_normals" current-regression)
include("${CMAKE_CURRENT_LIST_DIR}/native/Native.cmake")
add_executable(type25_post_gapm_host HostTest.cpp NodePrefixTest.cpp)
target_link_libraries(type25_post_gapm_host PRIVATE tl_radioss_type25_interface_surface
  tl_radioss_type25_current_normals type25_post_gapm_oracle type25_current_normals_oracle GTest::gtest_main)
target_compile_features(type25_post_gapm_host PRIVATE cxx_std_17)
add_test(NAME type25_post_gapm_host COMMAND type25_post_gapm_host)
add_executable(type25_post_gapm_consumer Consumer.cpp)
target_link_libraries(type25_post_gapm_consumer PRIVATE tl_radioss_type25_current_normals)
add_test(NAME type25_post_gapm_consumer COMMAND type25_post_gapm_consumer)
add_test(NAME type25_post_gapm_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/check.py")
set_tests_properties(type25_post_gapm_host type25_post_gapm_consumer type25_post_gapm_source
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
if(TYPE25_CURRENT_NORMALS_CUDA)
  add_executable(type25_post_gapm_cuda CudaTest.cu)
  target_link_libraries(type25_post_gapm_cuda PRIVATE tl_radioss_type25_interface_surface
    tl_radioss_type25_current_normals type25_post_gapm_oracle type25_current_normals_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_post_gapm_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME type25_post_gapm_cuda COMMAND type25_post_gapm_cuda)
  set_tests_properties(type25_post_gapm_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
endif()
