cmake_minimum_required(VERSION 3.22)
project(TLNativeMultiInterface LANGUAGES C CXX CUDA)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(PHYSICAL_PUBLICATION_CUDA ON)
set(NATIVE_CONTACT_LEGACY_CUDA ON)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../physical_publication" publication-regression)
set(TYPE25_SEARCH_CUDA ON)
include("${TL_ROOT}/lib_src/collision/RadiossType25Transaction.cmake")
set(TYPE25_NATIVE_SCENE_APP_ROOT "" CACHE PATH "Pinned qualified app material bridge")
if(NOT EXISTS "${TYPE25_NATIVE_SCENE_APP_ROOT}/modelio/native_contact_scene/MaterialBridge.cmake")
  message(FATAL_ERROR "Native group tests require the explicit qualified material bridge source")
endif()
set(ROBO_DYNA_TL_ROOT "${TL_ROOT}" CACHE PATH "Explicit TL checkout" FORCE)
include("${TYPE25_NATIVE_SCENE_APP_ROOT}/modelio/native_contact_scene/MaterialBridge.cmake")
set(runtime "${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_runtime")
set(observed "${CMAKE_CURRENT_BINARY_DIR}/observed-fixture")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${runtime}/prepare_fixture.py"
  --output "${observed}" RESULT_VARIABLE status)
if(NOT status EQUAL 0)
  message(FATAL_ERROR "Native fixture source preparation failed")
endif()
file(GLOB fixture_inputs CONFIGURE_DEPENDS "${runtime}/fixture/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${fixture_inputs} "${runtime}/prepare_fixture.py")
add_executable(native_group_host HostTest.cpp)
target_link_libraries(native_group_host PRIVATE tl_radioss_type25_transaction GTest::gtest_main)
add_test(NAME native_group_host COMMAND native_group_host)
add_executable(native_group_cuda CudaTest.cu)
add_executable(native_group_single_legacy_cuda "${runtime}/MovingSceneTest.cu")
foreach(target native_group_cuda native_group_single_legacy_cuda)
  target_include_directories(${target} PRIVATE "${observed}")
  target_compile_definitions(${target} PRIVATE TYPE25_NATIVE_REFERENCE_FILE="${observed}/native-reference.bin")
  target_link_libraries(${target} PRIVATE tl_radioss_type25_transaction
    robo_dyna_native_scene_material GTest::gtest_main CUDA::cudart)
  set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(${target} PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME ${target} COMMAND ${target})
endforeach()
set_tests_properties(native_group_host native_group_cuda native_group_single_legacy_cuda
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 300)
add_test(NAME native_group_observed_sources COMMAND "${Python3_EXECUTABLE}" -B
  "${runtime}/prepare_fixture.py" --output "${observed}" --check)
