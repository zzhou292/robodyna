cmake_minimum_required(VERSION 3.22)
project(ShellFilteredZeroC LANGUAGES CXX)
option(TL_FILTERED_ZERO_C_NATIVE "Build complete native zero-C filter comparisons" OFF)
option(TL_FILTERED_ZERO_C_CUDA "Build device-owned zero-C history comparisons" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
add_executable(shell_filtered_zero_c_values_check ZeroCValueTest.cpp)
target_include_directories(shell_filtered_zero_c_values_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_filtered_zero_c_values_check PRIVATE cxx_std_17)
target_compile_options(shell_filtered_zero_c_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_filtered_zero_c_values_check PRIVATE GTest::gtest_main)
add_test(NAME shell_filtered_zero_c_values COMMAND shell_filtered_zero_c_values_check)
include("${CMAKE_CURRENT_SOURCE_DIR}/../plasticity_binding/PlasticityBinding.cmake")
if(TL_FILTERED_ZERO_C_NATIVE)
  enable_language(Fortran)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/law44" native-law44)
  add_library(shell_filtered_zero_c_native STATIC NativeZeroC.F90)
  foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(value law44_point_native ${property})
    set_property(TARGET shell_filtered_zero_c_native PROPERTY ${property} "${value}")
  endforeach()
  get_target_property(modules law44_point_native Fortran_MODULE_DIRECTORY)
  target_include_directories(shell_filtered_zero_c_native PRIVATE "${modules}")
  target_link_libraries(shell_filtered_zero_c_native PUBLIC law44_point_native)
  add_executable(shell_filtered_zero_c_native_check ZeroCNativeTest.cpp)
  target_link_libraries(shell_filtered_zero_c_native_check PRIVATE shell_filtered_zero_c_native GTest::gtest_main)
  target_compile_options(shell_filtered_zero_c_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME shell_filtered_zero_c_native COMMAND shell_filtered_zero_c_native_check)
endif()
if(TL_FILTERED_ZERO_C_CUDA)
  if(NOT TL_FILTERED_ZERO_C_NATIVE)
    message(FATAL_ERROR "CUDA zero-C qualification requires the complete native oracle")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_filtered_zero_c_cuda_check ZeroCCudaTest.cu)
  target_link_libraries(shell_filtered_zero_c_cuda_check PRIVATE shell_filtered_zero_c_native CUDA::cudart GTest::gtest_main)
  target_compile_features(shell_filtered_zero_c_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_filtered_zero_c_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_filtered_zero_c_cuda COMMAND shell_filtered_zero_c_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
