cmake_minimum_required(VERSION 3.22)
project(TLType45Joint LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
add_executable(type45_joint_host HostTest.cpp)
target_compile_features(type45_joint_host PRIVATE cxx_std_17)
target_include_directories(type45_joint_host PRIVATE "${TL_ROOT}")
target_compile_options(type45_joint_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(type45_joint_host PRIVATE GTest::gtest_main)
add_test(NAME type45_joint_host COMMAND type45_joint_host)
set_tests_properties(type45_joint_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(TL_TYPE45_NATIVE "Build complete native joint leaf and caller comparison" OFF)
option(TL_TYPE45_CUDA "Build device-owned joint histories against independent native packets" OFF)
if(TL_TYPE45_CUDA AND NOT TL_TYPE45_NATIVE)
  message(FATAL_ERROR "TYPE45 CUDA gate requires independent native packets")
endif()
if(TL_TYPE45_NATIVE)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_subdirectory(native)
  add_library(type45_oracle STATIC NativeOracle.cpp NativeCompare.cpp NativeLengthRoundoff.cpp)
  target_compile_features(type45_oracle PUBLIC cxx_std_17)
  target_include_directories(type45_oracle PUBLIC "${TL_ROOT}")
  target_compile_options(type45_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(type45_oracle PUBLIC GTest::gtest type45_joint_native)
  add_executable(type45_joint_native_tests NativeTest.cpp NativeLengthWitnessTest.cpp)
  target_compile_options(type45_joint_native_tests PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(type45_joint_native_tests PRIVATE type45_oracle GTest::gtest_main)
  add_test(NAME type45_joint_native_tests COMMAND type45_joint_native_tests)
  set_tests_properties(type45_joint_native_tests PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
if(TL_TYPE45_CUDA)
  enable_language(CUDA)
  add_executable(type45_joint_cuda CudaTest.cu)
  target_compile_features(type45_joint_cuda PRIVATE cuda_std_17)
  target_compile_options(type45_joint_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-fno-fast-math,-ffp-contract=off)
  target_link_libraries(type45_joint_cuda PRIVATE type45_oracle GTest::gtest_main)
  add_test(NAME type45_joint_cuda COMMAND type45_joint_cuda)
  set_tests_properties(type45_joint_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
option(TL_TYPE45_ORIGINAL "Compare all original 44 joint geometries with supplied context" OFF)
set(TL_YARIS_ASSETS "" CACHE PATH "Immutable original canonical Yaris assets")
if(TL_TYPE45_ORIGINAL)
  if(NOT TL_TYPE45_NATIVE OR NOT TL_YARIS_ASSETS)
    message(FATAL_ERROR "Original TYPE45 geometry requires native oracle and TL_YARIS_ASSETS")
  endif()
  set(source_prepared "${CMAKE_CURRENT_BINARY_DIR}/source_prepared")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/source_fixture/prepare.py" --assets "${TL_YARIS_ASSETS}"
    --output "${source_prepared}" RESULT_VARIABLE source_status)
  if(NOT source_status EQUAL 0)
    message(FATAL_ERROR "Original TYPE45 joint geometry identity failed")
  endif()
  add_executable(type45_joint_source SourceGeometryTest.cpp)
  target_include_directories(type45_joint_source PRIVATE "${source_prepared}")
  target_compile_options(type45_joint_source PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(type45_joint_source PRIVATE type45_oracle GTest::gtest_main)
  add_test(NAME type45_joint_source COMMAND type45_joint_source)
  add_test(NAME type45_joint_source_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/source_fixture/prepare.py" --assets "${TL_YARIS_ASSETS}"
    --output "${source_prepared}" --check)
  set_tests_properties(type45_joint_source type45_joint_source_identity
    PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
