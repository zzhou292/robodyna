cmake_minimum_required(VERSION 3.22)
project(TLTiedPostKinChk LANGUAGES CXX)
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/../../../lib_src/constraints/tied_shell/TiedPostKinChk.cmake")
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(tied_post_kinchk_host_test ValueTest.cpp)
target_link_libraries(tied_post_kinchk_host_test PRIVATE tl_tied_post_kinchk GTest::gtest_main)
add_test(NAME tied_post_kinchk_values COMMAND tied_post_kinchk_host_test)
add_test(NAME tied_post_kinchk_source_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/verify_sources.py")
option(TL_TIED_POST_KINCHK_NATIVE "Build complete native KINCHK oracle" OFF)
if(TL_TIED_POST_KINCHK_NATIVE)
  enable_language(Fortran)
  include("${CMAKE_CURRENT_LIST_DIR}/Native.cmake")
  add_executable(tied_post_kinchk_native_test NativeTest.cpp)
  target_link_libraries(tied_post_kinchk_native_test PRIVATE tl_tied_post_kinchk tied_post_kinchk_oracle GTest::gtest_main)
  add_test(NAME tied_post_kinchk_native COMMAND tied_post_kinchk_native_test)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
