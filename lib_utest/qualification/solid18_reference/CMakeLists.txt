cmake_minimum_required(VERSION 3.22)
project(TLSolid18Reference LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(solid18_reference_host HostTest.cpp)
target_compile_features(solid18_reference_host PRIVATE cxx_std_17)
target_include_directories(solid18_reference_host PRIVATE "${TL_ROOT}")
target_compile_options(solid18_reference_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(solid18_reference_host PRIVATE GTest::gtest_main)
add_test(NAME solid18_reference_host COMMAND solid18_reference_host)
set_tests_properties(solid18_reference_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(TL_SOLID18_REFERENCE_ENABLE_NATIVE "Build independent native starter oracle" OFF)
option(TL_SOLID18_REFERENCE_ENABLE_CUDA "Build actual device reference gate" OFF)
if(TL_SOLID18_REFERENCE_ENABLE_CUDA AND NOT TL_SOLID18_REFERENCE_ENABLE_NATIVE)
  message(FATAL_ERROR "CUDA reference gate requires independent native oracle")
endif()
if(TL_SOLID18_REFERENCE_ENABLE_NATIVE)
  enable_language(Fortran)
  add_subdirectory(native)
  add_library(solid18_reference_oracle STATIC NativeOracle.cpp)
  target_compile_features(solid18_reference_oracle PUBLIC cxx_std_17)
  target_include_directories(solid18_reference_oracle PUBLIC "${TL_ROOT}")
  target_compile_options(solid18_reference_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(solid18_reference_oracle PUBLIC solid18_reference_native)
  foreach(group IN ITEMS Native SourceNative)
    string(TOLOWER "${group}" group_lower)
    set(target "solid18_reference_${group_lower}_test")
    add_executable(${target} "${group}Test.cpp")
    target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off)
    target_link_libraries(${target} PRIVATE solid18_reference_oracle GTest::gtest_main)
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
  endforeach()
endif()
if(TL_SOLID18_REFERENCE_ENABLE_CUDA)
  enable_language(CUDA)
  add_executable(solid18_reference_cuda CudaTest.cu)
  target_compile_features(solid18_reference_cuda PRIVATE cuda_std_17)
  target_compile_options(solid18_reference_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  target_link_libraries(solid18_reference_cuda PRIVATE solid18_reference_oracle GTest::gtest_main)
  add_test(NAME solid18_reference_cuda COMMAND solid18_reference_cuda)
  set_tests_properties(solid18_reference_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
add_test(NAME solid18_reference_source_identity
  COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
