cmake_minimum_required(VERSION 3.22)
project(tl_t3_one_point_qualification LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/t3/T3OnePoint.cmake")
find_package(GTest REQUIRED)
add_executable(t3_one_point_host_test MathTest.cpp)
target_link_libraries(t3_one_point_host_test PRIVATE tl_t3_one_point GTest::gtest_main)
target_compile_options(t3_one_point_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME t3_one_point_host COMMAND t3_one_point_host_test)
option(T3_ONE_POINT_NATIVE_CHECKS "Independent complete native one-point caller" ON)
if(T3_ONE_POINT_NATIVE_CHECKS)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(QBAT_FORCE_NATIVE_CHECKS ON CACHE BOOL "" FORCE)
  set(QBAT_FORCE_CUDA_CHECKS OFF CACHE BOOL "" FORCE)
  add_subdirectory("../qbat_force" qbat_force)
  set(T3_R2_BUILD ON CACHE BOOL "" FORCE)
  set(T3_R3_BUILD ON CACHE BOOL "" FORCE)
  add_subdirectory("../native/t3" t3_native)
  include("${CMAKE_CURRENT_LIST_DIR}/OnePointNative.cmake")
  add_executable(t3_one_point_native_test NativeTest.cpp NativeOracle.cpp)
  target_link_libraries(t3_one_point_native_test PRIVATE tl_t3_one_point t3_one_point_native GTest::gtest_main)
  target_compile_options(t3_one_point_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME t3_one_point_native COMMAND t3_one_point_native_test)
  add_test(NAME t3_one_point_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${one_native}/verify_sources.py")
endif()
option(T3_ONE_POINT_CUDA_CHECKS "Root-scheduled actual one-point CUDA checks" OFF)
if(T3_ONE_POINT_CUDA_CHECKS)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(t3_one_point_cuda_test MathCudaTest.cu)
  target_link_libraries(t3_one_point_cuda_test PRIVATE tl_t3_one_point GTest::gtest_main CUDA::cudart)
  set_target_properties(t3_one_point_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(t3_one_point_cuda_test PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME t3_one_point_cuda COMMAND t3_one_point_cuda_test)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
