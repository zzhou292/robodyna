cmake_minimum_required(VERSION 3.22)
project(TLNodalAcceptedRawMass LANGUAGES CXX CUDA Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(TL_TIED_CIN_RUNTIME_NATIVE ON CACHE BOOL "Native mass phase regression" FORCE)
set(TL_TIED_CIN_RUNTIME_CUDA ON CACHE BOOL "Actual existing CIN owner regression" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../tied_cin_runtime" cin-regression)
add_executable(nodal_accepted_raw_mass_host HostTest.cpp)
target_link_libraries(nodal_accepted_raw_mass_host PRIVATE tl_explicit_nodal_state GTest::gtest_main)
add_test(NAME nodal_accepted_raw_mass_host COMMAND nodal_accepted_raw_mass_host)
add_executable(nodal_accepted_raw_mass_consumer Consumer.cpp)
target_link_libraries(nodal_accepted_raw_mass_consumer PRIVATE tl_explicit_nodal_state)
add_test(NAME nodal_accepted_raw_mass_consumer COMMAND nodal_accepted_raw_mass_consumer)
add_executable(nodal_accepted_raw_mass_cuda OwnerTest.cu RigidTest.cu ../rigid_assembly_owner/Fixture.cpp)
target_link_libraries(nodal_accepted_raw_mass_cuda PRIVATE tl_explicit_nodal_state tied_cin_runtime_native GTest::gtest_main)
set_target_properties(nodal_accepted_raw_mass_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
target_compile_options(nodal_accepted_raw_mass_cuda PRIVATE
  "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
  "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>")
add_test(NAME nodal_accepted_raw_mass_cuda COMMAND nodal_accepted_raw_mass_cuda)
set_tests_properties(nodal_accepted_raw_mass_host nodal_accepted_raw_mass_consumer nodal_accepted_raw_mass_cuda
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
