cmake_minimum_required(VERSION 3.22)
project(TLConnectorMappedAssembly LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(connector_assembly_layout STATIC
  "${tl_root}/lib_src/elements/type13/resident/Arena.cpp"
  "${tl_root}/lib_src/elements/type25/Type25BatchArena.cpp")
target_include_directories(connector_assembly_layout PUBLIC "${tl_root}")
target_link_libraries(connector_assembly_layout PUBLIC CUDA::toolkit)
target_compile_features(connector_assembly_layout PUBLIC cxx_std_17)
target_compile_options(connector_assembly_layout PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(connector_mapped_assembly_host HostTest.cpp)
target_link_libraries(connector_mapped_assembly_host PRIVATE connector_assembly_layout GTest::gtest_main)
target_compile_options(connector_mapped_assembly_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME connector_mapped_assembly_host COMMAND connector_mapped_assembly_host)
set_tests_properties(connector_mapped_assembly_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME connector_mapped_assembly_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
option(CONNECTOR_ASSEMBLY_CUDA "GPU gather compared to frozen original serial assembly" OFF)
if(CONNECTOR_ASSEMBLY_CUDA)
  enable_language(CUDA)
  add_executable(connector_mapped_assembly_cuda CudaTest.cu
    "${tl_root}/lib_src/elements/type13/resident/Assembly.cu"
    "${tl_root}/lib_src/elements/type25/mapped/Kernels.cu")
  target_link_libraries(connector_mapped_assembly_cuda PRIVATE connector_assembly_layout GTest::gtest_main CUDA::cudart)
  set_target_properties(connector_mapped_assembly_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(connector_mapped_assembly_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME connector_mapped_assembly_cuda COMMAND connector_mapped_assembly_cuda)
  set_tests_properties(connector_mapped_assembly_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
endif()
