cmake_minimum_required(VERSION 3.22)
project(TLCinPhysicalMainSummary LANGUAGES CXX)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
add_executable(cin_physical_main_host ValuesTest.cpp)
target_include_directories(cin_physical_main_host PRIVATE "${TL_ROOT}")
target_compile_features(cin_physical_main_host PRIVATE cxx_std_17)
target_compile_options(cin_physical_main_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(cin_physical_main_host PRIVATE GTest::gtest_main CUDA::cudart)
add_test(NAME cin_physical_main_host COMMAND cin_physical_main_host)
option(TL_CIN_PHYSICAL_MAIN_NATIVE "Build independent native plain-body scalar comparison" OFF)
option(TL_CIN_PHYSICAL_MAIN_CUDA "Build actual prepared TT0 query gate" OFF)
if(TL_CIN_PHYSICAL_MAIN_NATIVE)
  enable_language(Fortran)
  include(Native.cmake)
endif()
if(TL_CIN_PHYSICAL_MAIN_CUDA)
  enable_language(CUDA)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" owner)
  add_executable(cin_physical_main_cuda OwnerTest.cu ../rigid_assembly_owner/Fixture.cpp)
  target_link_libraries(cin_physical_main_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main)
  set_target_properties(cin_physical_main_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(cin_physical_main_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>")
  add_test(NAME cin_physical_main_cuda COMMAND cin_physical_main_cuda)
  set_tests_properties(cin_physical_main_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
