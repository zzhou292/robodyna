cmake_minimum_required(VERSION 3.22)
project(TLQephMappedGather LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(qeph_mapped_gather_host HostTest.cpp)
target_include_directories(qeph_mapped_gather_host PRIVATE "${tl_root}")
target_link_libraries(qeph_mapped_gather_host PRIVATE GTest::gtest_main CUDA::toolkit)
target_compile_features(qeph_mapped_gather_host PRIVATE cxx_std_17)
target_compile_options(qeph_mapped_gather_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qeph_mapped_gather_host COMMAND qeph_mapped_gather_host)
set_tests_properties(qeph_mapped_gather_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(QEPH_GATHER_CUDA "Ordered gather against frozen serial mapped assembly" OFF)
if(QEPH_GATHER_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(NOT TARGET tl_explicit_nodal_state)
    add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  endif()
  add_subdirectory("${tl_root}/lib_src/elements/qeph" qeph-values)
  if(NOT TARGET tl_qeph_batch)
    include("${tl_root}/lib_src/elements/qeph/QephBatch.cmake")
  endif()
  add_executable(qeph_mapped_gather_cuda CudaTest.cu)
  target_link_libraries(qeph_mapped_gather_cuda PRIVATE tl_qeph_batch GTest::gtest_main CUDA::cudart)
  set_target_properties(qeph_mapped_gather_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(qeph_mapped_gather_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME qeph_mapped_gather_cuda COMMAND qeph_mapped_gather_cuda)
  set_tests_properties(qeph_mapped_gather_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME qeph_mapped_gather_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(qeph_mapped_gather_source_identity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
