cmake_minimum_required(VERSION 3.22)
project(TLSelfContactSurface LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/SelfContactSurfaceBinding.cmake")
add_executable(self_contact_surface_host HostTest.cpp ExecutionTest.cpp)
target_link_libraries(self_contact_surface_host PRIVATE tl_self_contact_surface_binding GTest::gtest_main)
target_compile_options(self_contact_surface_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME self_contact_surface_host COMMAND self_contact_surface_host)
set_tests_properties(self_contact_surface_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
