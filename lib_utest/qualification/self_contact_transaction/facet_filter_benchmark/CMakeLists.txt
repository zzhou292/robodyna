cmake_minimum_required(VERSION 3.22)
project(TLFacetFilterAdapterBenchmark LANGUAGES CXX CUDA)
get_filename_component(DEFAULT_TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(TL_FILTER_ROOT "${DEFAULT_TL_ROOT}" CACHE PATH "Pinned old-span or compact TL source")
if(NOT EXISTS "${TL_FILTER_ROOT}/lib_src/collision/self_contact_transaction/FacetFilters.h")
  message(FATAL_ERROR "TL_FILTER_ROOT must provide the transaction facet-filter adapter")
endif()
find_package(GTest REQUIRED)
find_package(Git REQUIRED)
include("${TL_FILTER_ROOT}/lib_src/collision/SelfContactTransaction.cmake")
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${TL_FILTER_ROOT}" rev-parse HEAD
  OUTPUT_VARIABLE _filter_commit OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _pin_status)
if(NOT _pin_status EQUAL 0)
  message(FATAL_ERROR "Unable to pin the selected TL source")
endif()
add_executable(facet_filter_adapter_benchmark Main.cpp Cases.cpp Copies.cpp Cases.h Copies.h Evaluate.h Results.h)
target_include_directories(facet_filter_adapter_benchmark PRIVATE "${TL_FILTER_ROOT}")
target_link_libraries(facet_filter_adapter_benchmark PRIVATE tl_self_contact_transaction CUDA::cudart GTest::gtest)
target_compile_features(facet_filter_adapter_benchmark PRIVATE cxx_std_17)
target_compile_options(facet_filter_adapter_benchmark PRIVATE -fno-fast-math -ffp-contract=off)
target_compile_definitions(facet_filter_adapter_benchmark PRIVATE TL_FILTER_SOURCE_COMMIT="${_filter_commit}")
target_link_options(facet_filter_adapter_benchmark PRIVATE -Wl,--wrap=cudaMemcpyAsync)
