cmake_minimum_required(VERSION 3.22)
project(TLType25InitialState LANGUAGES CXX Fortran)
enable_testing()
get_filename_component(TYPE25_INITIAL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
include("${TYPE25_INITIAL_ROOT}/lib_src/collision/RadiossType25InitialStateValues.cmake")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_fixed_main_startup" startup)
include("${CMAKE_CURRENT_LIST_DIR}/native/Native.cmake")
add_executable(type25_initial_state_host HostTest.cpp)
target_compile_features(type25_initial_state_host PRIVATE cxx_std_17)
target_link_libraries(type25_initial_state_host PRIVATE type25_initial_state_native type25_startup_oracle GTest::gtest_main)
target_compile_options(type25_initial_state_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type25_initial_state_host COMMAND type25_initial_state_host)
set_tests_properties(type25_initial_state_host type25_initial_state_source PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)

add_executable(type25_initial_state_consumer Consumer.cpp)
target_compile_features(type25_initial_state_consumer PRIVATE cxx_std_17)
target_link_libraries(type25_initial_state_consumer PRIVATE tl_radioss_type25_initial_state_values)
# Pure production header closure: no GTest/native/CUDA link.
add_test(NAME type25_initial_state_consumer COMMAND type25_initial_state_consumer)
set_tests_properties(type25_initial_state_consumer PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)

option(TYPE25_INITIAL_CUDA "Compile owning actual CUDA value-stage gate" ON)
if(TYPE25_INITIAL_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(type25_initial_state_cuda CudaTest.cu)
  set_target_properties(type25_initial_state_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_link_libraries(type25_initial_state_cuda PRIVATE type25_initial_state_native type25_startup_oracle GTest::gtest_main CUDA::cudart)
  target_compile_options(type25_initial_state_cuda PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-Xcompiler=-ffp-contract=off>" )
  add_test(NAME type25_initial_state_cuda COMMAND type25_initial_state_cuda)
  set_tests_properties(type25_initial_state_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 2 TIMEOUT 120)
endif()
