cmake_minimum_required(VERSION 3.22)
project(TLType25InterfaceSurface LANGUAGES CXX Fortran)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/RadiossType25InterfaceSurface.cmake")
add_subdirectory("../radioss_type25_fixed_main_startup" startup-regression)
add_subdirectory("../radioss_type25_surface_source" surface-regression)
include("${CMAKE_CURRENT_LIST_DIR}/native/Native.cmake")
add_executable(type25_interface_surface_host HostTest.cpp NativeTest.cpp)
target_link_libraries(type25_interface_surface_host PRIVATE type25_interface_oracle GTest::gtest_main)
target_compile_options(type25_interface_surface_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type25_interface_surface_host COMMAND type25_interface_surface_host)
set_tests_properties(type25_interface_surface_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)

add_test(NAME type25_interface_surface_consumer COMMAND type25_interface_surface_consumer)
