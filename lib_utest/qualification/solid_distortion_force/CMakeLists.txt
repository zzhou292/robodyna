cmake_minimum_required(VERSION 3.22)
project(TLSolidDistortionForce LANGUAGES CXX C Fortran)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(TL_DISTORTION_ORACLE_ROOT "" CACHE PATH "Qualified controlled H24 native checkout")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
enable_testing()
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/../solid_distortion_parameters/source_check.py"
  --oracle-root "${TL_DISTORTION_ORACLE_ROOT}" COMMAND_ERROR_IS_FATAL ANY)
set(TL_SOLID24_FORCE_NATIVE ON CACHE BOOL "Independent native material/context" FORCE)
set(TL_SOLID24_FORCE_CUDA OFF CACHE BOOL "No unrelated GPU targets" FORCE)
add_subdirectory("${TL_DISTORTION_ORACLE_ROOT}/lib_utest/qualification/solid24_force" legacy_heph)
add_subdirectory("${TL_DISTORTION_ORACLE_ROOT}/lib_utest/qualification/solid24_icontrol/native" native_icontrol)
add_library(distortion_force_probe STATIC native/ForceProbe.F90)
target_compile_options(distortion_force_probe PRIVATE -ffree-line-length-none -fcheck=bounds -fno-fast-math -ffp-contract=off)
target_include_directories(distortion_force_probe PRIVATE "$<TARGET_PROPERTY:heph_force_native,Fortran_MODULE_DIRECTORY>"
  "$<TARGET_PROPERTY:solid24_ic1_native,Fortran_MODULE_DIRECTORY>")
add_dependencies(distortion_force_probe solid24_ic1_native)
target_link_libraries(distortion_force_probe PUBLIC solid24_ic1_native)
add_executable(solid_distortion_force_host HostTest.cpp)
target_compile_features(solid_distortion_force_host PRIVATE cxx_std_17)
target_compile_options(solid_distortion_force_host PRIVATE -fno-fast-math -ffp-contract=off)
target_include_directories(solid_distortion_force_host PRIVATE "${TL_ROOT}")
target_link_libraries(solid_distortion_force_host PRIVATE distortion_force_probe GTest::gtest_main)
add_test(NAME solid_distortion_force_host COMMAND solid_distortion_force_host)
option(TL_DISTORTION_FORCE_CUDA "Run actual CUDA values against full independent native distortion" OFF)
if(TL_DISTORTION_FORCE_CUDA)
  enable_language(CUDA)
  add_executable(solid_distortion_force_cuda CudaTest.cu)
  target_compile_features(solid_distortion_force_cuda PRIVATE cuda_std_17)
  target_compile_options(solid_distortion_force_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true -Xcompiler=-fno-fast-math,-ffp-contract=off)
  target_include_directories(solid_distortion_force_cuda PRIVATE "${TL_ROOT}")
  target_link_libraries(solid_distortion_force_cuda PRIVATE distortion_force_probe GTest::gtest_main)
  add_test(NAME solid_distortion_force_cuda COMMAND solid_distortion_force_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
