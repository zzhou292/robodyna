cmake_minimum_required(VERSION 3.22)
project(TLNativeContactRowLaunch LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(native_contact_row_launch HostTest.cpp)
target_include_directories(native_contact_row_launch PRIVATE "${tl_root}")
target_link_libraries(native_contact_row_launch PRIVATE GTest::gtest_main)
target_compile_features(native_contact_row_launch PRIVATE cxx_std_17)
add_test(NAME native_contact_row_launch COMMAND native_contact_row_launch)
