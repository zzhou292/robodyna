cmake_minimum_required(VERSION 3.22)
project(TLBeam18MeasurementReadTile LANGUAGES CXX C)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(oracle "${CMAKE_CURRENT_BINARY_DIR}/oracle")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" --output "${oracle}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(beam18_read_tile_host HostTest.cpp)
target_include_directories(beam18_read_tile_host PRIVATE "${TL_ROOT}" "${oracle}" ${CUDAToolkit_INCLUDE_DIRS})
target_compile_features(beam18_read_tile_host PRIVATE cxx_std_17)
target_compile_options(beam18_read_tile_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(beam18_read_tile_host PRIVATE GTest::gtest_main)
add_test(NAME beam18_read_tile_host COMMAND beam18_read_tile_host)
add_test(NAME beam18_read_tile_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py")
option(BEAM18_READ_TILE_CUDA "Literal frozen CUDA finalizer comparison" OFF)
option(BEAM18_READ_TILE_OWNER "Actual native owner and rollback gate" OFF)
if(BEAM18_READ_TILE_CUDA)
  enable_language(CUDA)
  add_executable(beam18_read_tile_cuda CudaTest.cu)
  target_include_directories(beam18_read_tile_cuda PRIVATE "${TL_ROOT}" "${oracle}")
  target_link_libraries(beam18_read_tile_cuda PRIVATE GTest::gtest_main CUDA::cudart)
  set_target_properties(beam18_read_tile_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(beam18_read_tile_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true --ftz=false --ptxas-options=-v -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME beam18_read_tile_cuda COMMAND beam18_read_tile_cuda)
endif()
if(BEAM18_READ_TILE_OWNER)
  if(NOT BEAM18_READ_TILE_CUDA)
    message(FATAL_ERROR "Owner requires CUDA")
  endif()
  set(BEAM18_RESIDENT_CUDA ON CACHE BOOL "" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../beam18_resident" beam18-owner)
endif()
get_property(checks DIRECTORY PROPERTY TESTS)
set_tests_properties(${checks} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
