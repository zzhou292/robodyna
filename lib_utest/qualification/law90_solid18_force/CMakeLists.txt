cmake_minimum_required(VERSION 3.22)
project(TLLaw90Solid18Force LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(force_target target)
  target_include_directories(${target} PRIVATE "${TL_ROOT}")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
  target_link_libraries(${target} PRIVATE GTest::gtest_main)
endfunction()
add_executable(law90_solid18_force_host HostTest.cpp)
force_target(law90_solid18_force_host)
add_test(NAME law90_solid18_force_host COMMAND law90_solid18_force_host)
add_test(NAME law90_solid18_force_owning_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
option(TL_LAW90_SOLID18_FORCE_NATIVE "Independent complete native caller and force oracle" OFF)
option(TL_LAW90_SOLID18_FORCE_CUDA "Actual CUDA native recurrence and late failure" OFF)
set(TL_LAW90_RADIATOR_FIXTURE "" CACHE PATH "Authenticated original yaris-radiator-geometry-1 directory")
if(TL_LAW90_SOLID18_FORCE_CUDA AND NOT TL_LAW90_SOLID18_FORCE_NATIVE)
  message(FATAL_ERROR "Actual CUDA force values require the independent native oracle")
endif()
if(TL_LAW90_SOLID18_FORCE_NATIVE)
  if(NOT TL_LAW90_RADIATOR_FIXTURE)
    message(FATAL_ERROR "Native force gate requires authenticated original geometry")
  endif()
  set(fixture "${CMAKE_CURRENT_BINARY_DIR}/source_fixture")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/../law90_solid18_reference/source_fixture/prepare_fixture.py"
    --input "${TL_LAW90_RADIATOR_FIXTURE}" --output "${fixture}/RadiatorSourceFixture.h" RESULT_VARIABLE result)
  if(NOT result EQUAL 0)
    message(FATAL_ERROR "LAW90 original force fixture authentication failed")
  endif()
  add_test(NAME law90_solid18_force_original_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/../law90_solid18_reference/source_fixture/prepare_fixture.py"
    --input "${TL_LAW90_RADIATOR_FIXTURE}" --output "${fixture}/RadiatorSourceFixture.h" --check)
  enable_language(C)
  enable_language(Fortran)
  add_subdirectory(../law90_preparation/native preparation_native)
  add_subdirectory(../law90_point/native point_native)
  add_subdirectory(../solid18_reference/native reference_native)
  add_subdirectory(../law90_solid18_reference/native kinematics_native)
  add_subdirectory(native)
  foreach(group IN ITEMS Native SourceNative)
    string(TOLOWER "${group}" lower)
    set(target law90_solid18_force_${lower})
    add_executable(${target} "${group}Test.cpp")
    force_target(${target})
    target_include_directories(${target} PRIVATE "${fixture}")
    target_link_libraries(${target} PRIVATE law90_solid18_force_native)
    add_test(NAME ${target} COMMAND ${target})
  endforeach()
endif()
if(TL_LAW90_SOLID18_FORCE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(law90_solid18_force_cuda CudaTest.cu)
  force_target(law90_solid18_force_cuda)
  target_include_directories(law90_solid18_force_cuda PRIVATE "${fixture}")
  target_compile_features(law90_solid18_force_cuda PRIVATE cuda_std_17)
  target_compile_options(law90_solid18_force_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xptxas=-v;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(law90_solid18_force_cuda PRIVATE law90_solid18_force_native CUDA::cudart)
  add_test(NAME law90_solid18_force_cuda COMMAND law90_solid18_force_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 240)
