cmake_minimum_required(VERSION 3.18)
project(TLVehiclePlasticityCatalogQualification LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/ShellBatchPlasticityBinding.cmake")
add_executable(vehicle_plasticity_catalog_check VehicleCatalogAdmissionTest.cpp VehicleCatalogSourceSizeTest.cpp
  "${CMAKE_CURRENT_LIST_DIR}/../host_shell_collection/AllocationFailure.cpp")
target_include_directories(vehicle_plasticity_catalog_check PRIVATE "${tl_root}")
target_link_libraries(vehicle_plasticity_catalog_check PRIVATE tl_shell_batch_plasticity_binding GTest::gtest_main)
target_compile_options(vehicle_plasticity_catalog_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME vehicle_plasticity_catalog_small COMMAND vehicle_plasticity_catalog_check --gtest_filter=VehiclePlasticityCatalog.*)
add_test(NAME vehicle_plasticity_catalog_source_size COMMAND vehicle_plasticity_catalog_check --gtest_filter=VehiclePlasticityCatalogSourceSize.*)
set_tests_properties(vehicle_plasticity_catalog_small PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
set_tests_properties(vehicle_plasticity_catalog_source_size PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
include("${CMAKE_CURRENT_LIST_DIR}/../vehicle_shell_host/VehicleShellHost.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/../plasticity_binding/PlasticityBinding.cmake")
