cmake_minimum_required(VERSION 3.22)
project(TLLaw44AnalyticQualification LANGUAGES CXX)
option(TL_LAW44_ANALYTIC_NATIVE "Build independent MFUNC0 native oracle" ON)
option(TL_LAW44_ANALYTIC_CUDA "Build actual CUDA point/native check" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
add_executable(law44_analytic_preparation_check AnalyticPreparationTest.cpp)
target_include_directories(law44_analytic_preparation_check PRIVATE "${TL_ROOT}")
target_compile_features(law44_analytic_preparation_check PRIVATE cxx_std_17)
target_compile_options(law44_analytic_preparation_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(law44_analytic_preparation_check PRIVATE GTest::gtest_main)
add_test(NAME law44_analytic_preparation_check COMMAND law44_analytic_preparation_check)
include("${CMAKE_CURRENT_SOURCE_DIR}/../plasticity_binding/PlasticityBinding.cmake")
if(TL_LAW44_ANALYTIC_NATIVE)
  enable_language(Fortran)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/law44" native-law44)
endif()
if(TL_LAW44_ANALYTIC_CUDA)
  if(NOT TL_LAW44_ANALYTIC_NATIVE)
    message(FATAL_ERROR "The CUDA oracle requires the native reference")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(law44_analytic_cuda_check AnalyticCudaTest.cu)
  target_include_directories(law44_analytic_cuda_check PRIVATE "${TL_ROOT}")
  target_compile_features(law44_analytic_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(law44_analytic_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(law44_analytic_cuda_check PRIVATE law44_point_native CUDA::cudart GTest::gtest_main)
  add_test(NAME law44_analytic_cuda_check COMMAND law44_analytic_cuda_check)
  set_tests_properties(law44_analytic_cuda_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
endif()
