cmake_minimum_required(VERSION 3.22)
project(TLFailureSourceExtent LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/elements/ShellBatchPlasticity.cmake")
set(oracle "${CMAKE_CURRENT_BINARY_DIR}/oracle")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" "${CMAKE_CURRENT_LIST_DIR}/source-manifest.json")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" --output "${oracle}"
  COMMAND_ERROR_IS_FATAL ANY)
foreach(kind IN ITEMS host cuda)
  if(kind STREQUAL "host")
    set(source HostTest.cpp)
  else()
    set(source StorageCudaTest.cpp)
  endif()
  add_executable(failure_source_extent_${kind} "${source}")
  target_include_directories(failure_source_extent_${kind} PRIVATE "${TL_ROOT}" "${oracle}")
  target_link_libraries(failure_source_extent_${kind} PRIVATE tl_shell_batch_plasticity GTest::gtest_main)
  target_compile_features(failure_source_extent_${kind} PRIVATE cxx_std_17)
  target_compile_options(failure_source_extent_${kind} PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME failure_source_extent_${kind} COMMAND failure_source_extent_${kind})
endforeach()
add_test(NAME failure_source_extent_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py")
set_tests_properties(failure_source_extent_host failure_source_extent_cuda failure_source_extent_source PROPERTIES
  RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
