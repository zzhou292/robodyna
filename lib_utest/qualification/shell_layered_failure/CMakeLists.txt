cmake_minimum_required(VERSION 3.22)
project(ShellLayeredFailureQualification LANGUAGES CXX)
option(TL_LAYERED_FAILURE_NATIVE "Build exact native LAW44/failure caller qualification" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
add_executable(shell_layered_failure_values_check FailureSectionValueTest.cpp)
target_include_directories(shell_layered_failure_values_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_layered_failure_values_check PRIVATE cxx_std_17)
target_compile_options(shell_layered_failure_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_layered_failure_values_check PRIVATE GTest::gtest_main)
add_test(NAME shell_layered_failure_values COMMAND shell_layered_failure_values_check)

if(TL_LAYERED_FAILURE_NATIVE)
  enable_language(Fortran)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shell_constant_failure" constant-failure)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(prepared "${CMAKE_CURRENT_BINARY_DIR}/prepared")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/native/prepare_sources.py"
    --output "${prepared}" RESULT_VARIABLE status ERROR_VARIABLE diagnostic)
  if(NOT status EQUAL 0)
    message(FATAL_ERROR "Native caller source preparation failed: ${diagnostic}")
  endif()
  add_custom_target(shell_layered_failure_sources
    COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/native/prepare_sources.py"
      --output "${prepared}" --check VERBATIM)
  add_library(shell_layered_failure_native STATIC
    native/NativeParent.F native/NativeInterfaces.F90 native/NativeCaller.F90)
  add_dependencies(shell_layered_failure_native shell_layered_failure_sources)
  foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
    get_target_property(value law44_point_native ${property})
    set_property(TARGET shell_layered_failure_native PROPERTY ${property} "${value}")
  endforeach()
  get_target_property(modules law44_point_native Fortran_MODULE_DIRECTORY)
  target_include_directories(shell_layered_failure_native BEFORE PRIVATE "${prepared}" "${modules}")
  set_target_properties(shell_layered_failure_native PROPERTIES
    Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/caller-modules")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/caller-modules")
  target_link_libraries(shell_layered_failure_native PUBLIC shell_constant_failure_native)
  add_executable(shell_layered_failure_native_check
    FailureNativeTest.cpp FailureNativeFixture.cpp FailureNativeChecks.cpp)
  target_link_libraries(shell_layered_failure_native_check PRIVATE shell_layered_failure_native GTest::gtest_main)
  target_compile_options(shell_layered_failure_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME shell_layered_failure_native COMMAND shell_layered_failure_native_check)
endif()
set_tests_properties(shell_layered_failure_values PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
