cmake_minimum_required(VERSION 3.18)
project(TLRigidAssemblyBinding LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/constraints/NodalRigidAssemblyBinding.cmake")
add_executable(rigid_assembly_binding_check ValueTest.cpp CapacityTest.cpp EmptyTest.cpp)
target_link_libraries(rigid_assembly_binding_check PRIVATE tl_nodal_rigid_assembly_binding GTest::gtest_main)
target_compile_options(rigid_assembly_binding_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME rigid_assembly_binding COMMAND rigid_assembly_binding_check)
set_tests_properties(rigid_assembly_binding PROPERTIES PROCESSORS 1 RUN_SERIAL TRUE TIMEOUT 120)
