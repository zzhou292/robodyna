cmake_minimum_required(VERSION 3.22)
project(TLType45Resident LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
include("${TL_ROOT}/lib_src/elements/type45/resident/Values.cmake")
include("${TL_ROOT}/lib_src/constraints/tied_shell/TiedCinAttachment.cmake")
add_executable(type45_resident_host StateTest.cpp LayoutTest.cpp ../rigid_assembly_owner/Fixture.cpp)
target_link_libraries(type45_resident_host PRIVATE tl_type45_batch_values tl_tied_cin_attachment GTest::gtest_main CUDA::cudart)
target_compile_options(type45_resident_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type45_resident_host COMMAND type45_resident_host)
set_tests_properties(type45_resident_host PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE PROCESSORS 1)
add_executable(type45_virgin_host VirginTest.cpp)
target_include_directories(type45_virgin_host PRIVATE "${TL_ROOT}")
target_compile_features(type45_virgin_host PRIVATE cxx_std_17)
target_compile_options(type45_virgin_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(type45_virgin_host PRIVATE GTest::gtest_main)
add_test(NAME type45_virgin_host COMMAND type45_virgin_host)
set_tests_properties(type45_virgin_host PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
option(TL_TYPE45_RESIDENT_NATIVE "Build independent TT0 caller and existing TYPE45 native regressions" OFF)
option(TL_TYPE45_RESIDENT_CUDA "Build actual seven-participant owner/native TYPE45 gate" OFF)
if(TL_TYPE45_RESIDENT_CUDA)
  enable_language(CUDA)
  set(TL_TYPE45_RESIDENT_NATIVE ON)
endif()
if(TL_TYPE45_RESIDENT_NATIVE)
  enable_language(Fortran)
  set(TL_TYPE45_NATIVE ON)
  add_subdirectory("../type45_joint" "joint-values")
  add_library(type45_constructor_native STATIC native/Constructor.F90)
  target_include_directories(type45_constructor_native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/joint-values/native/modules")
  target_link_libraries(type45_constructor_native PUBLIC type45_joint_native)
  target_compile_options(type45_constructor_native PRIVATE -ffree-line-length-none -fcheck=bounds -fbacktrace
      -fno-fast-math -ffp-contract=off)
  add_executable(type45_constructor_native_tests ConstructorNativeTest.cpp)
  target_link_libraries(type45_constructor_native_tests PRIVATE type45_constructor_native type45_oracle GTest::gtest_main)
  target_compile_options(type45_constructor_native_tests PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME type45_constructor_native_tests COMMAND type45_constructor_native_tests)
  set_tests_properties(type45_constructor_native_tests PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE PROCESSORS 1)
endif()
if(TL_TYPE45_RESIDENT_CUDA)
  set(PHYSICAL_PUBLICATION_CUDA ON)
  add_subdirectory("../physical_publication" "physical-publication")
  include("${CMAKE_CURRENT_LIST_DIR}/Cuda.cmake")
endif()
