cmake_minimum_required(VERSION 3.22)
project(TLQephObserverReduction LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(qeph_observer_host HostTest.cpp)
target_include_directories(qeph_observer_host PRIVATE "${tl_root}")
target_link_libraries(qeph_observer_host PRIVATE GTest::gtest_main CUDA::toolkit)
target_compile_features(qeph_observer_host PRIVATE cxx_std_17)
target_compile_options(qeph_observer_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qeph_observer_host COMMAND qeph_observer_host)
set_tests_properties(qeph_observer_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(QEPH_OBSERVER_CUDA "Mapped observer tree versus independent serial diagnostics" OFF)
if(QEPH_OBSERVER_CUDA)
  enable_language(CUDA)
  add_subdirectory("${tl_root}/lib_src/solvers" nodal-owner)
  add_subdirectory("${tl_root}/lib_src/elements/qeph" qeph-values)
  include("${tl_root}/lib_src/elements/qeph/QephBatch.cmake")
  add_executable(qeph_observer_cuda CudaTest.cu Truth.cpp)
  target_include_directories(qeph_observer_cuda PRIVATE "${tl_root}")
  target_link_libraries(qeph_observer_cuda PRIVATE tl_qeph_batch GTest::gtest_main CUDA::cudart)
  set_target_properties(qeph_observer_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_features(qeph_observer_cuda PRIVATE cxx_std_17)
  target_compile_options(qeph_observer_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME qeph_observer_cuda COMMAND qeph_observer_cuda)
  set_tests_properties(qeph_observer_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME qeph_observer_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(qeph_observer_identity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
