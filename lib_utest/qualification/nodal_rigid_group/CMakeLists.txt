cmake_minimum_required(VERSION 3.22)
project(nodal_rigid_group_qualification LANGUAGES CXX Fortran)
option(TL_NODAL_RIGID_CUDA_CHECKS "Build bounded nodal-rigid pure-math CUDA checks" OFF)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(tl_rigid_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${tl_rigid_root}/lib_src/constraints/NodalRigidGroup.cmake")
enable_testing()
add_executable(nodal_rigid_group_check GroupModelTest.cpp GroupMathTest.cpp)
target_link_libraries(nodal_rigid_group_check PRIVATE tl_nodal_rigid_group GTest::gtest_main)
target_compile_options(nodal_rigid_group_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME nodal_rigid_group_check COMMAND nodal_rigid_group_check)
set_tests_properties(nodal_rigid_group_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)

if(NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Native nodal-rigid arithmetic currently admits GNU Fortran only")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native/verify_sources.py"
  RESULT_VARIABLE rigid_verify ERROR_VARIABLE rigid_verify_error)
if(NOT rigid_verify EQUAL 0)
  message(FATAL_ERROR "Pinned nodal-rigid source verification failed: ${rigid_verify_error}")
endif()
add_custom_target(nodal_rigid_verify_sources
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/native/verify_sources.py" VERBATIM)
add_library(nodal_rigid_native STATIC native/NativeRigidGroup.F
  native/NativeFrameStep.F native/NativePrimaryStep.F native/NativeMemberStep.F)
add_dependencies(nodal_rigid_native nodal_rigid_verify_sources)
target_include_directories(nodal_rigid_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_compile_options(nodal_rigid_native PRIVATE -cpp -ffixed-line-length-none -fcheck=bounds
  -fbacktrace -fno-fast-math -ffp-contract=off)
add_executable(nodal_rigid_group_native_check GroupNativeTest.cpp)
target_link_libraries(nodal_rigid_group_native_check PRIVATE nodal_rigid_native tl_nodal_rigid_group GTest::gtest_main)
target_compile_options(nodal_rigid_group_native_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME nodal_rigid_group_native_check COMMAND nodal_rigid_group_native_check)
set_tests_properties(nodal_rigid_group_native_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)


add_executable(nodal_rigid_step_check GroupStepMathTest.cpp)
target_link_libraries(nodal_rigid_step_check PRIVATE tl_nodal_rigid_group GTest::gtest_main)
target_compile_options(nodal_rigid_step_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME nodal_rigid_step_check COMMAND nodal_rigid_step_check)
set_tests_properties(nodal_rigid_step_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)

add_executable(nodal_rigid_step_native_check GroupStepNativeTest.cpp GroupStepNativeFixture.cpp)
target_link_libraries(nodal_rigid_step_native_check PRIVATE nodal_rigid_native tl_nodal_rigid_group GTest::gtest_main)
target_compile_options(nodal_rigid_step_native_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME nodal_rigid_step_native_check COMMAND nodal_rigid_step_native_check)
set_tests_properties(nodal_rigid_step_native_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
