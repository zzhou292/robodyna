cmake_minimum_required(VERSION 3.22)
project(TLLayeredLaw1ForceQualification LANGUAGES CXX)
option(TL_LAYERED_LAW1_FORCE_NATIVE "Build independent elastic native force-stage oracle" ON)
option(TL_LAYERED_LAW1_FORCE_CUDA "Compare actual CUDA elastic forces against native recurrence" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
add_executable(shell_layered_law1_force_value_check Law1ForceValueTest.cpp)
target_include_directories(shell_layered_law1_force_value_check PRIVATE "${TL_ROOT}")
target_link_libraries(shell_layered_law1_force_value_check PRIVATE GTest::gtest_main)
target_compile_features(shell_layered_law1_force_value_check PRIVATE cxx_std_17)
target_compile_options(shell_layered_law1_force_value_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME shell_layered_law1_force_values COMMAND shell_layered_law1_force_value_check)
if(TL_LAYERED_LAW1_FORCE_NATIVE)
  enable_language(Fortran)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/law1" native-law1)
  set(TL_SHELL_LAYERED_J2_ENABLE_CUDA ${TL_LAYERED_LAW1_FORCE_CUDA})
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_j2" native-existing)
  include("NativeForce.cmake")
  add_executable(shell_layered_law1_force_native_check Law1ForceNativeTest.cpp)
  target_link_libraries(shell_layered_law1_force_native_check PRIVATE shell_layered_law1_native_reference GTest::gtest_main)
  target_compile_options(shell_layered_law1_force_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME shell_layered_law1_force_native COMMAND shell_layered_law1_force_native_check)
endif()
if(TL_LAYERED_LAW1_FORCE_CUDA)
  if(NOT TL_LAYERED_LAW1_FORCE_NATIVE)
    message(FATAL_ERROR "LAW1 force CUDA qualification requires independent native oracle")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_layered_law1_force_cuda_check Law1ForceCudaTest.cu)
  target_link_libraries(shell_layered_law1_force_cuda_check PRIVATE shell_layered_law1_native_reference GTest::gtest_main CUDA::cudart)
  target_compile_features(shell_layered_law1_force_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_layered_law1_force_cuda_check PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_layered_law1_force_cuda COMMAND shell_layered_law1_force_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
