cmake_minimum_required(VERSION 3.18)
project(TLNodalVehicleQualification LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(nodal_vehicle_layout_check LayoutTest.cpp)
target_compile_features(nodal_vehicle_layout_check PRIVATE cxx_std_17)
target_include_directories(nodal_vehicle_layout_check PRIVATE "${TL_ROOT}")
target_link_libraries(nodal_vehicle_layout_check PRIVATE GTest::gtest_main)
add_test(NAME nodal_vehicle_layout COMMAND nodal_vehicle_layout_check)
set_tests_properties(nodal_vehicle_layout PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
option(TL_NODAL_VEHICLE_ENABLE_CUDA "Actual vehicle-sized owner allocation/step/readback gate" OFF)
if(TL_NODAL_VEHICLE_ENABLE_CUDA)
  enable_language(CUDA)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" nodal-owner)
  add_executable(nodal_vehicle_owner_check VehicleOwnerFixture.cu VehicleOwnerTest.cu)
  target_link_libraries(nodal_vehicle_owner_check PRIVATE tl_explicit_nodal_state GTest::gtest_main)
  set_target_properties(nodal_vehicle_owner_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(nodal_vehicle_owner_check PRIVATE
    --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME nodal_vehicle_owner COMMAND nodal_vehicle_owner_check)
  set_tests_properties(nodal_vehicle_owner PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
endif()
