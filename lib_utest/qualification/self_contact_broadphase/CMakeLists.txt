cmake_minimum_required(VERSION 3.22)
project(TLSelfContactBroadphase LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(SELF_CONTACT_BROADPHASE_CUDA "Actual S0 borrowed-view broadphase and legacy CUDA gate" OFF)
if(SELF_CONTACT_BROADPHASE_CUDA)
  enable_language(CUDA)
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/SelfContactBroadphase.cmake")
add_executable(self_contact_broadphase_host HostTest.cpp)
target_link_libraries(self_contact_broadphase_host PRIVATE tl_self_contact_broadphase_values GTest::gtest_main)
target_compile_options(self_contact_broadphase_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME self_contact_broadphase_host COMMAND self_contact_broadphase_host)
set_tests_properties(self_contact_broadphase_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
if(SELF_CONTACT_BROADPHASE_CUDA)
  add_executable(self_contact_broadphase_cuda CudaTest.cu)
  target_link_libraries(self_contact_broadphase_cuda PRIVATE tl_self_contact_broadphase GTest::gtest_main)
  set_target_properties(self_contact_broadphase_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(self_contact_broadphase_cuda PRIVATE
    --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME self_contact_broadphase_cuda COMMAND self_contact_broadphase_cuda)
  set_tests_properties(self_contact_broadphase_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
  include("${TL_ROOT}/lib_src/collision/HydroelasticBroadphase.cmake")
  add_executable(self_contact_broadphase_legacy "${TL_ROOT}/lib_utest/utest_surface_broadphase.cu")
  target_link_libraries(self_contact_broadphase_legacy PRIVATE tl_collision_broadphase GTest::gtest_main)
  set_target_properties(self_contact_broadphase_legacy PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME self_contact_broadphase_legacy COMMAND self_contact_broadphase_legacy)
  set_tests_properties(self_contact_broadphase_legacy PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
add_test(NAME self_contact_broadphase_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(self_contact_broadphase_identity PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
