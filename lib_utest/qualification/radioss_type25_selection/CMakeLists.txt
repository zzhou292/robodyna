cmake_minimum_required(VERSION 3.22)
project(TLRadiossType25Selection LANGUAGES CXX Fortran)
enable_testing()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TYPE25_SELECTION_CUDA "Build actual retained native selection CUDA qualification" OFF)
if(TYPE25_SELECTION_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(NOT CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA")
    message(FATAL_ERROR "Selection CUDA qualification requires NVIDIA nvcc")
  endif()
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Selected precision qualification requires GNU C++")
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(TYPE25_GEOMETRY_CUDA ${TYPE25_SELECTION_CUDA})
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_local_geometry" geometry)
include("${TL_ROOT}/lib_src/collision/RadiossType25Selection.cmake")
set(native "${CMAKE_CURRENT_BINARY_DIR}/native")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE prepared ERROR_VARIABLE diagnostic)
if(NOT prepared EQUAL 0)
  message(FATAL_ERROR "Pinned retained oracle preparation failed: ${diagnostic}")
endif()
file(GLOB native_pins CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/native/original/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  native/prepare.py native/Sources.py native/source-manifest.json
  native/LocalBoundary.F90 native/Observations.F90 native/RetainedWrapper.F90
  ../radioss_type25_normal/native/original/constant_mod.F ${native_pins})
add_library(type25_selection_oracle STATIC NativeOracle.cpp
  "${native}/Constants.F90" "${native}/LocalBoundary.F90" "${native}/Observations.F90"
  "${native}/CorRetained.F" "${native}/DstRetained.F" "${native}/GlobRetained.F"
  "${native}/RetainedWrapper.F90")
target_include_directories(type25_selection_oracle PRIVATE "${native}")
target_link_libraries(type25_selection_oracle PUBLIC tl_radioss_type25_selection)
target_compile_options(type25_selection_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(type25_selection_host HostTest.cpp)
target_link_libraries(type25_selection_host PRIVATE type25_selection_oracle GTest::gtest_main)
add_test(NAME type25_selection_host COMMAND type25_selection_host)
add_executable(type25_selection_consumer_host ConsumerTest.cpp)
target_link_libraries(type25_selection_consumer_host PRIVATE tl_radioss_type25_selection GTest::gtest_main)
add_test(NAME type25_selection_consumer_host COMMAND type25_selection_consumer_host)
add_test(NAME type25_selection_sources COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
set_tests_properties(type25_selection_host type25_selection_consumer_host type25_selection_sources
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
if(TYPE25_SELECTION_CUDA)
  add_executable(type25_selection_cuda CudaTest.cu)
  target_link_libraries(type25_selection_cuda PRIVATE type25_selection_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_selection_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME type25_selection_cuda COMMAND type25_selection_cuda)
  add_executable(type25_selection_consumer_cuda ConsumerCudaTest.cu)
  target_link_libraries(type25_selection_consumer_cuda PRIVATE tl_radioss_type25_selection GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_selection_consumer_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME type25_selection_consumer_cuda COMMAND type25_selection_consumer_cuda)
  set_tests_properties(type25_selection_cuda type25_selection_consumer_cuda PROPERTIES
    RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
