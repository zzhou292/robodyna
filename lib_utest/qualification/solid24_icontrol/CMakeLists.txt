cmake_minimum_required(VERSION 3.22)
project(TLSolid24Icontrol LANGUAGES CXX C)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
enable_testing()
add_test(NAME solid24_icontrol_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_check.py")
add_test(NAME solid24_icontrol_source_tamper COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/test_sources.py")
option(TL_SOLID24_ICONTROL_NATIVE "Complete controlled HEPH native caller; no product link" OFF)
if(TL_SOLID24_ICONTROL_NATIVE)
  enable_language(Fortran)
  set(TL_SOLID24_FORCE_NATIVE ON CACHE BOOL "Reuse qualified independent material/geometry" FORCE)
  set(TL_SOLID24_FORCE_CUDA OFF CACHE BOOL "No CUDA product work in this native milestone" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../solid24_force" legacy_heph)
  add_subdirectory(native)
  add_executable(solid24_icontrol_native_test NativeTest.cpp)
  target_include_directories(solid24_icontrol_native_test PRIVATE "${TL_ROOT}")
  target_compile_features(solid24_icontrol_native_test PRIVATE cxx_std_17)
  target_compile_options(solid24_icontrol_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(solid24_icontrol_native_test PRIVATE solid24_ic1_native Eigen3::Eigen GTest::gtest_main)
  add_test(NAME solid24_icontrol_native_test COMMAND solid24_icontrol_native_test)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
