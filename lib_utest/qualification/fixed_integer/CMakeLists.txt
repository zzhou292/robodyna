cmake_minimum_required(VERSION 3.22)
project(TLFixedInteger LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(fixed_integer_oracle STATIC Oracle.cpp)
target_include_directories(fixed_integer_oracle PUBLIC "${TL_ROOT}")
target_compile_features(fixed_integer_oracle PUBLIC cxx_std_17)
target_link_libraries(fixed_integer_oracle PUBLIC GTest::gtest)
target_compile_options(fixed_integer_oracle PRIVATE -fno-fast-math -ffp-contract=off)
add_executable(fixed_integer_host HostTest.cpp)
target_link_libraries(fixed_integer_host PRIVATE fixed_integer_oracle GTest::gtest_main)
add_test(NAME fixed_integer_host COMMAND fixed_integer_host)
add_test(NAME fixed_integer_extraction COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_extraction.py")
set_tests_properties(fixed_integer_host fixed_integer_extraction PROPERTIES
  RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 90 LABELS "unit;host;exact-integer")
option(FIXED_INTEGER_CUDA "Build bounded real CUDA primitive parity" OFF)
if(FIXED_INTEGER_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(fixed_integer_cuda CudaTest.cu)
  target_link_libraries(fixed_integer_cuda PRIVATE fixed_integer_oracle CUDA::cudart GTest::gtest_main)
  set_target_properties(fixed_integer_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(fixed_integer_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME fixed_integer_cuda COMMAND fixed_integer_cuda)
  set_tests_properties(fixed_integer_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1
    TIMEOUT 120 LABELS "unit;cuda;exact-integer;differential")
endif()
