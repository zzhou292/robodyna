cmake_minimum_required(VERSION 3.22)
project(TLLayeredLaw1Qualification LANGUAGES CXX)
option(TL_LAYERED_LAW1_NATIVE "Independent native SIGEPS01C and NIP3 qualification" ON)
option(TL_LAYERED_LAW1_CUDA "Actual CUDA layered LAW1/native comparison" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
add_executable(shell_layered_law1_value_check Law1ValueTest.cpp)
target_include_directories(shell_layered_law1_value_check PRIVATE "${TL_ROOT}")
target_link_libraries(shell_layered_law1_value_check PRIVATE GTest::gtest_main)
target_compile_features(shell_layered_law1_value_check PRIVATE cxx_std_17)
target_compile_options(shell_layered_law1_value_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME shell_layered_law1_values COMMAND shell_layered_law1_value_check)
if(TL_LAYERED_LAW1_NATIVE)
  enable_language(Fortran)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/law1" native-law1)
  add_executable(shell_layered_law1_native_check Law1NativeTest.cpp)
  target_link_libraries(shell_layered_law1_native_check PRIVATE law1_point_native GTest::gtest_main)
  target_compile_options(shell_layered_law1_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME shell_layered_law1_native COMMAND shell_layered_law1_native_check)
  # Existing J2 qualification is a direct regression gate for shared NIP3 helpers.
  add_executable(shell_layered_j2_check "${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_j2/ShellLayeredJ2Test.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_j2/ShellLayeredJ2CovarianceTest.cpp")
  target_link_libraries(shell_layered_j2_check PRIVATE law44_point_native GTest::gtest_main)
  target_compile_options(shell_layered_j2_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME shell_layered_j2_check COMMAND shell_layered_j2_check)
endif()
if(TL_LAYERED_LAW1_CUDA)
  if(NOT TL_LAYERED_LAW1_NATIVE)
    message(FATAL_ERROR "CUDA qualification requires independent native oracle")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_layered_law1_cuda_check Law1CudaTest.cu)
  target_link_libraries(shell_layered_law1_cuda_check PRIVATE law1_point_native GTest::gtest_main CUDA::cudart)
  target_compile_features(shell_layered_law1_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_layered_law1_cuda_check PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_layered_law1_cuda COMMAND shell_layered_law1_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
