cmake_minimum_required(VERSION 3.22)
project(TLFixedContactFacets LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/FixedContactFacetBinding.cmake")
add_executable(fixed_contact_facets_host BindingTest.cpp MappingTest.cpp ApproximationTest.cpp RigidSkinTest.cpp)
target_link_libraries(fixed_contact_facets_host PRIVATE tl_fixed_contact_facets GTest::gtest_main)
target_compile_options(fixed_contact_facets_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME fixed_contact_facets_host COMMAND fixed_contact_facets_host)
set_tests_properties(fixed_contact_facets_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
