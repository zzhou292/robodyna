cmake_minimum_required(VERSION 3.22)
project(TLRadiossType25TiedRemoval LANGUAGES C CXX Fortran)
enable_testing()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "The pinned serial native oracle qualifies GNU C++ and Fortran")
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/RadiossType25TiedRemoval.cmake")
set(native "${CMAKE_CURRENT_BINARY_DIR}/native")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE code ERROR_VARIABLE error)
if(NOT code EQUAL 0)
  message(FATAL_ERROR "Native tied-removal source preparation failed: ${error}")
endif()
file(GLOB donors CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/native/original/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${donors}
  native/prepare.py native/source-manifest.json native/Boundary.F90 native/Wrapper.F90
  ../radioss_type25_search_startup/native/Sources.py ../radioss_type25_selection/native/Sources.py)
set(generated Constants.F90 Boundary.F90 REMN_I2OP.F PRE_I2.F ZERONM_TAGD.F UPGRADE_REMNODE2.F Orders.c Wrapper.F90)
list(TRANSFORM generated PREPEND "${native}/")
add_library(type25_tied_removal_oracle STATIC NativeOracle.cpp ${generated})
set_target_properties(type25_tied_removal_oracle PROPERTIES Fortran_MODULE_DIRECTORY "${native}")
target_include_directories(type25_tied_removal_oracle PRIVATE "${native}")
target_link_libraries(type25_tied_removal_oracle PUBLIC tl_radioss_type25_tied_removal)
target_compile_options(type25_tied_removal_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-cpp;-ffixed-line-length-none;-ffree-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(type25_tied_removal_host HostTest.cpp)
target_link_libraries(type25_tied_removal_host PRIVATE type25_tied_removal_oracle GTest::gtest_main)
add_test(NAME type25_tied_removal_host COMMAND type25_tied_removal_host)
add_executable(type25_tied_removal_consumer Consumer.cpp)
target_link_libraries(type25_tied_removal_consumer PRIVATE tl_radioss_type25_tied_removal)
add_test(NAME type25_tied_removal_consumer COMMAND type25_tied_removal_consumer)
add_test(NAME type25_tied_removal_source COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
set_tests_properties(type25_tied_removal_host type25_tied_removal_consumer type25_tied_removal_source
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_search_startup" search_regressions)
