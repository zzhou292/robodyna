cmake_minimum_required(VERSION 3.22)
project(RigidContactResponse LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/collision/RigidNormalResponse.cmake")
add_executable(rigid_contact_response_host HostTest.cpp DenseTest.cpp)
target_link_libraries(rigid_contact_response_host PRIVATE tl_rigid_normal_response Eigen3::Eigen GTest::gtest_main)
target_compile_options(rigid_contact_response_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME rigid_contact_response_host COMMAND rigid_contact_response_host)
option(RIGID_CONTACT_NATIVE "Reuse pinned rigid wrench and principal acceleration oracle" OFF)
if(RIGID_CONTACT_NATIVE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_language(Fortran)
  add_subdirectory("../nodal_rigid_group" "native-rigid" EXCLUDE_FROM_ALL)
  add_executable(rigid_contact_response_native NativeTest.cpp)
  target_link_libraries(rigid_contact_response_native PRIVATE tl_rigid_normal_response nodal_rigid_native GTest::gtest_main)
  target_compile_options(rigid_contact_response_native PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME rigid_contact_response_native COMMAND rigid_contact_response_native)
  add_test(NAME rigid_contact_response_source_identity COMMAND "${Python3_EXECUTABLE}"
    "${tl_root}/lib_utest/qualification/nodal_rigid_group/native/verify_sources.py")
endif()
option(RIGID_CONTACT_CUDA "Execute rigid normal response values on CUDA" OFF)
if(RIGID_CONTACT_CUDA)
  enable_language(CUDA)
  add_executable(rigid_contact_response_cuda CudaTest.cu)
  target_link_libraries(rigid_contact_response_cuda PRIVATE tl_rigid_normal_response GTest::gtest_main)
  set_target_properties(rigid_contact_response_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(rigid_contact_response_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME rigid_contact_response_cuda COMMAND rigid_contact_response_cuda)
endif()
