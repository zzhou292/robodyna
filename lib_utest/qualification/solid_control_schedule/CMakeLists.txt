cmake_minimum_required(VERSION 3.22)
project(TLSolidControlSchedule LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
enable_testing()
add_subdirectory("${TL_ROOT}/lib_src/elements/solids/resident" resident)
add_executable(solid_control_schedule_host HostTest.cpp)
target_link_libraries(solid_control_schedule_host PRIVATE tl_solid_batch_values GTest::gtest_main)
target_compile_options(solid_control_schedule_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME solid_control_schedule_host COMMAND solid_control_schedule_host)
add_executable(solid_control_schedule_legacy ../solid_model/HostTest.cpp)
target_link_libraries(solid_control_schedule_legacy PRIVATE tl_solid_model GTest::gtest_main)
add_test(NAME solid_control_schedule_legacy COMMAND solid_control_schedule_legacy)
add_executable(solid_control_schedule_extended ../extended_solid_model/ValueTest.cpp ../extended_solid_model/IdentityTest.cpp ../extended_solid_model/BudgetTest.cpp ../extended_solid_model/ResidentGuardTest.cpp)
target_link_libraries(solid_control_schedule_extended PRIVATE tl_solid_batch_values GTest::gtest_main)
add_test(NAME solid_control_schedule_extended COMMAND solid_control_schedule_extended)
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
