cmake_minimum_required(VERSION 3.22)
project(TLType13MeasurementReadTile LANGUAGES CXX C)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(oracle "${CMAKE_CURRENT_BINARY_DIR}/oracle")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py" --output "${oracle}" COMMAND_ERROR_IS_FATAL ANY)
add_executable(type13_read_tile_host HostTest.cpp)
target_include_directories(type13_read_tile_host PRIVATE "${TL_ROOT}" "${oracle}" ${CUDAToolkit_INCLUDE_DIRS})
target_compile_features(type13_read_tile_host PRIVATE cxx_std_17)
target_compile_options(type13_read_tile_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(type13_read_tile_host PRIVATE GTest::gtest_main)
add_test(NAME type13_read_tile_host COMMAND type13_read_tile_host)
add_test(NAME type13_read_tile_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/source_proof.py")
option(TYPE13_READ_TILE_CUDA "Frozen serial CUDA control comparison" OFF)
option(TYPE13_READ_TILE_OWNER "Actual native owner rollback and full production caller" OFF)
if(TYPE13_READ_TILE_CUDA)
  enable_language(CUDA)
  add_executable(type13_read_tile_cuda CudaTest.cu)
  target_include_directories(type13_read_tile_cuda PRIVATE "${TL_ROOT}" "${oracle}")
  target_link_libraries(type13_read_tile_cuda PRIVATE GTest::gtest_main CUDA::cudart)
  set_target_properties(type13_read_tile_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(type13_read_tile_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true --ftz=false --ptxas-options=-v -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME type13_read_tile_cuda COMMAND type13_read_tile_cuda)
endif()
if(TYPE13_READ_TILE_OWNER)
  if(NOT TYPE13_READ_TILE_CUDA)
    message(FATAL_ERROR "Owner requires CUDA")
  endif()
  set(TYPE13_RESIDENT_NATIVE ON CACHE BOOL "" FORCE)
  set(TYPE13_RESIDENT_CUDA ON CACHE BOOL "" FORCE)
  set(TYPE13_RESIDENT_ORIGINAL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../type13_resident" type13-owner)
endif()
get_property(checks DIRECTORY PROPERTY TESTS)
set_tests_properties(${checks} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
