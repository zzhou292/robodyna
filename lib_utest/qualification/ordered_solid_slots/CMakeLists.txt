cmake_minimum_required(VERSION 3.22)
project(TLOrderedSolidSlots LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(ordered_solid_slots_host HostTest.cpp ../tied_shell_patch/RepeatedAssemblyTest.cpp)
target_include_directories(ordered_solid_slots_host PRIVATE "${TL_ROOT}")
target_compile_features(ordered_solid_slots_host PRIVATE cxx_std_17)
target_compile_options(ordered_solid_slots_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(ordered_solid_slots_host PRIVATE GTest::gtest_main)
add_test(NAME ordered_solid_slots_host COMMAND ordered_solid_slots_host)
option(TL_ORDERED_SOLID_SLOTS_CUDA "Run actual ordered source-slot CUDA gate" OFF)
if(TL_ORDERED_SOLID_SLOTS_CUDA)
  enable_language(CUDA)
  add_executable(ordered_solid_slots_cuda CudaTest.cu ../tied_shell_patch/RepeatedAssemblyCudaTest.cu)
  target_include_directories(ordered_solid_slots_cuda PRIVATE "${TL_ROOT}")
  target_compile_features(ordered_solid_slots_cuda PRIVATE cuda_std_17)
  target_compile_options(ordered_solid_slots_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;-Xcompiler=-fno-fast-math,-ffp-contract=off>)
  target_link_libraries(ordered_solid_slots_cuda PRIVATE GTest::gtest_main)
  add_test(NAME ordered_solid_slots_cuda COMMAND ordered_solid_slots_cuda)
endif()
