cmake_minimum_required(VERSION 3.22)
project(TLT3Qualification LANGUAGES CXX Fortran)
option(TL_T3_ENABLE_CUDA "Build and require actual T3 device qualification" ON)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_sources.py"
  RESULT_VARIABLE t3_port_verify_status ERROR_VARIABLE t3_port_verify_error)
if(NOT t3_port_verify_status EQUAL 0)
  message(FATAL_ERROR "T3 port verification failed: ${t3_port_verify_error}")
endif()
add_custom_target(t3_verify_port COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/verify_sources.py" VERBATIM)
add_subdirectory("${TL_ROOT}/lib_src/elements/t3" "tl-t3")
# Native R1/R2 regression targets stay enabled; no edits to their owning package.
set(T3_R2_BUILD ON CACHE BOOL "Build native prescribed rates" FORCE)
set(T3_R3_BUILD OFF CACHE BOOL "Force/history outside this port stage" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/t3" "native-t3")
foreach(stage Startup Kinematics)
  string(TOLOWER "${stage}" lower)
  add_executable(t3_${lower}_port_check "T3${stage}Test.cpp")
  add_dependencies(t3_${lower}_port_check t3_verify_port)
  target_link_libraries(t3_${lower}_port_check PRIVATE tl_t3 t3_r2_native GTest::gtest_main)
  target_compile_options(t3_${lower}_port_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME t3_${lower}_port_check COMMAND t3_${lower}_port_check)
  set_tests_properties(t3_${lower}_port_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 45)
endforeach()
if(TL_T3_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(t3_port_cuda_check T3PortCudaTest.cu)
  add_dependencies(t3_port_cuda_check t3_verify_port)
  set_target_properties(t3_port_cuda_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_link_libraries(t3_port_cuda_check PRIVATE tl_t3 t3_r2_native CUDA::cudart GTest::gtest_main)
  target_compile_options(t3_port_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME t3_port_cuda_check COMMAND t3_port_cuda_check)
  set_tests_properties(t3_port_cuda_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
endif()
