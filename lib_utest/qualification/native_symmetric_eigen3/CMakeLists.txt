cmake_minimum_required(VERSION 3.22)
project(TLNativeSymmetricSpectrum LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
function(spectrum_target name)
  target_include_directories(${name} PRIVATE "${TL_ROOT}")
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_link_libraries(${name} PRIVATE GTest::gtest_main)
  target_compile_options(${name} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
endfunction()
add_executable(native_spectrum_host_check HostTest.cpp)
spectrum_target(native_spectrum_host_check)
add_test(NAME native_spectrum_host COMMAND native_spectrum_host_check)
add_test(NAME native_spectrum_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
option(SPECTRUM_NATIVE "Independent complete VALPVEC_V oracle" OFF)
option(SPECTRUM_CUDA "Actual CUDA values against the native oracle" OFF)
if(SPECTRUM_NATIVE OR SPECTRUM_CUDA)
  enable_language(C)
  enable_language(Fortran)
  add_subdirectory(native)
  add_executable(native_spectrum_reference_check NativeTest.cpp)
  spectrum_target(native_spectrum_reference_check)
  target_link_libraries(native_spectrum_reference_check PRIVATE symmetric_spectrum_native)
  add_test(NAME native_spectrum_reference COMMAND native_spectrum_reference_check)
endif()
if(SPECTRUM_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(native_spectrum_cuda_check CudaTest.cu)
  spectrum_target(native_spectrum_cuda_check)
  target_compile_features(native_spectrum_cuda_check PRIVATE cuda_std_17)
  target_compile_options(native_spectrum_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(native_spectrum_cuda_check PRIVATE symmetric_spectrum_native CUDA::cudart)
  add_test(NAME native_spectrum_cuda COMMAND native_spectrum_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
