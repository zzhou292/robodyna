cmake_minimum_required(VERSION 3.22)
project(TLBeam18Model LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
option(BEAM18_MODEL_LEGACY "Build owning old ledger regressions" ON)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/assembly/NodalCoefficientLedger.cmake")
include("${TL_ROOT}/lib_src/assembly/NodalMassBinding.cmake")
if(BEAM18_MODEL_LEGACY)
  add_subdirectory(../extended_solid_ledger legacy)
endif()
add_executable(beam18_model_host ModelTest.cpp LedgerTest.cpp SolidOrderTest.cpp)
target_link_libraries(beam18_model_host PRIVATE tl_nodal_coefficient_ledger tl_nodal_mass_binding GTest::gtest_main)
target_compile_options(beam18_model_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME beam18_model_host COMMAND beam18_model_host)
option(BEAM18_MODEL_OUTPUT "Host immutable output-range checks (CUDA headers only)" ON)
if(BEAM18_MODEL_OUTPUT)
  include("${TL_ROOT}/lib_src/elements/ShellPhysicalOwnerValues.cmake")
  add_executable(beam18_model_output OutputTest.cpp)
  target_link_libraries(beam18_model_output PRIVATE tl_shell_physical_owner_values GTest::gtest_main)
  target_compile_options(beam18_model_output PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME beam18_model_output COMMAND beam18_model_output)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(BEAM18_MODEL_NATIVE "Independent qualified PMASS and original142 snapshot/ledger gate" OFF)
set(BEAM18_MODEL_SOURCE_FIXTURE "" CACHE PATH "Authenticated original142 geometry")
if(BEAM18_MODEL_NATIVE)
  enable_language(Fortran)
  add_subdirectory(../beam18_reference/native native)
  add_executable(beam18_model_native NativeTest.cpp ../beam18_reference/NativeOracle.cpp)
  target_include_directories(beam18_model_native PRIVATE "${TL_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/../beam18_reference")
  target_link_libraries(beam18_model_native PRIVATE tl_nodal_coefficient_ledger beam18_native GTest::gtest_main)
  target_compile_options(beam18_model_native PRIVATE -fno-fast-math -ffp-contract=off)
  if(BEAM18_MODEL_SOURCE_FIXTURE)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/../beam18_force/prepare_fixture.py"
      --fixture "${BEAM18_MODEL_SOURCE_FIXTURE}" --output "${CMAKE_CURRENT_BINARY_DIR}/fixture/OriginalFixture.h"
      RESULT_VARIABLE status)
    if(NOT status EQUAL 0)
      message(FATAL_ERROR "Beam18 model source authentication failed")
    endif()
    target_include_directories(beam18_model_native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture")
    target_compile_definitions(beam18_model_native PRIVATE BEAM18_MODEL_ORIGINAL)
  endif()
  add_test(NAME beam18_model_native COMMAND beam18_model_native)
endif()
add_test(NAME beam18_model_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_source.py")
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
