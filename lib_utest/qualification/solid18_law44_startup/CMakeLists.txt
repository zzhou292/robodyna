cmake_minimum_required(VERSION 3.22)
project(TLRearSolid18Startup LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
option(REAR18_STARTUP_NATIVE "Independent native TT0 and carried intervals" OFF)
option(REAR18_STARTUP_CUDA "Actual device external scratch, TT0 and rollback" OFF)
set(REAR18_FORCE_NATIVE "${REAR18_STARTUP_NATIVE}")
set(REAR18_FORCE_CUDA "${REAR18_STARTUP_CUDA}")
add_subdirectory(../solid18_law44_force recurrence)

function(rear_startup_host target)
  add_executable(${target} ${ARGN})
  target_include_directories(${target} PRIVATE "${TL_ROOT}")
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(${target} PRIVATE GTest::gtest_main)
  add_test(NAME ${target} COMMAND ${target})
endfunction()
rear_startup_host(rear18_startup_host HostTest.cpp ../solid18_law44_force/PacketValues.cpp)
rear_startup_host(rear18_startup_point_host ../solid_law44_point/HostTest.cpp)

if(REAR18_STARTUP_NATIVE OR REAR18_STARTUP_CUDA)
  add_library(rear18_startup_oracle STATIC NativeOracle.cpp)
  target_include_directories(rear18_startup_oracle PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(rear18_startup_oracle PUBLIC rear18_force_oracle)
  target_compile_options(rear18_startup_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  add_executable(rear18_startup_native NativeTest.cpp PointNativeTest.cpp)
  target_link_libraries(rear18_startup_native PRIVATE rear18_startup_oracle GTest::gtest_main)
  target_compile_options(rear18_startup_native PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME rear18_startup_native COMMAND rear18_startup_native)
  rear_startup_host(rear18_startup_point_native ../solid_law44_point/NativeTest.cpp
                    ../solid_law44_point/NativeOracle.cpp)
  target_link_libraries(rear18_startup_point_native PRIVATE solid_law44_native)
endif()
if(REAR18_STARTUP_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(rear18_startup_cuda CudaTest.cu)
  target_compile_features(rear18_startup_cuda PRIVATE cxx_std_17 cuda_std_17)
  target_link_libraries(rear18_startup_cuda PRIVATE rear18_startup_oracle CUDA::cudart GTest::gtest_main)
  add_executable(rear18_startup_point_cuda ../solid_law44_point/CudaTest.cu
                  ../solid_law44_point/NativeOracle.cpp)
  target_include_directories(rear18_startup_point_cuda PRIVATE "${TL_ROOT}")
  target_compile_features(rear18_startup_point_cuda PRIVATE cxx_std_17 cuda_std_17)
  target_link_libraries(rear18_startup_point_cuda PRIVATE solid_law44_native CUDA::cudart GTest::gtest_main)
  foreach(target rear18_startup_cuda rear18_startup_point_cuda)
    target_compile_options(${target} PRIVATE
      "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>"
      "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
    add_test(NAME ${target} COMMAND ${target})
  endforeach()
endif()
