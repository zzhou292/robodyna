cmake_minimum_required(VERSION 3.22)
project(TLCinLimiter LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(cin_limiter_host ValuesTest.cpp)
target_include_directories(cin_limiter_host PRIVATE "${TL_ROOT}")
target_link_libraries(cin_limiter_host PRIVATE GTest::gtest_main CUDA::toolkit)
target_compile_features(cin_limiter_host PRIVATE cxx_std_17)
target_compile_options(cin_limiter_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME cin_limiter_host COMMAND cin_limiter_host)
option(CIN_LIMITER_CUDA "Actual owner diagnostic and rollback gate" OFF)
if(CIN_LIMITER_CUDA)
  enable_language(CUDA)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" owner)
  add_executable(cin_limiter_cuda OwnerTest.cu ../rigid_assembly_owner/Fixture.cpp)
  target_link_libraries(cin_limiter_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main)
  set_target_properties(cin_limiter_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(cin_limiter_cuda PRIVATE
    $<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>
    $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>)
  add_test(NAME cin_limiter_cuda COMMAND cin_limiter_cuda)
endif()
add_test(NAME cin_limiter_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
