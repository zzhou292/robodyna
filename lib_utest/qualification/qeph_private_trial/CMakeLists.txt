cmake_minimum_required(VERSION 3.22)
project(QephPrivateTrial LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(QEPH_PRIVATE_TRIAL_CUDA "Build real resident owner/native regressions" OFF)
set(TL_RESIDENT_TAB1_NATIVE ON CACHE BOOL "" FORCE)
set(TL_RESIDENT_TAB1_CUDA ${QEPH_PRIVATE_TRIAL_CUDA} CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../resident_shell_tab1" resident-regressions)
add_executable(qeph_private_trial_host PrivateTrialTest.cpp)
target_include_directories(qeph_private_trial_host PRIVATE "${TL_ROOT}")
target_compile_features(qeph_private_trial_host PRIVATE cxx_std_17)
target_compile_options(qeph_private_trial_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(qeph_private_trial_host PRIVATE tl_shell_batch_plasticity GTest::gtest_main)
add_test(NAME qeph_private_trial_host COMMAND qeph_private_trial_host)
set_tests_properties(qeph_private_trial_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME qeph_private_trial_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(qeph_private_trial_source PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
if(QEPH_PRIVATE_TRIAL_CUDA)
  target_sources(resident_shell_tab1_cuda_check PRIVATE "${CMAKE_CURRENT_LIST_DIR}/PrivateTrialCudaTest.cu")
endif()
