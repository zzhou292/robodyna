cmake_minimum_required(VERSION 3.22)
project(TLHephLaw42Force LANGUAGES CXX C)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(solid24_force_host HostTest.cpp)
target_compile_features(solid24_force_host PRIVATE cxx_std_17)
target_include_directories(solid24_force_host PRIVATE "${TL_ROOT}")
target_compile_options(solid24_force_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(solid24_force_host PRIVATE Eigen3::Eigen GTest::gtest_main)
add_test(NAME solid24_force_host COMMAND solid24_force_host)
option(TL_SOLID24_FORCE_NATIVE "Build complete independent selected HEPH force oracle" OFF)
option(TL_SOLID24_FORCE_CUDA "Build device-owned force recurrence versus native" OFF)
if(TL_SOLID24_FORCE_CUDA AND NOT TL_SOLID24_FORCE_NATIVE)
  message(FATAL_ERROR "CUDA force gate requires the independent native oracle")
endif()
if(TL_SOLID24_FORCE_NATIVE)
  enable_language(Fortran)
  if(NOT TARGET law42_native)
    add_subdirectory(../solid_law42_point/native "${CMAKE_CURRENT_BINARY_DIR}/law42_point")
  endif()
  if(NOT TARGET law42_solid_caller_native)
    add_subdirectory(../solid_law42_caller/native "${CMAKE_CURRENT_BINARY_DIR}/law42_caller")
  endif()
  if(NOT TARGET solid24_reference_native)
    add_subdirectory(../solid24_reference/native "${CMAKE_CURRENT_BINARY_DIR}/reference")
  endif()
  add_subdirectory(native)
  add_executable(solid24_force_native_test NativeTest.cpp SourceNativeTest.cpp)
  target_include_directories(solid24_force_native_test PRIVATE "${TL_ROOT}")
  target_compile_features(solid24_force_native_test PRIVATE cxx_std_17)
  target_compile_options(solid24_force_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(solid24_force_native_test PRIVATE heph_force_native Eigen3::Eigen GTest::gtest_main)
  add_test(NAME solid24_force_native_test COMMAND solid24_force_native_test)
endif()
if(TL_SOLID24_FORCE_CUDA)
  enable_language(CUDA)
  add_executable(solid24_force_cuda CudaTest.cu SourceCudaTest.cu)
  target_include_directories(solid24_force_cuda PRIVATE "${TL_ROOT}")
  target_compile_features(solid24_force_cuda PRIVATE cuda_std_17)
  target_compile_options(solid24_force_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-fno-fast-math,-ffp-contract=off)
  target_link_libraries(solid24_force_cuda PRIVATE heph_force_native Eigen3::Eigen GTest::gtest_main)
  add_test(NAME solid24_force_cuda COMMAND solid24_force_cuda)
endif()
add_test(NAME solid24_force_fixture_identity
  COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/../solid_common/source_fixture/verify_fixture.py")
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
