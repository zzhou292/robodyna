cmake_minimum_required(VERSION 3.22)
project(NativeGpuBatchBenchmark LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(NATIVE_BENCHMARK_ROOT "" CACHE PATH "Pinned maintained native benchmark directory")
foreach(source Cases.cpp Cases.h Results.cpp Results.h)
  if(NOT EXISTS "${NATIVE_BENCHMARK_ROOT}/${source}")
    message(FATAL_ERROR "NATIVE_BENCHMARK_ROOT must identify the maintained bounded-cohort benchmark")
  endif()
endforeach()
include("${TL_ROOT}/lib_src/collision/RepresentedIntervalCrossingGpu.cmake")
add_executable(native_gpu_batch_benchmark Main.cpp Selection.h
  "${NATIVE_BENCHMARK_ROOT}/Cases.cpp" "${NATIVE_BENCHMARK_ROOT}/Results.cpp")
target_include_directories(native_gpu_batch_benchmark PRIVATE "${NATIVE_BENCHMARK_ROOT}")
target_link_libraries(native_gpu_batch_benchmark PRIVATE tl_represented_interval_crossing_gpu)
target_compile_features(native_gpu_batch_benchmark PRIVATE cxx_std_17)
target_compile_options(native_gpu_batch_benchmark PRIVATE -fno-fast-math -ffp-contract=off)
