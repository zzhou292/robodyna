cmake_minimum_required(VERSION 3.22)
project(TLMappedWallScatter LANGUAGES CXX)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
option(MAPPED_WALL_SCATTER_CUDA "Frozen serial CUDA comparison" OFF)
add_executable(mapped_wall_scatter_host HostTest.cpp Fixture.cpp)
target_include_directories(mapped_wall_scatter_host PRIVATE "${TL_ROOT}" "${CUDAToolkit_INCLUDE_DIRS}")
target_compile_features(mapped_wall_scatter_host PRIVATE cxx_std_17)
target_compile_options(mapped_wall_scatter_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(mapped_wall_scatter_host PRIVATE GTest::gtest_main)
add_test(NAME mapped_wall_scatter_host COMMAND mapped_wall_scatter_host)
if(MAPPED_WALL_SCATTER_CUDA)
  enable_language(CUDA)
  add_executable(mapped_wall_scatter_cuda CudaTest.cu Fixture.cpp)
  target_include_directories(mapped_wall_scatter_cuda PRIVATE "${TL_ROOT}" "${CUDAToolkit_INCLUDE_DIRS}")
  target_link_libraries(mapped_wall_scatter_cuda PRIVATE GTest::gtest_main CUDA::cudart)
  set_target_properties(mapped_wall_scatter_cuda PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17)
  target_compile_options(mapped_wall_scatter_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME mapped_wall_scatter_cuda COMMAND mapped_wall_scatter_cuda)
  set_tests_properties(mapped_wall_scatter_cuda PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE PROCESSORS 1)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/Verify.cmake")
add_test(NAME mapped_wall_scatter_identity COMMAND "${CMAKE_COMMAND}" "-DTL_ROOT=${TL_ROOT}"
  -P "${CMAKE_CURRENT_LIST_DIR}/Verify.cmake")
set_tests_properties(mapped_wall_scatter_host mapped_wall_scatter_identity PROPERTIES
  TIMEOUT 30 RUN_SERIAL TRUE PROCESSORS 1)
