cmake_minimum_required(VERSION 3.22)
project(TLShellLayeredJ2Qualification LANGUAGES CXX Fortran)
option(TL_SHELL_LAYERED_J2_ENABLE_CUDA "Build the one-thread layered adapter device check" ON)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
if(NOT TARGET law44_point_native)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../native/law44" native-law44)
endif()
add_executable(shell_layered_j2_check ShellLayeredJ2Test.cpp)
target_include_directories(shell_layered_j2_check PRIVATE "${TL_ROOT}")
target_link_libraries(shell_layered_j2_check PRIVATE law44_point_native GTest::gtest_main)
target_compile_options(shell_layered_j2_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME shell_layered_j2_check COMMAND shell_layered_j2_check)
set_tests_properties(shell_layered_j2_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
if(TL_SHELL_LAYERED_J2_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_layered_j2_cuda_check ShellLayeredJ2CudaTest.cu)
  target_include_directories(shell_layered_j2_cuda_check PRIVATE "${TL_ROOT}")
  target_compile_features(shell_layered_j2_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_link_libraries(shell_layered_j2_cuda_check PRIVATE GTest::gtest_main CUDA::cudart)
  target_compile_options(shell_layered_j2_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_layered_j2_cuda_check COMMAND shell_layered_j2_cuda_check)
  set_tests_properties(shell_layered_j2_cuda_check PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
endif()
