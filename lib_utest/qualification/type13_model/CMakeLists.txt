cmake_minimum_required(VERSION 3.18)
project(TLType13Model LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/type13/Type13Model.cmake")
find_package(GTest REQUIRED)
add_executable(type13_model_test ValueTest.cpp)
target_link_libraries(type13_model_test PRIVATE tl_type13_model GTest::gtest_main)
target_compile_options(type13_model_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type13_model_values COMMAND type13_model_test)
option(TYPE13_MODEL_ORIGINAL_NATIVE "Build original source model/native coefficient checks" OFF)
if(TYPE13_MODEL_ORIGINAL_NATIVE)
  enable_language(Fortran)
  set(TYPE13_NATIVE_CHECKS ON CACHE BOOL "Native startup oracle required" FORCE)
  set(TYPE13_CUDA_CHECKS OFF CACHE BOOL "No GPU in host model gate" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../type13" native-startup)
  add_executable(type13_model_original_test OriginalTest.cpp "../type13/NativeOracle.cpp")
  target_link_libraries(type13_model_original_test PRIVATE tl_type13_model type13_native GTest::gtest_main)
  target_compile_options(type13_model_original_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME type13_model_original COMMAND type13_model_original_test)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES PROCESSORS 1 RUN_SERIAL TRUE TIMEOUT 120)
