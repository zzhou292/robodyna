cmake_minimum_required(VERSION 3.22)
project(TLRadiossType25Normal LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "P1 native block oracle qualifies GNU Fortran only")
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/RadiossType25Normal.cmake")
set(native "${CMAKE_CURRENT_BINARY_DIR}/native/NormalReference.F")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE prepared ERROR_VARIABLE diagnostic)
if(NOT prepared EQUAL 0)
  message(FATAL_ERROR "Pinned normal oracle preparation failed: ${diagnostic}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  native/prepare.py native/Wrapper.F.in native/source-manifest.json
  native/original/i25for3.F native/original/constant_mod.F)
add_library(type25_normal_oracle STATIC NativeOracle.cpp "${native}")
target_link_libraries(type25_normal_oracle PUBLIC tl_radioss_type25_normal)
target_include_directories(type25_normal_oracle PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_compile_options(type25_normal_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>"
  "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
add_executable(type25_normal_host HostTest.cpp UnitsTest.cpp HeaderTest.cpp)
target_link_libraries(type25_normal_host PRIVATE type25_normal_oracle GTest::gtest_main)
target_compile_options(type25_normal_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type25_normal_host COMMAND type25_normal_host)
add_test(NAME type25_normal_sources COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
set_tests_properties(type25_normal_host type25_normal_sources PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(TYPE25_NORMAL_CUDA "Build actual selected native/host/CUDA response parity" OFF)
if(TYPE25_NORMAL_CUDA)
  get_property(languages GLOBAL PROPERTY ENABLED_LANGUAGES)
  if(NOT "CUDA" IN_LIST languages)
    enable_language(CUDA)
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_executable(type25_normal_cuda CudaTest.cu)
  target_link_libraries(type25_normal_cuda PRIVATE type25_normal_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_normal_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(type25_normal_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME type25_normal_cuda COMMAND type25_normal_cuda)
  set_tests_properties(type25_normal_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
