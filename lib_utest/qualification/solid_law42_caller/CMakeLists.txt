cmake_minimum_required(VERSION 3.22)
project(TLLaw42Caller LANGUAGES CXX C)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(law42_caller_host_check HostTest.cpp)
target_include_directories(law42_caller_host_check PRIVATE "${TL_ROOT}")
target_compile_features(law42_caller_host_check PRIVATE cxx_std_17)
target_compile_options(law42_caller_host_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(law42_caller_host_check PRIVATE GTest::gtest_main Eigen3::Eigen)
add_test(NAME law42_caller_host COMMAND law42_caller_host_check)
set_tests_properties(law42_caller_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)

option(LAW42_CALLER_NATIVE "Complete independent native solid caller" OFF)
if(LAW42_CALLER_NATIVE)
  enable_language(Fortran)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(NOT TARGET law42_native)
    add_subdirectory(../solid_law42_point/native law42_point_native)
  endif()
  add_subdirectory(native)
  add_executable(law42_caller_native_check NativeTest.cpp)
  target_include_directories(law42_caller_native_check PRIVATE "${TL_ROOT}")
  target_compile_features(law42_caller_native_check PRIVATE cxx_std_17)
  target_compile_options(law42_caller_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(law42_caller_native_check PRIVATE GTest::gtest_main Eigen3::Eigen law42_solid_caller_native)
  add_test(NAME law42_caller_native COMMAND law42_caller_native_check)
  set_tests_properties(law42_caller_native PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
endif()
