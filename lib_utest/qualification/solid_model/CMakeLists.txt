cmake_minimum_required(VERSION 3.22)
project(TLSolidMechanicsModel LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${TL_ROOT}/lib_src/elements/solids" "${CMAKE_CURRENT_BINARY_DIR}/model")

target_link_libraries(solid_model_host PRIVATE tl_solid_model GTest::gtest_main)
target_compile_options(solid_model_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME solid_model_host COMMAND solid_model_host)
option(TL_SOLID_MODEL_SOURCE "Build all-original2412 immutable model gate" OFF)
if(TL_SOLID_MODEL_SOURCE)
  add_executable(solid_model_source OriginalTest.cpp)
  target_link_libraries(solid_model_source PRIVATE tl_solid_model GTest::gtest_main)
  target_compile_options(solid_model_source PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME solid_model_source COMMAND solid_model_source)
  add_test(NAME solid_model_rubber_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/../solid_common/source_fixture/verify_fixture.py")
  add_test(NAME solid_model_adhesive_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/../solid18_reference/verify_source.py")
  add_test(NAME solid_model_material_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_SOURCE_DIR}/../solid_law36_point/verify_source.py")
