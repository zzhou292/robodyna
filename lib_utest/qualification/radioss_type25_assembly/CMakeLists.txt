cmake_minimum_required(VERSION 3.22)
project(TLType25Assembly LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TYPE25_ASSEMBLY_CUDA "Build actual ASS0 CUDA qualification" OFF)
if(TYPE25_ASSEMBLY_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(NOT CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA")
    message(FATAL_ERROR "Selected qualification requires NVIDIA nvcc")
  endif()
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Pinned native oracle qualifies GNU C++ and Fortran")
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/collision/RadiossType25Assembly.cmake")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(native "${CMAKE_CURRENT_BINARY_DIR}/NativeAssembly.F")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE prepared ERROR_VARIABLE diagnostic)
if(NOT prepared EQUAL 0)
  message(FATAL_ERROR "Native ASS0 preparation failed: ${diagnostic}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  native/prepare.py native/source-manifest.json native/Wrapper.F.in native/original/i25ass3.F
  ../radioss_type25_normal/native/original/i25for3.F)
add_library(type25_assembly_oracle STATIC NativeOracle.cpp "${native}")
target_link_libraries(type25_assembly_oracle PUBLIC tl_radioss_type25_assembly)
target_compile_options(type25_assembly_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(type25_assembly_host HostTest.cpp)
target_link_libraries(type25_assembly_host PRIVATE type25_assembly_oracle GTest::gtest_main)
add_test(NAME type25_assembly_host COMMAND type25_assembly_host)
add_executable(type25_assembly_consumer HeaderTest.cpp)
target_link_libraries(type25_assembly_consumer PRIVATE tl_radioss_type25_assembly GTest::gtest_main)
add_test(NAME type25_assembly_consumer COMMAND type25_assembly_consumer)
add_test(NAME type25_assembly_source COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
set_tests_properties(type25_assembly_host type25_assembly_consumer type25_assembly_source
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 90)
if(TYPE25_ASSEMBLY_CUDA)
  add_executable(type25_assembly_cuda CudaTest.cu)
  target_link_libraries(type25_assembly_cuda PRIVATE type25_assembly_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_assembly_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME type25_assembly_cuda COMMAND type25_assembly_cuda)
  set_tests_properties(type25_assembly_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
