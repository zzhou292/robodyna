cmake_minimum_required(VERSION 3.22)
project(TLSolidLaw44Point LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(solid_law44_host_check HostTest.cpp)
target_include_directories(solid_law44_host_check PRIVATE "${TL_ROOT}")
target_compile_features(solid_law44_host_check PRIVATE cxx_std_17)
target_compile_options(solid_law44_host_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(solid_law44_host_check PRIVATE GTest::gtest_main)
add_test(NAME solid_law44_host COMMAND solid_law44_host_check)
add_executable(solid_law44_shared_host_check ../solid_law36_point/HostTest.cpp)
target_include_directories(solid_law44_shared_host_check PRIVATE "${TL_ROOT}")
target_compile_features(solid_law44_shared_host_check PRIVATE cxx_std_17)
target_compile_options(solid_law44_shared_host_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(solid_law44_shared_host_check PRIVATE GTest::gtest_main)
add_test(NAME solid_law44_shared_host COMMAND solid_law44_shared_host_check)
add_test(NAME solid_law44_source_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
option(SOLID_LAW44_NATIVE "Complete independent native material caller" OFF)
option(SOLID_LAW44_CUDA "Actual CUDA point recurrence and rollback" OFF)
if(SOLID_LAW44_NATIVE OR SOLID_LAW44_CUDA)
  enable_language(Fortran)
  add_subdirectory(native)
  add_executable(solid_law44_native_check NativeTest.cpp NativeOracle.cpp)
  target_include_directories(solid_law44_native_check PRIVATE "${TL_ROOT}")
  target_compile_features(solid_law44_native_check PRIVATE cxx_std_17)
  target_compile_options(solid_law44_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(solid_law44_native_check PRIVATE solid_law44_native GTest::gtest_main)
  add_test(NAME solid_law44_native COMMAND solid_law44_native_check)
endif()
if(SOLID_LAW44_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(solid_law44_cuda_check CudaTest.cu NativeOracle.cpp)
  target_include_directories(solid_law44_cuda_check PRIVATE "${TL_ROOT}")
  target_compile_features(solid_law44_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(solid_law44_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
  target_link_libraries(solid_law44_cuda_check PRIVATE solid_law44_native CUDA::cudart GTest::gtest_main)
  add_test(NAME solid_law44_cuda COMMAND solid_law44_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
