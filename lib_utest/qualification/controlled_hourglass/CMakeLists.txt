cmake_minimum_required(VERSION 3.22)
project(TLControlledHourglass LANGUAGES CXX C Fortran)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory(native)
add_executable(controlled_hourglass_host HostTest.cpp)
target_compile_features(controlled_hourglass_host PRIVATE cxx_std_17)
target_include_directories(controlled_hourglass_host PRIVATE "${TL_ROOT}")
target_compile_options(controlled_hourglass_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(controlled_hourglass_host PRIVATE controlled_hourglass_native GTest::gtest_main)
add_test(NAME controlled_hourglass_host COMMAND controlled_hourglass_host)
add_test(NAME controlled_hourglass_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/source_check.py")
option(TL_CONTROLLED_HOURGLASS_CUDA "Actual CUDA values versus independent native leaf" OFF)
if(TL_CONTROLLED_HOURGLASS_CUDA)
  enable_language(CUDA)
  add_executable(controlled_hourglass_cuda CudaTest.cu)
  target_compile_features(controlled_hourglass_cuda PRIVATE cuda_std_17)
  target_include_directories(controlled_hourglass_cuda PRIVATE "${TL_ROOT}")
  target_compile_options(controlled_hourglass_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-fno-fast-math,-ffp-contract=off)
  target_link_libraries(controlled_hourglass_cuda PRIVATE controlled_hourglass_native GTest::gtest_main)
  add_test(NAME controlled_hourglass_cuda COMMAND controlled_hourglass_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
