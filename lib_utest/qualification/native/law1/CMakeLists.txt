cmake_minimum_required(VERSION 3.22)
project(law1_layered_native_reference LANGUAGES CXX Fortran)
if(NOT TARGET law44_point_native)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../law44" native-law44)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(law1_prepared "${CMAKE_CURRENT_BINARY_DIR}/prepared")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py"
  --output "${law1_prepared}" RESULT_VARIABLE status ERROR_VARIABLE diagnostic)
if(NOT status EQUAL 0)
  message(FATAL_ERROR "Native LAW1 source preparation failed: ${diagnostic}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS prepare_sources.py source-manifest.json)
add_custom_target(law1_point_verify_sources COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sources.py" --output "${law1_prepared}" --check VERBATIM)
add_library(law1_point_native STATIC NativeReference.cpp NativePoint.F NativeSection.F90 "${law1_prepared}/Sigeps01c.F")
add_dependencies(law1_point_native law1_point_verify_sources)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
set_target_properties(law1_point_native PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
foreach(property INCLUDE_DIRECTORIES COMPILE_DEFINITIONS COMPILE_OPTIONS)
  get_target_property(value law44_point_native ${property})
  set_property(TARGET law1_point_native PROPERTY ${property} "${value}")
endforeach()
get_target_property(law1_modules law44_point_native Fortran_MODULE_DIRECTORY)
target_include_directories(law1_point_native BEFORE PRIVATE "${law1_prepared}" "${law1_modules}")
target_include_directories(law1_point_native PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(law1_point_native PUBLIC cxx_std_17)
target_link_libraries(law1_point_native PUBLIC law44_point_native)
