cmake_minimum_required(VERSION 3.22)
project(TLMappedWallRemovalEvents LANGUAGES CXX)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(CUDAToolkit REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(MAPPED_WALL_REMOVAL_CUDA "Frozen complete removal caller and timing" OFF)
add_executable(mapped_wall_removal_host HostTest.cpp)
target_include_directories(mapped_wall_removal_host PRIVATE "${TL_ROOT}" "${CUDAToolkit_INCLUDE_DIRS}")
target_compile_features(mapped_wall_removal_host PRIVATE cxx_std_17)
target_compile_options(mapped_wall_removal_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(mapped_wall_removal_host PRIVATE GTest::gtest_main)
add_test(NAME mapped_wall_removal_host COMMAND mapped_wall_removal_host)
if(MAPPED_WALL_REMOVAL_CUDA)
  enable_language(CUDA)
  add_executable(mapped_wall_removal_cuda CudaTest.cu)
  target_include_directories(mapped_wall_removal_cuda PRIVATE "${TL_ROOT}" "${CUDAToolkit_INCLUDE_DIRS}")
  target_link_libraries(mapped_wall_removal_cuda PRIVATE CUDA::cudart GTest::gtest_main)
  set_target_properties(mapped_wall_removal_cuda PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17)
  target_compile_options(mapped_wall_removal_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME mapped_wall_removal_cuda COMMAND mapped_wall_removal_cuda)
  set_tests_properties(mapped_wall_removal_cuda PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE PROCESSORS 1)
endif()
add_test(NAME mapped_wall_removal_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
set_tests_properties(mapped_wall_removal_host mapped_wall_removal_identity PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
