cmake_minimum_required(VERSION 3.22)
project(TLSolid6zForce LANGUAGES CXX C)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(solid6z_force_host HostTest.cpp FrameTensorTest.cpp)
target_compile_features(solid6z_force_host PRIVATE cxx_std_17)
target_include_directories(solid6z_force_host PRIVATE "${TL_ROOT}")
target_compile_options(solid6z_force_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(solid6z_force_host PRIVATE GTest::gtest_main Eigen3::Eigen)
add_test(NAME solid6z_force_host COMMAND solid6z_force_host)
set_tests_properties(solid6z_force_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TL_SOLID6Z_FORCE_NATIVE "Build corrected native S6Z caller and original mapped source gate" OFF)
option(TL_SOLID6Z_FORCE_CUDA "Build CUDA versus independent native force gate" OFF)
if(TL_SOLID6Z_FORCE_CUDA AND NOT TL_SOLID6Z_FORCE_NATIVE)
  message(FATAL_ERROR "CUDA force qualification requires native oracle")
endif()
if(TL_SOLID6Z_FORCE_NATIVE)
  enable_language(Fortran)
  add_subdirectory(../solid_law42_point/native law42_native)
  add_subdirectory(../solid_law42_caller/native law42_caller_native)
  add_subdirectory(../solid6z_reference/native solid6z_reference_native)
  add_subdirectory(native)
  add_executable(solid6z_force_native_test NativeTest.cpp SourceNativeTest.cpp)
  target_compile_features(solid6z_force_native_test PRIVATE cxx_std_17)
  target_include_directories(solid6z_force_native_test PRIVATE "${TL_ROOT}")
  target_compile_options(solid6z_force_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(solid6z_force_native_test PRIVATE solid6z_force_native solid6z_reference_native
    GTest::gtest_main Eigen3::Eigen)
  add_test(NAME solid6z_force_native_test COMMAND solid6z_force_native_test --gtest_filter=Solid6zForceNative.*)
  add_test(NAME solid6z_force_source_test COMMAND solid6z_force_native_test --gtest_filter=Solid6zForceSource.*)
endif()
if(TL_SOLID6Z_FORCE_CUDA)
  enable_language(CUDA)
  add_executable(solid6z_force_cuda CudaTest.cu)
  target_compile_features(solid6z_force_cuda PRIVATE cuda_std_17)
  target_include_directories(solid6z_force_cuda PRIVATE "${TL_ROOT}")
  target_compile_options(solid6z_force_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-fno-fast-math,-ffp-contract=off)
  target_link_libraries(solid6z_force_cuda PRIVATE solid6z_force_native solid6z_reference_native
    GTest::gtest_main Eigen3::Eigen)
  add_test(NAME solid6z_force_cuda COMMAND solid6z_force_cuda)
endif()
add_test(NAME solid6z_force_source_identity COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_SOURCE_DIR}/../solid_common/source_fixture/verify_fixture.py")
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
