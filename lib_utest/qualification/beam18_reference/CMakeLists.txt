cmake_minimum_required(VERSION 3.22)
project(TLBeam18Reference LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
option(BEAM18_NATIVE "Independent complete native reference/mass values" OFF)
option(BEAM18_CUDA "Actual CUDA reference and rejection" OFF)
set(BEAM18_SOURCE_FIXTURE "" CACHE PATH "Original142 authenticated geometry fixture")
add_executable(beam18_host HostTest.cpp)
target_include_directories(beam18_host PRIVATE "${TL_ROOT}")
target_compile_features(beam18_host PRIVATE cxx_std_17)
target_compile_options(beam18_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(beam18_host PRIVATE GTest::gtest_main)
add_test(NAME beam18_host COMMAND beam18_host)
if(BEAM18_NATIVE OR BEAM18_CUDA)
  enable_language(Fortran)
  add_subdirectory(native)
  add_library(beam18_oracle STATIC NativeOracle.cpp)
  target_include_directories(beam18_oracle PUBLIC "${TL_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_features(beam18_oracle PUBLIC cxx_std_17)
  target_compile_options(beam18_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(beam18_oracle PUBLIC beam18_native GTest::gtest)
  add_executable(beam18_native_check NativeTest.cpp)
  target_link_libraries(beam18_native_check PRIVATE beam18_oracle GTest::gtest_main)
  target_compile_options(beam18_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME beam18_native_check COMMAND beam18_native_check)
endif()
if(BEAM18_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(beam18_cuda_check CudaTest.cu)
  target_compile_features(beam18_cuda_check PRIVATE cuda_std_17)
  target_compile_options(beam18_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  target_link_libraries(beam18_cuda_check PRIVATE beam18_oracle CUDA::cudart GTest::gtest_main)
  add_test(NAME beam18_cuda_check COMMAND beam18_cuda_check)
endif()
if(BEAM18_SOURCE_FIXTURE)
  set(fixture_header "${CMAKE_CURRENT_BINARY_DIR}/fixture/OriginalFixture.h")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_fixture.py"
    --fixture "${BEAM18_SOURCE_FIXTURE}" --output "${fixture_header}" RESULT_VARIABLE status)
  if(NOT status EQUAL 0)
    message(FATAL_ERROR "Original142 beam fixture authentication failed")
  endif()
  foreach(target beam18_native_check beam18_cuda_check)
    if(TARGET ${target})
      target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture")
      target_compile_definitions(${target} PRIVATE BEAM18_ORIGINAL)
    endif()
  endforeach()
endif()
add_test(NAME beam18_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
