cmake_minimum_required(VERSION 3.22)
project(TLType13ResidentQualification LANGUAGES CXX)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
include("${TL_ROOT}/lib_src/elements/type13/resident/Values.cmake")
add_executable(type13_resident_host_test HostTest.cpp)
target_link_libraries(type13_resident_host_test PRIVATE tl_type13_batch_values GTest::gtest_main)
target_compile_options(type13_resident_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME type13_resident_host COMMAND type13_resident_host_test)
option(TYPE13_RESIDENT_NATIVE "Build the independent native TYPE13 resident oracle" OFF)
option(TYPE13_RESIDENT_ORIGINAL "Build the original 4442-connection gate (root lane)" OFF)
option(TYPE13_RESIDENT_CUDA "Build actual common-owner CUDA qualification (root lane)" OFF)
if(TYPE13_RESIDENT_NATIVE)
  enable_language(Fortran)
  include("${CMAKE_CURRENT_LIST_DIR}/Native.cmake")
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME type13_resident_source_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/native/verify_sources.py")
  add_test(NAME type13_resident_original_fixture_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../type13/source_fixture/verify_fixture.py")
  add_executable(type13_resident_native_test NativeTest.cpp)
  target_link_libraries(type13_resident_native_test PRIVATE tl_type13_batch_values
    type13_resident_native GTest::gtest_main)
  target_compile_options(type13_resident_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME type13_resident_native COMMAND type13_resident_native_test)
  if(TYPE13_RESIDENT_ORIGINAL)
    add_executable(type13_resident_original_test OriginalTest.cpp)
    target_link_libraries(type13_resident_original_test PRIVATE tl_type13_batch_values
      type13_resident_native GTest::gtest_main)
    target_compile_options(type13_resident_original_test PRIVATE -fno-fast-math -ffp-contract=off)
    add_test(NAME type13_resident_original COMMAND type13_resident_original_test)
  endif()
elseif(TYPE13_RESIDENT_ORIGINAL OR TYPE13_RESIDENT_CUDA)
  message(FATAL_ERROR "Actual/original qualification requires the independent native oracle")
endif()
if(TYPE13_RESIDENT_CUDA)
  enable_language(CUDA)
  add_subdirectory("${TL_ROOT}/lib_src/solvers" nodal-owner)
  include("${TL_ROOT}/lib_src/elements/type13/resident/Batch.cmake")
  add_executable(type13_resident_cuda_test OwnerFixture.cu OwnerCompare.cu OwnerTest.cu CinOwnerTest.cu)
  if(TYPE13_RESIDENT_ORIGINAL)
    target_sources(type13_resident_cuda_test PRIVATE OriginalOwnerTest.cu)
  endif()
  target_link_libraries(type13_resident_cuda_test PRIVATE tl_type13_batch
    type13_resident_native GTest::gtest_main)
  set_target_properties(type13_resident_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(type13_resident_cuda_test PRIVATE
    --fmad=false --prec-div=true --prec-sqrt=true --ftz=false -Xcompiler=-fno-fast-math,-ffp-contract=off)
  add_test(NAME type13_resident_cuda COMMAND type13_resident_cuda_test)
  set_tests_properties(type13_resident_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 300)
endif()
