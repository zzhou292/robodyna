cmake_minimum_required(VERSION 3.22)
project(TLRadiossType25Friction LANGUAGES CXX Fortran)
enable_testing()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TYPE25_FRICTION_CUDA "Build actual coupled response/phase CUDA qualification" OFF)
if(TYPE25_FRICTION_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  if(NOT CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA")
    message(FATAL_ERROR "Selected CUDA qualification requires NVIDIA nvcc")
  endif()
endif()
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Selected host precision qualification requires GNU C++")
endif()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
# Reuse the native normal oracle and rerun its owning regressions after conversion extraction.
set(TYPE25_NORMAL_CUDA ${TYPE25_FRICTION_CUDA})
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../radioss_type25_normal" normal)
include("${TL_ROOT}/lib_src/collision/RadiossType25Friction.cmake")
set(native "${CMAKE_CURRENT_BINARY_DIR}/native")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py"
  --output "${native}" RESULT_VARIABLE prepared ERROR_VARIABLE diagnostic)
if(NOT prepared EQUAL 0)
  message(FATAL_ERROR "Pinned friction oracle preparation failed: ${diagnostic}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  native/prepare.py native/actual_fixture.py native/FrictionWrapper.F.in native/PhaseWrapper.F.in native/source-manifest.json
  native/original/i25irtlm.F native/original/i25mainf.F)
add_library(type25_friction_oracle STATIC NativeOracle.cpp NativePhaseOracle.cpp
  "${native}/FrictionReference.F" "${native}/PhaseReference.F")
target_include_directories(type25_friction_oracle PUBLIC "${native}")
target_link_libraries(type25_friction_oracle PUBLIC type25_normal_oracle tl_radioss_type25_friction)
target_compile_options(type25_friction_oracle PRIVATE
  "$<$<COMPILE_LANGUAGE:Fortran>:-ffixed-line-length-none;-fcheck=bounds;-fbacktrace;-fno-fast-math;-ffp-contract=off>")
add_executable(type25_friction_host HostTest.cpp HistoryTest.cpp FrictionUnitsTest.cpp)
target_link_libraries(type25_friction_host PRIVATE type25_friction_oracle GTest::gtest_main)
add_test(NAME type25_friction_host COMMAND type25_friction_host)
add_test(NAME type25_friction_sources COMMAND "${Python3_EXECUTABLE}" -B
  "${CMAKE_CURRENT_LIST_DIR}/native/prepare.py" --output "${native}" --check)
add_executable(type25_friction_consumer_host ConsumerTest.cpp)
# No private precision flags and no native/Fortran dependency on this consumer.
target_link_libraries(type25_friction_consumer_host PRIVATE tl_radioss_type25_friction GTest::gtest_main)
add_test(NAME type25_friction_consumer_host COMMAND type25_friction_consumer_host)
set_tests_properties(type25_friction_host type25_friction_sources type25_friction_consumer_host
  PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 90)
if(TYPE25_FRICTION_CUDA)
  add_executable(type25_friction_cuda CudaTest.cu HistoryCudaTest.cu)
  target_link_libraries(type25_friction_cuda PRIVATE type25_friction_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_friction_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME type25_friction_cuda COMMAND type25_friction_cuda)
  add_executable(type25_friction_consumer_cuda ConsumerCudaTest.cu)
  target_link_libraries(type25_friction_consumer_cuda PRIVATE tl_radioss_type25_friction GTest::gtest_main CUDA::cudart)
  set_target_properties(type25_friction_consumer_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME type25_friction_consumer_cuda COMMAND type25_friction_consumer_cuda)
  set_tests_properties(type25_friction_cuda type25_friction_consumer_cuda PROPERTIES
    RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
