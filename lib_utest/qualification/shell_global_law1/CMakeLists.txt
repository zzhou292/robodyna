cmake_minimum_required(VERSION 3.22)
project(TLShellGlobalLaw1 LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(TL_GLOBAL_LAW1_CUDA "Build actual device global LAW1 qualification" OFF)
set(TL_LAYERED_LAW1_FORCE_CUDA ${TL_GLOBAL_LAW1_CUDA})
set(TL_LAYERED_LAW1_FORCE_NATIVE ON)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../shell_layered_law1_force" layered-regressions)
include("${CMAKE_CURRENT_LIST_DIR}/native/Native.cmake")
add_executable(shell_global_law1_host ValueTest.cpp QephNativeTest.cpp T3NativeTest.cpp UnitsTest.cpp NptComparisonTest.cpp)
target_link_libraries(shell_global_law1_host PRIVATE shell_global_law1_native_reference shell_layered_law1_native_reference GTest::gtest_main)
target_include_directories(shell_global_law1_host PRIVATE "${TL_ROOT}")
target_compile_features(shell_global_law1_host PRIVATE cxx_std_17)
target_compile_options(shell_global_law1_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME shell_global_law1_host COMMAND shell_global_law1_host)
add_test(NAME shell_global_law1_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/verify_sources.py")
add_executable(shell_global_law1_consumer Consumer.cpp)
target_include_directories(shell_global_law1_consumer PRIVATE "${TL_ROOT}")
target_compile_features(shell_global_law1_consumer PRIVATE cxx_std_17)
target_compile_options(shell_global_law1_consumer PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME shell_global_law1_consumer COMMAND shell_global_law1_consumer)
if(TL_GLOBAL_LAW1_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_global_law1_cuda CudaTest.cu)
  target_link_libraries(shell_global_law1_cuda PRIVATE shell_global_law1_native_reference shell_layered_law1_native_reference CUDA::cudart GTest::gtest_main)
  target_include_directories(shell_global_law1_cuda PRIVATE "${TL_ROOT}")
  target_compile_features(shell_global_law1_cuda PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_global_law1_cuda PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_global_law1_cuda COMMAND shell_global_law1_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
