cmake_minimum_required(VERSION 3.22)
project(TLShellPhysicalBinding LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${TL_ROOT}/lib_src/assembly/ShellPhysicalBinding.cmake")
add_executable(shell_physical_binding_host HostTest.cpp)
target_link_libraries(shell_physical_binding_host PRIVATE tl_shell_physical_binding GTest::gtest_main)
target_compile_options(shell_physical_binding_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME shell_physical_binding_host COMMAND shell_physical_binding_host)
set_tests_properties(shell_physical_binding_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
