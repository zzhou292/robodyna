cmake_minimum_required(VERSION 3.22)
project(TLExplicitEmptyCin LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
option(EMPTY_CIN_CUDA "Build actual constrained physical-owner qualification" OFF)
set(QT_MAPPED_CUDA ${EMPTY_CIN_CUDA})
if(EMPTY_CIN_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../qt_mapped" qt-legacy)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../rigid_assembly_binding" rigid-binding)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(empty_cin_host HostTest.cpp)
target_link_libraries(empty_cin_host PRIVATE tl_shell_mapped_startup tl_tied_cin_attachment GTest::gtest_main)
target_compile_features(empty_cin_host PRIVATE cxx_std_17)
target_compile_options(empty_cin_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME empty_cin_host COMMAND empty_cin_host)
set_tests_properties(empty_cin_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
if(EMPTY_CIN_CUDA)
  add_executable(empty_cin_cuda CudaTest.cu)
  target_link_libraries(empty_cin_cuda PRIVATE tl_qeph_batch tl_t3_batch tl_tied_cin_attachment GTest::gtest_main CUDA::cudart)
  set_target_properties(empty_cin_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(empty_cin_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME empty_cin_cuda COMMAND empty_cin_cuda)
  add_executable(empty_cin_legacy_structural_cuda ../cin_physical_timestep/OwnerTest.cu ../rigid_assembly_owner/Fixture.cpp)
  target_link_libraries(empty_cin_legacy_structural_cuda PRIVATE tl_explicit_nodal_state GTest::gtest_main CUDA::cudart)
  set_target_properties(empty_cin_legacy_structural_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(empty_cin_legacy_structural_cuda PRIVATE
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false>")
  add_test(NAME empty_cin_legacy_structural_cuda COMMAND empty_cin_legacy_structural_cuda)
  set_tests_properties(empty_cin_cuda empty_cin_legacy_structural_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
