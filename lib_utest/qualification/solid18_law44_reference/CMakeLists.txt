cmake_minimum_required(VERSION 3.22)
project(TLRearSolid18Reference LANGUAGES CXX)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(rear18_host_check HostTest.cpp)
target_include_directories(rear18_host_check PRIVATE "${TL_ROOT}")
target_compile_features(rear18_host_check PRIVATE cxx_std_17)
target_compile_options(rear18_host_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(rear18_host_check PRIVATE GTest::gtest_main)
add_test(NAME rear18_host COMMAND rear18_host_check)
option(REAR18_NATIVE "Independent native reference and repeated-node mass" OFF)
option(REAR18_CUDA "Actual device reference and rollback" OFF)
set(REAR18_SOURCE_FIXTURE "" CACHE PATH "Authenticated original rear306 binary fixture")
if(REAR18_NATIVE OR REAR18_CUDA)
  enable_language(Fortran)
  add_subdirectory(../solid18_reference/native "${CMAKE_CURRENT_BINARY_DIR}/reference-native")
  add_library(rear18_native STATIC native/NativePacket.F90 native/NativeScatter.F)
  add_dependencies(rear18_native solid18_reference_native)
  target_compile_definitions(rear18_native PRIVATE MYREAL8 CPP_mach=CPP_p4linux964 COMP_GFORTRAN)
  target_compile_options(rear18_native PRIVATE -cpp -ffixed-line-length-none -ffree-line-length-none
    -fcheck=bounds -fbacktrace -fno-fast-math -ffp-contract=off)
  target_include_directories(rear18_native PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}/reference-native/prepared"
    "${CMAKE_CURRENT_BINARY_DIR}/reference-native/modules")
  target_link_libraries(rear18_native PUBLIC solid18_reference_native)
  add_executable(rear18_native_check NativeTest.cpp NativeOracle.cpp ../solid18_reference/NativeOracle.cpp)
  target_include_directories(rear18_native_check PRIVATE "${TL_ROOT}")
  target_compile_features(rear18_native_check PRIVATE cxx_std_17)
  target_compile_options(rear18_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  target_link_libraries(rear18_native_check PRIVATE rear18_native GTest::gtest_main)
  add_test(NAME rear18_native COMMAND rear18_native_check)
endif()
if(REAR18_SOURCE_FIXTURE)
  set(fixture_header "${CMAKE_CURRENT_BINARY_DIR}/fixture/OriginalFixture.h")
  execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/prepare_fixture.py"
    --fixture "${REAR18_SOURCE_FIXTURE}" --output "${fixture_header}" RESULT_VARIABLE status)
  if(NOT status EQUAL 0)
    message(FATAL_ERROR "Original rear306 fixture authentication failed")
  endif()
  if(TARGET rear18_native_check)
    target_sources(rear18_native_check PRIVATE SourceNativeTest.cpp)
    target_include_directories(rear18_native_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture" "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endif()
if(REAR18_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(rear18_cuda_check CudaTest.cu NativeOracle.cpp ../solid18_reference/NativeOracle.cpp)
  target_include_directories(rear18_cuda_check PRIVATE "${TL_ROOT}")
  target_compile_features(rear18_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(rear18_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>"
    "$<$<COMPILE_LANGUAGE:CXX>:-fno-fast-math;-ffp-contract=off>")
  target_link_libraries(rear18_cuda_check PRIVATE rear18_native CUDA::cudart GTest::gtest_main)
  if(REAR18_SOURCE_FIXTURE)
    target_compile_definitions(rear18_cuda_check PRIVATE REAR18_ORIGINAL)
    target_include_directories(rear18_cuda_check PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fixture" "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
  add_test(NAME rear18_cuda COMMAND rear18_cuda_check)
endif()
add_test(NAME rear18_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_source.py")
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
