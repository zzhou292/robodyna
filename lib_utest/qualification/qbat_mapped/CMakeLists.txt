cmake_minimum_required(VERSION 3.22)
project(TLQbatMappedQualification LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/qbat/QbatBatchValues.cmake")
include("${tl_root}/lib_src/constraints/NodalRigidAssemblyBinding.cmake")
include("${tl_root}/lib_src/constraints/tied_shell/TiedCinAttachment.cmake")
add_executable(qbat_mapped_host_test HostTest.cpp ../rigid_assembly_owner/Fixture.cpp
  ../qbat_resident/HostTest.cpp)
target_link_libraries(qbat_mapped_host_test PRIVATE tl_qbat_batch_values
  tl_nodal_rigid_assembly_binding tl_tied_cin_attachment GTest::gtest_main)
target_compile_options(qbat_mapped_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qbat_mapped_host COMMAND qbat_mapped_host_test)
set_tests_properties(qbat_mapped_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
option(QBAT_MAPPED_NATIVE "Independent native initial/current stiffness and complete scatter" OFF)
option(QBAT_MAPPED_CUDA "Actual mapped owner/source/stiffness/rollback gates" OFF)
if(QBAT_MAPPED_NATIVE)
  include("${CMAKE_CURRENT_LIST_DIR}/Native.cmake")
endif()
if(QBAT_MAPPED_CUDA)
  include("${CMAKE_CURRENT_LIST_DIR}/Cuda.cmake")
endif()

option(QBAT_MAPPED_ORIGINAL "Build original 4250+1 complete mapped host model gate" OFF)
if(QBAT_MAPPED_ORIGINAL)
  add_executable(qbat_mapped_original_test OriginalTest.cpp ../qbat_binding/OriginalFixture.cpp)
  target_link_libraries(qbat_mapped_original_test PRIVATE tl_qbat_batch_values GTest::gtest_main)
  target_compile_options(qbat_mapped_original_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME qbat_mapped_original COMMAND qbat_mapped_original_test)
  set_tests_properties(qbat_mapped_original PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
