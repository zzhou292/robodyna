cmake_minimum_required(VERSION 3.22)
project(ResidentShellTab1 LANGUAGES CXX)
enable_testing()
find_package(GTest REQUIRED)
get_filename_component(tl_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/elements/ShellBatchPlasticity.cmake")
add_executable(resident_shell_tab1_host_check BindingTest.cpp StateTest.cpp)
target_include_directories(resident_shell_tab1_host_check PRIVATE "${tl_root}")
target_link_libraries(resident_shell_tab1_host_check PRIVATE tl_shell_batch_plasticity GTest::gtest_main)
target_compile_options(resident_shell_tab1_host_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME resident_shell_tab1_host COMMAND resident_shell_tab1_host_check)
set_tests_properties(resident_shell_tab1_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
option(TL_RESIDENT_TAB1_NATIVE "Compare resident dispatcher with complete native placed force" OFF)
option(TL_RESIDENT_TAB1_CUDA "Qualify complete actual resident TAB1 collections" OFF)
if(TL_RESIDENT_TAB1_NATIVE OR TL_RESIDENT_TAB1_CUDA)
  set(TL_PLACEMENT_FORCE_NATIVE ON CACHE BOOL "" FORCE)
  set(TL_PLACEMENT_FORCE_CUDA OFF CACHE BOOL "" FORCE)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shell_placement_force" placed-reference)
  add_executable(resident_shell_tab1_native_check NativeTest.cpp)
  target_link_libraries(resident_shell_tab1_native_check PRIVATE tl_shell_batch_plasticity shell_placement_force_test_support)
  target_compile_options(resident_shell_tab1_native_check PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME resident_shell_tab1_native COMMAND resident_shell_tab1_native_check)
  set_tests_properties(resident_shell_tab1_native PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()
if(TL_RESIDENT_TAB1_CUDA)
  include(Cuda.cmake)
endif()
