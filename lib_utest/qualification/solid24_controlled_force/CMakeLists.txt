cmake_minimum_required(VERSION 3.22)
project(TLH24ControlledFull LANGUAGES CXX C Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
option(TL_H24_FULL_CUDA "Qualify complete controlled H24 on actual CUDA" OFF)
set(TL_H24_ADAPTER_CUDA ${TL_H24_FULL_CUDA} CACHE BOOL "Retain prefix CUDA regression" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../solid24_controlled_hourglass" prefix)
add_executable(h24_controlled_full_host HostTest.cpp)
target_compile_features(h24_controlled_full_host PRIVATE cxx_std_17)
target_include_directories(h24_controlled_full_host PRIVATE "${TL_ROOT}")
target_compile_options(h24_controlled_full_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(h24_controlled_full_host PRIVATE h24_adapter_native GTest::gtest_main Eigen3::Eigen)
add_test(NAME h24_controlled_full_host COMMAND h24_controlled_full_host)
if(TL_H24_FULL_CUDA)
 enable_language(CUDA)
 add_executable(h24_controlled_full_cuda CudaTest.cu)
 target_compile_features(h24_controlled_full_cuda PRIVATE cuda_std_17)
 target_include_directories(h24_controlled_full_cuda PRIVATE "${TL_ROOT}")
 target_compile_options(h24_controlled_full_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true -Xcompiler=-fno-fast-math,-ffp-contract=off)
 target_link_libraries(h24_controlled_full_cuda PRIVATE h24_adapter_native GTest::gtest_main Eigen3::Eigen)
 add_test(NAME h24_controlled_full_cuda COMMAND h24_controlled_full_cuda)
endif()
add_test(NAME h24_controlled_full_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/source_check.py")
