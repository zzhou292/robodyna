cmake_minimum_required(VERSION 3.22)
project(TLVehicleWallWeights LANGUAGES CXX)
enable_testing()
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
# Existing mixed-source fixture names CUDA owner types but runs no GPU work.
find_package(CUDAToolkit REQUIRED)
include("${TL_ROOT}/lib_src/collision/NodalWallContact.cmake")
add_executable(vehicle_wall_weights_check WeightProfileTest.cpp WeightSourceSizeTest.cpp)
target_include_directories(vehicle_wall_weights_check PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(vehicle_wall_weights_check PRIVATE tl_nodal_wall_contact GTest::gtest_main)
target_compile_options(vehicle_wall_weights_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME vehicle_wall_weights COMMAND vehicle_wall_weights_check)
set_tests_properties(vehicle_wall_weights PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE PROCESSORS 1)
