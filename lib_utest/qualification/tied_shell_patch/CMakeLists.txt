cmake_minimum_required(VERSION 3.22)
project(TLTiedShellPatch LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/constraints/tied_shell/TiedPatch.cmake")
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(tied_patch_host_test ValueTest.cpp CoefficientValueTest.cpp RepeatedAssemblyTest.cpp)
target_link_libraries(tied_patch_host_test PRIVATE tl_tied_shell_patch GTest::gtest_main)
target_compile_options(tied_patch_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME tied_patch_host COMMAND tied_patch_host_test)
add_test(NAME tied_patch_source_identity COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_LIST_DIR}/native/verify_sources.py")
option(TL_TIED_PATCH_NATIVE "Compile independent native force/motion oracle" ON)
option(TL_TIED_PATCH_CUDA "Compile actual GPU value qualification" OFF)
if(TL_TIED_PATCH_NATIVE OR TL_TIED_PATCH_CUDA)
  enable_language(Fortran)
  add_library(tied_patch_native STATIC native/NativeForce.F native/NativeMotion.F native/NativeMotionPacket.F90
    native/NativeCoefficients.F native/NativeAssembly.F)
  target_include_directories(tied_patch_native PRIVATE native native/stubs)
  target_compile_options(tied_patch_native PRIVATE -cpp -ffixed-line-length-none -ffree-line-length-none
    -fno-fast-math -ffp-contract=off -fcheck=all -finit-real=snan)
  add_library(tied_patch_oracle STATIC NativeOracle.cpp)
  target_link_libraries(tied_patch_oracle PUBLIC tl_tied_shell_patch tied_patch_native GTest::gtest)
  target_compile_options(tied_patch_oracle PRIVATE -fno-fast-math -ffp-contract=off)
  add_executable(tied_patch_native_test NativeTest.cpp CoefficientNativeTest.cpp RepeatedAssemblyNativeTest.cpp)
  target_link_libraries(tied_patch_native_test PRIVATE tied_patch_oracle GTest::gtest_main)
  target_compile_options(tied_patch_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME tied_patch_native COMMAND tied_patch_native_test)
endif()
if(TL_TIED_PATCH_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(tied_patch_cuda_test CudaTest.cu RepeatedAssemblyCudaTest.cu)
  target_link_libraries(tied_patch_cuda_test PRIVATE tied_patch_oracle GTest::gtest_main CUDA::cudart)
  set_target_properties(tied_patch_cuda_test PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(tied_patch_cuda_test PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME tied_patch_cuda COMMAND tied_patch_cuda_test)
  set_tests_properties(tied_patch_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 30)
endif()
