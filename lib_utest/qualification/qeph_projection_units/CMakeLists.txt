cmake_minimum_required(VERSION 3.22)
project(TLQephProjectionUnits LANGUAGES CXX Fortran)
enable_testing()
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(QEPH_Q1_BUILD_TESTS ON)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native/qeph" native-parent)
include("${CMAKE_CURRENT_LIST_DIR}/native/Native.cmake")
include("${TL_ROOT}/lib_src/elements/ShellBatchBinding.cmake")
add_executable(qeph_projection_replay_host HostTest.cpp IdentityTest.cpp)
target_include_directories(qeph_projection_replay_host PRIVATE "${TL_ROOT}")
target_link_libraries(qeph_projection_replay_host PRIVATE qeph_projection_replay tl_shell_batch_binding GTest::gtest_main)
target_compile_features(qeph_projection_replay_host PRIVATE cxx_std_17)
target_compile_options(qeph_projection_replay_host PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qeph_projection_replay_host COMMAND qeph_projection_replay_host)
set_tests_properties(qeph_projection_replay_host PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)

# Actual device evaluation is a separate guarded lane from host/source gates.
option(QEPH_PROJECTION_CUDA "Compile the native-backed device projection replay" OFF)
if(QEPH_PROJECTION_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(qeph_projection_profile_cuda CudaTest.cu)
  target_link_libraries(qeph_projection_profile_cuda PRIVATE qeph_projection_replay CUDA::cudart GTest::gtest_main)
  target_compile_options(qeph_projection_profile_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--expt-relaxed-constexpr>)
  set_target_properties(qeph_projection_profile_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  add_test(NAME qeph_projection_profile_cuda COMMAND qeph_projection_profile_cuda)
  set_tests_properties(qeph_projection_profile_cuda PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
endif()

# Inventory/placement regressions exercise every extended quadrilateral record.
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../qbat_binding" binding-regressions)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../shell_placement_force" placement-regressions)
