cmake_minimum_required(VERSION 3.22)
project(TLTiedSearchDriver LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${tl_root}/lib_src/constraints/tied_shell/search/TiedSearchDriverValues.cmake")
find_package(GTest REQUIRED)
add_executable(tied_search_driver_values_test ValueTest.cpp)
target_link_libraries(tied_search_driver_values_test PRIVATE tl_tied_search_driver_values GTest::gtest_main)
target_compile_options(tied_search_driver_values_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME tied_search_driver_values COMMAND tied_search_driver_values_test)
option(TL_TIED_SEARCH_DRIVER_CUDA "Build actual GPU driver checks" OFF)
if(TL_TIED_SEARCH_DRIVER_CUDA)
  enable_language(CUDA)
  include("${tl_root}/lib_src/constraints/tied_shell/search/TiedSearchDriver.cmake")
  add_executable(tied_search_driver_cuda_test CudaTest.cpp)
  target_link_libraries(tied_search_driver_cuda_test PRIVATE tl_tied_search_driver GTest::gtest_main)
  target_compile_options(tied_search_driver_cuda_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME tied_search_driver_cuda COMMAND tied_search_driver_cuda_test)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)
