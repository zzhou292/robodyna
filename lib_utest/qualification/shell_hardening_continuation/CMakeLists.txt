cmake_minimum_required(VERSION 3.22)
project(ShellHardeningContinuation LANGUAGES CXX)
option(TL_HARDENING_CONTINUATION_NATIVE "Build pinned native curve/point/failure comparisons" OFF)
option(TL_HARDENING_CONTINUATION_CUDA "Build actual CUDA continuation/failure comparisons" OFF)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_custom_target(shell_hardening_continuation_sources
  COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/verify_curves.py" VERBATIM)
add_executable(shell_hardening_continuation_values_check
  ContinuationValueTest.cpp ContinuationFailureValueTest.cpp)
add_dependencies(shell_hardening_continuation_values_check shell_hardening_continuation_sources)
target_include_directories(shell_hardening_continuation_values_check PRIVATE "${TL_ROOT}")
target_compile_features(shell_hardening_continuation_values_check PRIVATE cxx_std_17)
target_compile_options(shell_hardening_continuation_values_check PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(shell_hardening_continuation_values_check PRIVATE GTest::gtest_main)
add_test(NAME shell_hardening_continuation_values COMMAND shell_hardening_continuation_values_check)
include("${CMAKE_CURRENT_SOURCE_DIR}/../plasticity_binding/PlasticityBinding.cmake")

if(TL_HARDENING_CONTINUATION_NATIVE)
  enable_language(Fortran)
  set(TL_LAYERED_FAILURE_NATIVE ON)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_failure" layered-failure)
  # Shared C++ comparison functions and complete source caller; no duplicate law.
  add_library(continuation_failure_test_support STATIC ContinuationFailureNative.cpp
    "${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_failure/FailureNativeFixture.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/../shell_layered_failure/FailureNativeChecks.cpp")
  target_link_libraries(continuation_failure_test_support PUBLIC shell_layered_failure_native GTest::gtest_main)
  target_compile_options(continuation_failure_test_support PRIVATE -fno-fast-math -ffp-contract=off)
  add_executable(shell_hardening_continuation_native_check
    ContinuationNativeTest.cpp ContinuationFailureNativeTest.cpp NativeCurve.F90)
  add_dependencies(shell_hardening_continuation_native_check shell_hardening_continuation_sources)
  target_link_libraries(shell_hardening_continuation_native_check PRIVATE continuation_failure_test_support)
  target_compile_options(shell_hardening_continuation_native_check PRIVATE -fno-fast-math -ffp-contract=off
    "$<$<COMPILE_LANGUAGE:Fortran>:-fcheck=bounds;-fbacktrace;-ffree-line-length-none>")
  add_test(NAME shell_hardening_continuation_native COMMAND shell_hardening_continuation_native_check)
endif()
if(TL_HARDENING_CONTINUATION_CUDA)
  if(NOT TL_HARDENING_CONTINUATION_NATIVE)
    message(FATAL_ERROR "CUDA continuation qualification requires the complete native oracle")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_executable(shell_hardening_continuation_cuda_check ContinuationCudaTest.cu ContinuationFailureCudaTest.cu)
  add_dependencies(shell_hardening_continuation_cuda_check shell_hardening_continuation_sources)
  target_link_libraries(shell_hardening_continuation_cuda_check PRIVATE continuation_failure_test_support CUDA::cudart)
  target_compile_features(shell_hardening_continuation_cuda_check PRIVATE cxx_std_17 cuda_std_17)
  target_compile_options(shell_hardening_continuation_cuda_check PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--prec-div=true;--prec-sqrt=true;--ftz=false;-Xcompiler=-fno-fast-math,-ffp-contract=off>")
  add_test(NAME shell_hardening_continuation_cuda COMMAND shell_hardening_continuation_cuda_check)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
