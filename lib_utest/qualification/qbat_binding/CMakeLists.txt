cmake_minimum_required(VERSION 3.22)
project(TLQbatBindingQualification LANGUAGES CXX)
enable_testing()
get_filename_component(tl_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(GTest REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include("${tl_root}/lib_src/elements/ShellBatchPlasticityBinding.cmake")
include("${tl_root}/lib_src/assembly/NodalMassBinding.cmake")
add_executable(qbat_binding_host_test ValueTest.cpp FailureTest.cpp MassTest.cpp)
target_link_libraries(qbat_binding_host_test PRIVATE tl_shell_batch_plasticity_binding
  tl_nodal_mass_binding GTest::gtest_main)
target_compile_options(qbat_binding_host_test PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME qbat_binding_host COMMAND qbat_binding_host_test)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../t3/mixed_binding" legacy-binding)
include("${CMAKE_CURRENT_LIST_DIR}/../plasticity_binding/PlasticityBinding.cmake")

option(QBAT_BINDING_SOURCE_CHECKS "Build root-scheduled complete original 4250+1 binding" OFF)
option(QBAT_BINDING_NATIVE_CHECKS "Build root-scheduled independent existing native oracles" OFF)
option(QBAT_BINDING_CUDA_CHECKS "Build root-scheduled old participant rejection checks" OFF)
if(QBAT_BINDING_SOURCE_CHECKS OR QBAT_BINDING_NATIVE_CHECKS)
  add_executable(qbat_binding_original_test OriginalFixture.cpp OriginalTest.cpp)
  target_link_libraries(qbat_binding_original_test PRIVATE tl_shell_batch_binding GTest::gtest_main)
  target_compile_options(qbat_binding_original_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME qbat_binding_original COMMAND qbat_binding_original_test)
  add_test(NAME qbat_binding_source_identity COMMAND "${Python3_EXECUTABLE}" -B
    "${CMAKE_CURRENT_LIST_DIR}/../qbat/source_fixture/verify_fixture.py")
endif()
if(QBAT_BINDING_NATIVE_CHECKS)
  enable_language(Fortran)
  set(QBAT_NATIVE_CHECKS ON CACHE BOOL "Existing independent native reference" FORCE)
  set(QBAT_CUDA_CHECKS OFF CACHE BOOL "No unrelated device geometry gate" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../qbat" native-qbat)
  set(T3_R1_BUILD_TESTS ON CACHE BOOL "Affected original T3 native startup" FORCE)
  set(T3_R2_BUILD OFF CACHE BOOL "No T3 engine in host binding" FORCE)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native/t3" native-t3)
  add_executable(qbat_binding_native_test OriginalFixture.cpp NativeTest.cpp
    "${CMAKE_CURRENT_LIST_DIR}/../qbat/NativeOracle.cpp")
  target_link_libraries(qbat_binding_native_test PRIVATE tl_shell_batch_binding
    qbat_native t3_r1_native GTest::gtest_main)
  target_compile_options(qbat_binding_native_test PRIVATE -fno-fast-math -ffp-contract=off)
  add_test(NAME qbat_binding_native COMMAND qbat_binding_native_test)
endif()
if(QBAT_BINDING_CUDA_CHECKS)
  include("${CMAKE_CURRENT_LIST_DIR}/ClosedParticipants.cmake")
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 120)
