cmake_minimum_required(VERSION 3.22)
project(TLH24ControlledAdapter LANGUAGES CXX C Fortran)
get_filename_component(TL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
find_package(GTest REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(TL_H24_ADAPTER_CUDA "Qualify actual CUDA H24 controlled stage" OFF)
set(TL_SOLID24_FORCE_NATIVE ON CACHE BOOL "Native legacy regression" FORCE)
set(TL_SOLID24_FORCE_CUDA ${TL_H24_ADAPTER_CUDA} CACHE BOOL "CUDA legacy regression" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../solid24_force" legacy_heph)
add_subdirectory(native)
add_executable(h24_adapter_host HostTest.cpp WorkingHostTest.cpp)
target_compile_features(h24_adapter_host PRIVATE cxx_std_17)
target_include_directories(h24_adapter_host PRIVATE "${TL_ROOT}")
target_compile_options(h24_adapter_host PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(h24_adapter_host PRIVATE h24_adapter_native Eigen3::Eigen GTest::gtest_main)
add_test(NAME h24_adapter_host COMMAND h24_adapter_host)
add_executable(h24_complete_native_regression ../solid24_icontrol/NativeTest.cpp)
target_compile_features(h24_complete_native_regression PRIVATE cxx_std_17)
target_include_directories(h24_complete_native_regression PRIVATE "${TL_ROOT}")
target_compile_options(h24_complete_native_regression PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(h24_complete_native_regression PRIVATE h24_adapter_native Eigen3::Eigen GTest::gtest_main)
add_test(NAME h24_complete_native_regression COMMAND h24_complete_native_regression)
add_test(NAME h24_adapter_source COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/source_check.py")
if(TL_H24_ADAPTER_CUDA)
  enable_language(CUDA)
  add_executable(h24_adapter_cuda CudaTest.cu WorkingCudaTest.cu)
  target_compile_features(h24_adapter_cuda PRIVATE cuda_std_17)
  target_include_directories(h24_adapter_cuda PRIVATE "${TL_ROOT}")
  target_compile_options(h24_adapter_cuda PRIVATE --fmad=false --prec-div=true --prec-sqrt=true
    -Xcompiler=-fno-fast-math,-ffp-contract=off)
  target_link_libraries(h24_adapter_cuda PRIVATE h24_adapter_native Eigen3::Eigen GTest::gtest_main)
  add_test(NAME h24_adapter_cuda COMMAND h24_adapter_cuda)
endif()
get_property(tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${tests} PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 180)
