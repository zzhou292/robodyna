cmake_minimum_required(VERSION 3.22)
project(RoboDynaCudaApiTiming LANGUAGES CXX)
find_package(CUDAToolkit REQUIRED)
add_library(robo_dyna_cuda_api_timing SHARED Timing.cpp CudaApi.cpp Report.cpp)
target_compile_features(robo_dyna_cuda_api_timing PRIVATE cxx_std_17)
target_include_directories(robo_dyna_cuda_api_timing PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(robo_dyna_cuda_api_timing PRIVATE dl pthread)
target_link_options(robo_dyna_cuda_api_timing PRIVATE -Wl,-z,now)
set_target_properties(robo_dyna_cuda_api_timing PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
option(ROBO_DYNA_CUDA_API_TIMING_TESTS "CPU-only fake-runtime interposition checks" ON)
if(ROBO_DYNA_CUDA_API_TIMING_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_library(robo_dyna_fake_cudart SHARED tests/FakeCudart.cpp)
  target_compile_features(robo_dyna_fake_cudart PRIVATE cxx_std_17)
  target_include_directories(robo_dyna_fake_cudart PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
  set_target_properties(robo_dyna_fake_cudart PROPERTIES OUTPUT_NAME cudart SOVERSION 13)
  add_executable(robo_dyna_cuda_api_probe tests/Probe.cpp)
  target_compile_features(robo_dyna_cuda_api_probe PRIVATE cxx_std_17)
  target_include_directories(robo_dyna_cuda_api_probe PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
  target_link_libraries(robo_dyna_cuda_api_probe PRIVATE robo_dyna_fake_cudart pthread)
  add_test(NAME cuda_api_timing_fake_runtime COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tests/check.py"
    "$<TARGET_FILE:robo_dyna_cuda_api_timing>" "$<TARGET_FILE:robo_dyna_cuda_api_probe>")
  set_tests_properties(cuda_api_timing_fake_runtime PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE PROCESSORS 1)
endif()
