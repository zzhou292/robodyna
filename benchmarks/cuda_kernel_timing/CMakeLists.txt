cmake_minimum_required(VERSION 3.22)
project(RoboDynaCudaKernelTiming LANGUAGES CXX)
find_package(CUDAToolkit 13 REQUIRED)
if(NOT CUDAToolkit_VERSION VERSION_LESS 14)
  message(FATAL_ERROR "Internal kernel-handle ABI is verified for CUDA 13; review a new major version before use")
endif()
add_library(robo_dyna_cuda_kernel_timing SHARED Runtime.cpp Registry.cpp Timing.cpp Report.cpp)
target_compile_features(robo_dyna_cuda_kernel_timing PRIVATE cxx_std_17)
target_include_directories(robo_dyna_cuda_kernel_timing PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
target_link_libraries(robo_dyna_cuda_kernel_timing PRIVATE dl pthread)
target_link_options(robo_dyna_cuda_kernel_timing PRIVATE -Wl,-z,now)
set_target_properties(robo_dyna_cuda_kernel_timing PROPERTIES
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
option(ROBO_DYNA_CUDA_KERNEL_TIMING_TESTS "CPU-only fake runtime tests" ON)
if(ROBO_DYNA_CUDA_KERNEL_TIMING_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_library(robo_dyna_kernel_fake_cudart SHARED tests/FakeCudart.cpp)
  target_compile_features(robo_dyna_kernel_fake_cudart PRIVATE cxx_std_17)
  target_include_directories(robo_dyna_kernel_fake_cudart PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
  target_link_options(robo_dyna_kernel_fake_cudart PRIVATE
    "-Wl,--version-script=${CMAKE_CURRENT_LIST_DIR}/tests/cudart.map")
  set_target_properties(robo_dyna_kernel_fake_cudart PROPERTIES OUTPUT_NAME cudart SOVERSION 13)
  add_executable(robo_dyna_kernel_probe tests/Probe.cpp)
  target_compile_features(robo_dyna_kernel_probe PRIVATE cxx_std_17)
  target_include_directories(robo_dyna_kernel_probe PRIVATE "${CUDAToolkit_INCLUDE_DIRS}")
  target_link_libraries(robo_dyna_kernel_probe PRIVATE robo_dyna_kernel_fake_cudart pthread)
  add_test(NAME cuda_kernel_timing_fake_runtime COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_LIST_DIR}/tests/check.py" "$<TARGET_FILE:robo_dyna_cuda_kernel_timing>"
    "$<TARGET_FILE:robo_dyna_kernel_probe>")
  set_tests_properties(cuda_kernel_timing_fake_runtime PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE PROCESSORS 1)
  add_test(NAME cuda_kernel_timing_imports COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_LIST_DIR}/tests/check_imports.py")
  set_tests_properties(cuda_kernel_timing_imports PROPERTIES TIMEOUT 10 RUN_SERIAL TRUE PROCESSORS 1)
endif()
