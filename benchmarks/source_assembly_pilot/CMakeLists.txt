cmake_minimum_required(VERSION 3.22)
project(RoboDynaSourceAssemblyPilot LANGUAGES CXX)
enable_testing()
find_package(Chrono REQUIRED CONFIG)
find_package(GTest REQUIRED)
get_filename_component(robo_pilot_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${robo_pilot_root}/output" output)
include("${CMAKE_CURRENT_LIST_DIR}/SourceAssemblyPilot.cmake")
add_executable(robo_dyna_source_assembly_pilot_check tests/ComparisonTest.cpp)
target_link_libraries(robo_dyna_source_assembly_pilot_check PRIVATE robo_dyna_source_assembly_pilot GTest::gtest_main)
target_compile_options(robo_dyna_source_assembly_pilot_check PRIVATE -fno-fast-math -ffp-contract=off)
add_test(NAME source_assembly_pilot COMMAND robo_dyna_source_assembly_pilot_check)
set_tests_properties(source_assembly_pilot PROPERTIES RUN_SERIAL TRUE PROCESSORS 1 TIMEOUT 60)

set(ROBO_DYNA_PILOT_REFERENCE "" CACHE PATH "Optional fixed1024/65frame original assembly comparison fixture")
if(ROBO_DYNA_PILOT_REFERENCE)
  set_tests_properties(source_assembly_pilot PROPERTIES ENVIRONMENT "ROBO_DYNA_PILOT_REFERENCE=${ROBO_DYNA_PILOT_REFERENCE}")
endif()
